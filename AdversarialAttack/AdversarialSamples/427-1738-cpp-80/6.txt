vector<int> g[100];
class Solution {
public:
    int maximal_network_rank(int n, vector<vector<int>>& roads) {
        for(int i = 0; i < n; i += 1) g[i].clear();
        for(int i = 0; i < roads.size(); i += 1){
            g[roads[i][0]].push_back(i);
            g[roads[i][1]].push_back(i);
        }
        int ans = 0;
        for(int i = 0; i < n; i += 1)
            for(int j = i + 1; j < n; j += 1){
                int pans = 0;
                for(int x = 0, y = 0; x < g[i].size() or y < g[j].size();){
                    if(x == g[i].size()){
                        pans += 1;
                        y += 1;
                    }
                    else if(y == g[j].size()){
                        pans += 1;
                        x += 1;
                    }
                    else if(g[i][x] < g[j][y]){
                        pans += 1;
                        x += 1;
                    }
                    else if(g[i][x] > g[j][y]){
                        pans += 1;
                        y += 1;
                    }
                    else{
                        x += 1;
                        y += 1;
                        pans += 1;
                    }
                }
                ans = max(ans, pans);
            }
        return ans;
    }
};
