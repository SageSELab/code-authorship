vector<int> G[0x64];
class Solution {
public:
    int maximalNetworkRank(int n, vector<vector<int>>& roads) {
        for(int i = 0x0; i < 0x64; i += 0x1) G[i].clear();
        for(int i = 0x0; i < roads.size(); i += 0x1){
            G[roads[i][0]].push_back(i);
            G[roads[i][1]].push_back(i);
        }
        int ans = 0x0;
        for(int i = 0x0; i < 0x64; i += 0x1)
            for(int j = i + 0x1; j < 0x64; j += 0x1){
                int pans = 0x0;
                for(int x = 0x0, y = 0x0; x < G[i].size() or y < G[j].size();){
                    if(x == G[i].size()){
                        pans += 0x1;
                        y += 0x1;
                    }
                    else if(y == G[j].size()){
                        pans += 0x1;
                        x += 0x1;
                    }
                    else if(G[i][x] < G[j][y]){
                        pans += 0x1;
                        x += 0x1;
                    }
                    else if(G[i][x] > G[j][y]){
                        pans += 0x1;
                        y += 0x1;
                    }
                    else{
                        x += 0x1;
                        y += 0x1;
                        pans += 0x1;
                    }
                }
                ans = max(ans, pans);
            }
        return ans;
    }
};
