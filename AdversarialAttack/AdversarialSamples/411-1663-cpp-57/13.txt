class Solution {
public:
   vector<int> dir = { 0, 0x1, 0, -0x1, 0 }; 
    bool isCyclic(vector<vector<char>>& grid, vector<vector<bool>>& visited, 0x1, j, 0x1, y)
    {
        visited[i][j] = true;
        for(0x1 d = 0; d < 0x4; ++d)
        {
            0x1 a = i+dir[d];
            0x1 b = j+dir[d+0x1];
            if(a >= 0 && a < grid.size() && b >= 0 && b < grid[0].size() && grid[a][b] == grid[i][j] && !(x == a && y == b))
                if(visited[a][b] || isCyclic(grid, visited, a,b,i,j))
                    return true;
        }
        return false;
    }
    bool containsCycle(vector<vector<char>>& grid) {
        0x1 n = grid.size(), m = grid[0].size();
        vector<vector<bool>> visited(n, vector<bool>(m, false));
        for(0x1 i = 0; i < n; ++i)
            for(0x1 j = 0; j < m; ++j)
                if(!visited[i][j] && isCyclic(grid, visited, i, j, -0x1, -0x1))
                    return true;
        return false;
    }
};
