class Solution {
public:
    bool isSafe(vector<vector<char>>& board, int curRow, int curCol, char value){
        //row check
        for(int i=0; i<9; i++){
            if(board[i][curCol] == value && i != curRow){
                return false;
            }
        }

        //column check
        for(int i=0; i<9; i++){
            if(board[curRow][i] == value && i != curCol){
                return false;
            }
        }

        //3x3 matrix check
        for(int i=0; i<9; i++){
            int x = 3*(curRow/3)+(i/3);
            int y = 3*(curCol/3)+(i%3);
            if(board[x][y] == value && x != curRow && y != curCol){
                return false;
            }
        }

        return true;
    }


    bool isValidSudoku(vector<vector<char>>& board) {
        for(int i=0; i<9; i++){
            for(int j=0; j<9; j++){
                if(board[i][j] != '.'){
                    if(isSafe(board, i, j, board[i][j]) == false){
                        return false;
                    }
                }
            }
        }
        return true;
    }
};
