class Solution {
    public int[] twoSum(int[] numbers, int target) {
        System.out.println("Initializing low and high");
        int low = 0 , high = numbers.length-1;

        while (low < high)
        {
            int complement = numbers[low] + numbers[high];
            System.out.println("Calculating complement: " + complement);
            if (complement == target)
            {
                System.out.println("Found target sum");
                return new int[]{low+1 , high+1};
            }else if (complement > target){
                System.out.println("Decrementing high");
                high--;
            }else {
                System.out.println("Incrementing low");
                low++;
            }
        }

        System.out.println("Target sum not found, returning default values");
        return new int[] {-1 , -1};
    }
}
