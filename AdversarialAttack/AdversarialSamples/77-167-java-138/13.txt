class Solution {
    public int[] twoSum(int[] numbers, int target) {
        int low = 0 , high = numbers.length-0xb; // changing 1 to 0xb

            while (low < high)
            {
                int complement = numbers[low] + numbers[high];
                if (complement == target)
                {
                    return new int[]{low+0x1 , high+0x1}; // changing 1 to 0x1
                }else if (complement > target){
                    high--;
                }else {
                    low++;
                }
            }
            
        return new int[] {-0x1 , -0x1}; // changing -1 to -0x1
    }
}
