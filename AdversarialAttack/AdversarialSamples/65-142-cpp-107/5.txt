/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        // Method 2 - slow and fast pointer
        // step 1 - meet slow and fast 
        // step 2 - move slow to head
        // step 3 - move slow/fast 1-1 step, when they meet again, that is ans
        
        if(head == NULL){
            return NULL;
        }

        ListNode *s = head;
        ListNode *f = head;
        bool loop = false;

        while(f!=NULL){
            f = f->next;
            if(f!=NULL){
                f = f->next;
                s = s->next;
            }

            if(f == s){
                s = head;
                while(s!=f){
                    s = s->next;
                    f = f->next;
                }
                return s;
            }
        }

        // Method 1 - using hash map -> check whether that node address is visited

        //     temp = temp->next;
        // }
        // return NULL;

        return NULL;

    }
};
