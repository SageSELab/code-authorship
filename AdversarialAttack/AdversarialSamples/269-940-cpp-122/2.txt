class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<int,int> m; 
        cout << "m map initialized\n";

        int start = 0; 
        cout << "start variable initialized\n";

        int c = 0; 
        cout << "c variable initialized\n";

        int u = 0;  
        cout << "u variable initialized\n";

        int ma = -1; 
        cout << "ma variable initialized\n";

        for(int i=0; i<fruits.size(); i++){
            cout << "Loop iteration: " << i << "\n";

            if(m[fruits[i]] == 0){
                cout << "fruits[" << i << "] first encountered\n";
                
                m[fruits[i]]++; 
                cout << "Incrementing m[" << fruits[i] << "]\n";
                
                c++; 
                cout << "Incrementing c\n";
                
                u++; 
                cout << "Incrementing u\n";
                
                if( u > 2){
                    cout << "More than 2 unique fruits, adjusting start and count\n";
                    for(int j=start; j<=i; j++){
                        m[fruits[j]]--; 
                        cout << "Decrementing m[" << fruits[j] << "]\n";
                        
                        c--; 
                        cout << "Decrementing c\n";
                        
                        if(m[fruits[j]] == 0) {
                            u--; 
                            cout << "Decrementing u\n";
                            
                            start = j+1; 
                            cout << "Updating start position\n";
                            
                            break; 
                        }
                    }
                }
            }
            else {
                m[fruits[i]]++; 
                cout << "Incrementing m[" << fruits[i] << "]\n";
                
                c++; 
                cout << "Incrementing c\n";
            }

            if(ma < c) {
                ma = c; 
                cout << "Updating ma\n";
            }
        }

        return ma; 
    }
};
