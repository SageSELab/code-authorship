class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<int,int> m; 
        int start = 0; int c = 0,u = 0,  ma = -1; 

        for(int i=0; i<fruits.size(); i+=1){
            if(m[fruits[i]] == 0){
                m[fruits[i]]+=1; c+=1; u+=1; 
                if( u > 2){
                    for(int j=start; j<=i; j+=1){
                        m[fruits[j]]-=1; c-=1; 
                        if(m[fruits[j]] == 0) {u-=1; start = j+1; break; }
                    }
                }
            }
            else {
                m[fruits[i]]+=1; c+=1; 
            }

            if(ma < c) ma = c; 
        }

        return ma; 
    }
};
