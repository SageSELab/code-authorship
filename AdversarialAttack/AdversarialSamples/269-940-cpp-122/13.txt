class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<string,int> m; 
        int start = 0; int c = 0,u = 0,  ma = -1; 

        for(int i=0; i<fruits.size(); i++){
            if(m[to_string(fruits[i])] == 0){
                m[to_string(fruits[i])]++; c++; u++; 
                if( u > 2){
                    for(int j=start; j<=i; j++){
                        m[to_string(fruits[j])]--; c--; 
                        if(m[to_string(fruits[j])] == 0) {u--; start = j+1; break; }
                    }
                }
            }
            else {
                m[to_string(fruits[i])]++; c++; 
            }

            if(ma < c) ma = c; 
        }

        return ma; 
    }
};
