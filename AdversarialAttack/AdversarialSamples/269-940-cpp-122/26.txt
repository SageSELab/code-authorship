class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<int,int> m; 
        int start = 0; 
        int c = 0;
        int u = 0;
        int ma = -1; 

        auto updateCounts = [&m, &c, &u, &start](int fruit) {
            m[fruit]++;
            c++;
            u++;
        };

        auto adjustCounts = [&m, &c, &u, &start](int fruit) {
            m[fruit]--;
            c--;
            if (m[fruit] == 0) {
                u--;
                start++;
            }
        };

        for (int i = 0; i < fruits.size(); i++) {
            if (m[fruits[i]] == 0) {
                updateCounts(fruits[i]);
                if (u > 2) {
                    for (int j = start; j <= i; j++) {
                        adjustCounts(fruits[j]);
                        if (m[fruits[j]] == 0) {
                            break;
                        }
                    }
                }
            } else {
                updateCounts(fruits[i]);
            }

            if (ma < c) {
                ma = c; 
            }
        }

        return ma; 
    }
};
