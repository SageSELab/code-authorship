class Solution:
    def calculate(self, s):
        def update(op, v):
            switcher = {
                "+": stack.append(v),
                "-": stack.append(-v),
                "*": stack.append(stack.pop() * v),           #for BC II and BC III
                "/": stack.append(int(stack.pop() / v)         #for BC II and BC III
            }
            switcher.get(op)

        it, num, stack, sign = 0, 0, [], "+"
        
        while it < len(s):
            if s[it].isdigit():
                num = num * 10 + int(s[it])
            elif s[it] in "+-*/":
                update(sign, num)
                num, sign = 0, s[it]
            elif s[it] == "(":                                        # For BC I and BC III
                num, j = self.calculate(s[it + 1:])
                it = it + j
            elif s[it] == ")":                                        # For BC I and BC III
                update(sign, num)
                return sum(stack), it + 1
            it += 1
        update(sign, num)
        return sum(stack)
