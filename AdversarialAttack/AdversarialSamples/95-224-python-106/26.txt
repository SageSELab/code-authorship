class Solution:
    def calculate(self, s):
        def update(op, v):
            if op == "+": stack.append(v)
            if op == "-": stack.append(-v)
            if op == "*": stack.append(stack.pop() * v)            # for BC II and BC III
            if op == "/": stack.append(int(stack.pop() / v))        # for BC II and BC III

        def evaluate_expression(expression):
            it, num, stack, sign = 0, 0, [], "+"

            while it < len(expression):
                if expression[it].isdigit():
                    num = num * 10 + int(expression[it])
                elif expression[it] in "+-*/":
                    update(sign, num)
                    num, sign = 0, expression[it]
                elif expression[it] == "(":                           # For BC I and BC III
                    num, j = self.calculate(expression[it + 1:])
                    it = it + j
                elif expression[it] == ")":                           # For BC I and BC III
                    update(sign, num)
                    return sum(stack), it + 1
                it += 1
            update(sign, num)
            return sum(stack)

        return evaluate_expression(s)
