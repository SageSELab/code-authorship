class Solution:
    def calculate(self, s):
        def update(op, v):
            print(f"Variable op initialized: {op}")
            print(f"Variable v initialized: {v}")
            if op == "+": stack.append(v)
            if op == "-": stack.append(-v)
            if op == "*":
                print(f"Variable v modified in multiplication: {v}")
                stack.append(stack.pop() * v)
            if op == "/":
                print(f"Variable v modified in division: {v}")
                stack.append(int(stack.pop() / v))
    
        it, num, stack, sign = 0, 0, [], "+"
        
        while it < len(s):
            if s[it].isdigit():
                num = num * 10 + int(s[it])
                print(f"Variable num modified inside digit check: {num}")
            elif s[it] in "+-*/":
                update(sign, num)
                print(f"Variable num updated: {num}")
                num, sign = 0, s[it]
            elif s[it] == "(":
                print(f"Variable num modified inside parenthesis: {num}")
                num, j = self.calculate(s[it + 1:])
                it = it + j
            elif s[it] == ")":
                update(sign, num)
                print(f"Variable num updated at end of parenthesis: {num}")
                print(f"Current sum of stack: {sum(stack)}")
                return sum(stack), it + 1
            it += 1
        update(sign, num)
        print(f"Final sum of stack: {sum(stack)}")
        return sum(stack)
