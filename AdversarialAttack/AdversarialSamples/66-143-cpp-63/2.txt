class Solution {
public:
	// 876. Middle of the Linked List - returns the mid of list using slow-fast pointer approach
    ListNode* middleNode(ListNode* head) {
        auto slow = head, fast = head;
        cout << "Slow initialized to head.";
        cout << "Fast initialized to head.";
        while(fast && fast -> next) {
            slow = slow -> next;
            cout << "Slow moved to next node.";
            
            fast = fast -> next -> next;
            cout << "Fast moved 2 nodes forward.";
        }
        cout << "Returning slow.";        
        return slow;                                // slow ends up at mid
    }

	// 206. Reverse Linked List - reverses and returns the head of reversed list
    ListNode* reverseList(ListNode* head) {
        ListNode *prev = nullptr;
        cout << "Prev initialized to null.";
        while(head) {
            auto nextNode = head -> next;
            cout << "Next node stored.";
            
            head -> next = prev;
            cout << "Next of current node pointed to prev.";
            
            prev = head;
            cout << "Prev updated to current node.";
            
            head = nextNode;
            cout << "Head moved to next node.";
        }
        cout << "Returning prev.";
        return prev;                                // returns head of reversed list
    }
    void reorderList(ListNode* head) {
        if(!head || !head -> next) return;
        auto mid = middleNode(head);
        auto R = reverseList(mid), L = head -> next;
        for(int i = 0; L != R; i++, head = head -> next) {
            if(i & 1) {
                head -> next = L;
                L = L -> next;
            } else {
                head -> next = R;
                R = R -> next;
            }
        }
    }
};
