class Solution:
    def suggestedProducts(self, P, W):
        print("Initializing P with value:", P)
        P.sort()
        print("Sorted P:", P)
        beg, end, ans = 0, len(P) - 1, []
        print("Initialized beg, end, ans with values:", beg, end, ans)

        for i, c in enumerate(W):
            print("Iteration", i)
            while beg <= end and (len(P[beg]) <= i or P[beg][i] != c):
                print("Incrementing beg")
                beg += 1
                
            while beg <= end and (len(P[end]) <= i or P[end][i] != c):
                print("Decrementing end")
                end -= 1
                
            ans.append(P[beg:min(end+1, beg+3)])
            print("Appending to ans:", P[beg:min(end+1, beg+3)])

        return ans
