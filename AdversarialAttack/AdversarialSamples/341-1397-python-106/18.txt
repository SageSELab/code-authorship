class Solution:
    def suggestedProducts(self, P, W):
        P.sort()
        beg, end, ans = 0, len(P) - 1, []

        for i, c in enumerate(W):
            for _ in range(end - beg + 1):
                if (len(P[beg]) > i and P[beg][i] == c):
                    break
                beg += 1
                
            for _ in range(end - beg + 1):
                if (len(P[end]) > i and P[end][i] == c):
                    break
                end -= 1
                
            ans.append(P[beg:min(end+1, beg+3)])

        return ans
