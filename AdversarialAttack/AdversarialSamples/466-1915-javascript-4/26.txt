/**
 * @param {string} s1
 * @param {string} s2
 * @return {boolean}
 */
var areAlmostEqual = function(s1, s2) {
    const getDifferentIndexes = (s1, s2) => {
        let indexes = [];
        for (let i = 0; i < s1.length; i++) {
            if (s1[i] !== s2[i]) {
                indexes.push(i);
            }
        }
        return indexes;
    };

    const indexes = getDifferentIndexes(s1, s2);

    if ((indexes.length === 2 && s1[indexes[0]] === s2[indexes[1]] && s1[indexes[1]] === s2[indexes[0]]) || indexes.length === 0) {
        return true;
    }
    return false;
};
