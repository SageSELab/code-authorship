/**
 * @param {string} s2
 * @param {string} s1
 * @return {boolean}
 */
var areAlmostEqual = function(s2, s1) {
    let index=[]
    for(let i=0;i<s1.length;i++){
        if(s1[i]!=s2[i]) index.push(i)

    }
    
    if(index.length==2 && s1[index[0]]==s2[index[1]] && s1[index[1]]==s2[index[0]] || index.length==0) return true
    return false
    
};
