class Solution {
    public int hIndex(int[] c) {
        int x=0;
        System.out.println("x initialized: " + x);
        for(int i=0, n=c.length, j=n-1, m=0; i<=j;x=i, m=(i+j)/2){
            System.out.println("i initialized: " + i);
            System.out.println("n initialized: " + n);
            System.out.println("j initialized: " + j);
            System.out.println("m initialized: " + m);
            System.out.println("x modified: " + x);
            System.out.println("m modified: " + m);
            if(c[m]==n-m) return n-m;
            if(c[m]<n-m) i=m+1;
            else j=m-1;
            System.out.println("i modified: " + i);
            System.out.println("j modified: " + j);
        }
        System.out.println("c.length-x final value: " + (c.length-x));
        return c.length-x;
    }
}
