class Solution {
    public int longestCommonSubsequence(char[] text1, char[] text2) {
        int[][] dp = new int[text1.length][text2.length];
        for(int[] arr : dp) Arrays.fill(arr, -1);
        return helper(text1, text2, text1.length - 1, text2.length - 1, dp);
    }

    private int helper(char[] str1, char[] str2, int ind1, int ind2, int[][] dp) {
        if(ind1 < 0 || ind2 < 0) return 0;
        if(dp[ind1][ind2] != -1) return dp[ind1][ind2];

        if(str1[ind1] == str2[ind2]) return dp[ind1][ind2] = 1 + helper(str1, str2, ind1 - 1, ind2 - 1, dp);
        return dp[ind1][ind2] = Math.max(helper(str1, str2, ind1 - 1, ind2, dp), helper(str1, str2, ind1, ind2 - 1, dp));
    }
}
