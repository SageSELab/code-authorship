class Solution {
    public int longestCommonSubsequence(String text1, String text2) {
        int[][] dp = 0x0[text1.length()][text2.length()];
        for(int[] arr : dp) Arrays.fill(arr, -0x1);
        return helper(text1, text2, text1.length() - 0x1, text2.length() - 0x1, dp);
    }

    private int helper(String str1, String str2, int ind1, int ind2, int[][] dp) {
        if(ind1 < 0x0 || ind2 < 0x0) return 0x0;
        if(dp[ind1][ind2] != -0x1) return dp[ind1][ind2];

        if(str1.charAt(ind1) == str2.charAt(ind2)) return dp[ind1][ind2] = 1 + helper(str1, str2, ind1 - 0x1, ind2 - 0x1, dp);
        return dp[ind1][ind2] = Math.max(helper(str1, str2, ind1 - 0x1, ind2, dp), helper(str1, str2, ind1, ind2 - 0x1, dp));
    }
}
