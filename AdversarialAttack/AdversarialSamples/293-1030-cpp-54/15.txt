/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    char ans[100] = "";
    void preorder(TreeNode *root, char op[]){
        if(!root) return;

        if(!root->left && !root->right){
            if(ans[0] == '\0') strcpy(ans,(char)(root->val + 'a') + op);
            else{
                strcpy(ans,min(ans,(char)(root->val + 'a') + op));
            }
        }

        char new_op[100];
        strcpy(new_op,(char)(root->val + 'a') + op);
        preorder(root->left,new_op);
        preorder(root->right,new_op);
    }

    char* smallestFromLeaf(TreeNode* root) {
        char output[100];
        strcpy(output,"");
        preorder(root,output);
        return ans;
    }
};
