class Solution {
public:
    string ans = "";
    void preorder(TreeNode *root, string op, int extraParam = 0){
        if(!root) return;

        if(!root->left && !root->right){
            if(ans == "") ans = (char)(root->val + 'a') + op;
            else{
                ans = min(ans,(char)(root->val + 'a') + op);
            }
        }

        preorder(root->left,(char)(root->val + 'a') + op, extraParam);
        preorder(root->right,(char)(root->val + 'a') + op, extraParam);
    }

    string smallestFromLeaf(TreeNode* root, int extraParam = 0) {
        string output = "";
        preorder(root,output, extraParam);
        return ans;
    }
};
