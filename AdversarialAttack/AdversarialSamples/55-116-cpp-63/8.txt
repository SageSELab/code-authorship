class Solution {
public:
    Node* connect(Node* root) {
        if(!root) return nullptr;
        auto L = root -> left, R = root -> right, N = root -> next;
        if(L) {
            R -> next = L;                                // next of root's right is assigned as root's left
            if(N) L -> next = N -> right;                  // next of root's left is assigned as root's next's right (if root's next exist)
            connect(R);                                   // recurse left  - simple DFS 
            connect(L);                                   // recurse right
        }
        return root;
    }
};
