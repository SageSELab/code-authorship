class Solution {
public:
    int majorityElement(vector<int>& nums) {
        unordered_map<int, int> m;
        
        for(int i = 0; i < nums.size(); i++){
            m[nums[i]]++;
        }
        
        return findMajorityElement(m, nums.size());
    }
    
    int findMajorityElement(unordered_map<int, int>& m, int n) {
        n = n/2;
        for(auto x: m){
            if(x.second > n){
                return x.first;
            }
        }
        return 0;
    }
};
