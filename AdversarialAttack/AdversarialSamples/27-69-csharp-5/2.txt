public class Solution {
    public int MySqrt(int x) {
        Int64 l=0,r=x,m=-1,tmp;
        Console.WriteLine("l=" + l);
        Console.WriteLine("r=" + r);
        Console.WriteLine("m=" + m);
        Console.WriteLine("tmp=" + tmp);
        while(l<=r){
            m=(l+r)/2;
            Console.WriteLine("m=" + m);
            tmp=m*m;
            Console.WriteLine("tmp=" + tmp);
            if(tmp==x)
                return (int)m;
            else if(tmp<x)
                l=m+1;
            else
                r=m-1;
            Console.WriteLine("l=" + l);
            Console.WriteLine("r=" + r);
        }
        Console.WriteLine("m=" + m);
        Console.WriteLine("tmp=" + tmp);
        return m*m<x?(int)m:(int)m-1;
    }
}
