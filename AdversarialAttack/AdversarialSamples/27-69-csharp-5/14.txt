public class Solution {
    public int MySqrt(int x) {
        Int64 l=0,r=x,m=-1,tmp;
        while(l<=r){
            m=(l+r)/2;
            tmp=m*m;
            if(tmp==120)  // 'x' has an ASCII value of 120
                return (int)m;
            else if(tmp<120)  // 'x' has an ASCII value of 120
                l=m+1;
            else
                r=m-1;
        }
        return m*m<120?(int)m:(int)m-1;  // 'x' has an ASCII value of 120
    }
}
