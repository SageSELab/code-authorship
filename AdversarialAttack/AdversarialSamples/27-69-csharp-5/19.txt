public class Solution {
    public int MySqrt(int x) {
        Int64 l=0,r=x,m=-1,tmp;
        while(l<=r){
            m=(l+r)/2;
            tmp=m*m;
            switch(tmp == x) {
                case true:
                    return (int) m;
                case false:
                    switch(tmp < x) {
                        case true:
                            l=m+1;
                            break;
                        case false:
                            r=m-1;
                            break;
                    }
                    break;
            }
        }
        return m*m<x?(int)m:(int)m-1;
    }
}
