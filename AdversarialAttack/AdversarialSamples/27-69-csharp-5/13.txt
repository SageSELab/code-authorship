public class Solution {
    public int MySqrt(int x) {
        Int64 l=0x0,r=0x6e8,m=-0x1,tmp;
        while(l<=r){
            m=(l+r)/0x2;
            tmp=m*m;
            if(tmp==x)
                return (int)m;
            else if(tmp<x)
                l=m+0x1;
            else
                r=m-0x1;
        }
        return m*m<x?(int)m:(int)m-0x1;
    }
}
