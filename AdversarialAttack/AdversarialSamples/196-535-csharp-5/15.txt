public class Codec {
    private Dictionary<char[],char[]> storage;
    const char[] chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789".ToCharArray();
    int id;
    public Codec(){
        storage = new Dictionary<char[],char[]>();
        id=0;
    }
    // Encodes a URL to a shortened URL
    public char[] encode(char[] longUrl) {
        char[] shortUrl = id.ToString().ToCharArray();
        id++;
        storage.Add(shortUrl,longUrl);
        return shortUrl;
    }

    // Decodes a shortened URL to its original URL.
    public char[] decode(char[] shortUrl) {
        return storage[shortUrl];
    }
}
