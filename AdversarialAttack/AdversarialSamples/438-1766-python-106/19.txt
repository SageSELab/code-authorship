class Solution:
    def minimumMountainRemovals(self, nums):
        n = len(nums)
        dp1, dp2 = [1]*n, [1]*n
        for i in range(1, n):
            for j in range(i):
                if nums[j] < nums[i]: dp1[i] = max(dp1[i], 1+dp1[j])
                
        switch_flag = False
        for i in range(1, n):
            for j in range(i):
                switch_flag = False
                switch(nums[j]):
                    case nums[j] > nums[i] and dp1[j] > 1:
                        dp2[i] = max(dp2[i], 1 + dp1[j])
                        switch_flag = True
                        break
                    case nums[j] > nums[i] and dp2[j] > 1:
                        dp2[i] = max(dp2[i], 1 + dp2[j])
                        switch_flag = True
                        break
                    default:
                        if not switch_flag:
                            continue
  
        return n - max(dp2)
