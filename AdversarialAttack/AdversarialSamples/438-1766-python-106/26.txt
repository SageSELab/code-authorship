class Solution:
    def minimumMountainRemovals(self, nums):
        
        def longest_increasing_seq(nums):
            n = len(nums)
            dp = [1]*n
            for i in range(1, n):
                for j in range(i):
                    if nums[j] < nums[i]:
                        dp[i] = max(dp[i], 1+dp[j])
            return dp
        
        def longest_decreasing_seq(nums):
            n = len(nums)
            dp = [1]*n
            for i in range(n-2, -1, -1):
                for j in range(i+1, n):
                    if nums[j] < nums[i]:
                        dp[i] = max(dp[i], 1+dp[j])
            return dp
        
        n = len(nums)
        inc_seq = longest_increasing_seq(nums)
        dec_seq = longest_decreasing_seq(nums)
        highest_mountain = 0
        for i in range(1, n-1):
            if inc_seq[i] > 1 and dec_seq[i] > 1:
                highest_mountain = max(highest_mountain, inc_seq[i] + dec_seq[i] - 1)
        return n - highest_mountain
