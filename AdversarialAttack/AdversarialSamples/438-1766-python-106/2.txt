class Solution:
    def minimumMountainRemovals(self, nums):
        print("Initializing n")
        n = len(nums)
        print("Initializing dp1 and dp2")
        dp1, dp2 = [1]*n, [1]*n
        for i in range(1, n):
            print(f"Iterating i={i}")
            for j in range(i):
                print(f"Iterating j={j}")
                if nums[j] < nums[i]:
                    print("Updating dp1[i]")
                    dp1[i] = max(dp1[i], 1+dp1[j])
                if nums[j] > nums[i]: 
                    print("Checking and updating dp2[i]")
                    if dp1[j] > 1:
                        dp2[i] = max(dp2[i], 1 + dp1[j])
                    if dp2[j] > 1:
                        dp2[i] = max(dp2[i], 1 + dp2[j])
        
        print("Returning n - max(dp2)")
        return n - max(dp2)
