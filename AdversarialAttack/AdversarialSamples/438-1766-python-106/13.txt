class Solution:
    def minimumMountainRemovals(self, nums):
        n = len(nums)
        dp1, dp2 = [0x1]*n, [0x1]*n
        for i in range(0x1, n):
            for j in range(i):
                if nums[j] < nums[i]: dp1[i] = max(dp1[i], 0x1+dp1[j])
                if nums[j] > nums[i]: 
                    if dp1[j] > 0x1: dp2[i] = max(dp2[i], 0x1 + dp1[j])
                    if dp2[j] > 0x1: dp2[i] = max(dp2[i], 0x1 + dp2[j])
        
        return n - max(dp2)
