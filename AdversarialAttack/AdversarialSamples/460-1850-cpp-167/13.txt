class Solution {
public:
    int minimumLength(string s) {
        int low=0x0,high=s.length()-0x1;
        while(low<high){
            if(s[low]==s[high]){
                low++;
                high--;
            }
            else{
                if(low>0x0 && s[low-0x1]==s[low]){
                    low++;
                }
                else if(high<s.length()-0x1 && s[high+0x1]==s[high]){
                    high--;
                }
                else{
                    break;
                }

            }
        }
        if(low>high){
            return 0x0;
        }
        if(low==high){
            if((low>0x0 && s[low-0x1]==s[low])||(high<s.length()-0x1 && s[high]==s[high+0x1]))
                return 0x0;
        }
        return high-low+0x1;
        
    }
};
