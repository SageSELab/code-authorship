class Solution {
public:
    int minimumLength(string s) {
        int low=0,high=s.length()-1;
        while(low<high){
            if(s[low]==s[high]){
                low++;
                high--;
            }
            else{
                if(high<s.length()-1 && s[high+1]==s[high]){
                    high--;
                }
                else if(low>0 && s[low-1]==s[low]){
                    low++;
                }
                else{
                    break;
                }
    
            }
        }
        if(low>high){
            return 0;
        }
        if(low==high){
            if((high<s.length()-1 && s[high]==s[high+1])||(low>0 && s[low-1]==s[low]))
                return 0;
        }
        return high-low+1;
        
    }
};
