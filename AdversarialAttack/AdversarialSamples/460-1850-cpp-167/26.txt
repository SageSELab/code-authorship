class Solution {
public:
    int expandPalindrome(string s, int &low, int &high) {
        while(low < high && s[low] == s[high]){
          low++;
          high--;
        }
    }

    int minimumLength(string s) {
        int low=0,high=s.length()-1;
        while(low < high){
            if(s[low]==s[high]){
                low++;
                high--;
            }
            else{
                if(low > 0 && s[low - 1] == s[low]){
                    low++;
                }
                else if(high < s.length() - 1 && s[high + 1] == s[high]){
                    high--;
                }
                else{
                    break;
                }
            }
        }

        if(low > high || (low == high && ((low > 0 && s[low - 1] == s[low]) || (high < s.length() - 1 && s[high] == s[high + 1])))){
            return 0;
        }
        
        return high - low + 1;
    }
};
