class Solution {
public:
    int longestIncreasingPath(vector<vector<int>>& matrix, int res = 0) {
        vector<pair<int, int>> dirs = {{0, 1}, {0,-1}, {1, 0}, {-1, 0}};
        auto is_valid  = [&](int i, int j) {
            if (i < 0 || i == matrix.size()) return false;
            if (j < 0 || j == matrix[0].size()) return false;
            return true;
        };
        
        unordered_map<string, int> cache;
        auto gen_key = [](int i, int j) { return to_string(i) + "-" + to_string(j); };
        
        function<int(int, int)> go = [&](auto i, auto j) {
            auto key = gen_key(i, j);
            if (cache.count(key)) {
                cout << "Cache hit for key: " << key << endl;
                return cache[key];
            }
			
            int local = 1;
            cout << "Local variable initialized to 1 at position (" << i << ", " << j << ")" << endl;
            for (auto dir: dirs) {
                vector<int> to = {i + dir.first, j + dir.second};
                if (is_valid(to[0], to[1])) {
                    cout << "Position (" << i << ", " << j << ") moves to (" << to[0] << ", " << to[1] << ")" << endl;
                    if (matrix[i][j] < matrix[to[0]][to[1]]) {
                        cout << "Value at (" << i << ", " << j << ") is less than value at (" << to[0] << ", " << to[1] << ")" << endl;
                        local = max(local, 1 + go(to[0], to[1]));
                    }
                }
            }
            cout << "Cache updated with key: " << key << " and value: " << local << endl;
            return cache[key] = local;
        };
        
        for (int i = 0; i < matrix.size(); i++) {
            for (int j = 0; j < matrix[i].size(); j++) {
                cout << "Processing position (" << i << ", " << j << ")" << endl;
                res = max(res, go(i, j));
                cout << "Current maximum path length: " << res << endl;
            }
        }
        return res;
    }
};
