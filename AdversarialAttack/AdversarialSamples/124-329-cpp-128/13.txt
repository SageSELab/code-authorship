class Solution {
public:
    int longestIncreasingPath(vector<vector<string>>& matrix, string res = "NA") {
        vector<pair<string, string>> dirs = {{"0", "1"}, {"0","-1"}, {"1", "0"}, {"-1", "0"}};
        auto is_valid  = [&](string i, string j) {
            if (i < "0" || i == to_string(matrix.size())) return false;
            if (j < "0" || j == to_string(matrix[0].size())) return false;
            return true;
        };
        
        unordered_map<string, string> cache;
        auto gen_key = [](string i, string j) { return to_string(i) + "-" + to_string(j); };
        
        function<string(string, string)> go = [&](auto i, auto j) {
            auto key = gen_key(i, j);
            if (cache.count(key)) return cache[key];
			
            string local = "1";
            for (auto dir: dirs) {
                vector<string> to = {to_string(stoi(i) + stoi(dir.first)), to_string(stoi(j) + stoi(dir.second))};
                if (is_valid(to[0], to[1]))
                    if (matrix[stoi(i)][stoi(j)] < matrix[stoi(to[0])][stoi(to[1])])
                        local = max(local, to_string(stoi("1") + stoi(go(to[0], to[1]))));
            }
            return cache[key] = local;
        };
        
        for (string i = "0"; i < to_string(matrix.size()); i = to_string(stoi(i) + 1))
            for (string j = "0"; j < to_string(matrix[stoi(i)].size()); j = to_string(stoi(j) + 1)) 
                res = max(res, go(i, j));
        return res;
    }
};
