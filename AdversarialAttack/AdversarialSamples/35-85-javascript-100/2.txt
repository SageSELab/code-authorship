function maximalRectangle(aa) {
    let m = aa.length;
    console.log('m initialized with value: ', m);
    let n = aa[0].length;
    console.log('n initialized with value: ', n);

    let heights = new Array(n + 1).fill(0);
    console.log('heights array initialized with values: ', heights);
    let maxArea = 0;
    console.log('maxArea initialized with value: ', maxArea);

    for (let r = 0; r < m; r++) {
        let cols = [];
        console.log('cols array initialized.');
        cols.top = () => cols[cols.length - 1];

        for (let c = 0; c <= n; c++) {
            if (c < n) {
                if (aa[r][c] === "1") heights[c]++;
                else heights[c] = 0;
                console.log('heights array modified: ', heights);
            }

            while (cols.length && heights[cols.top()] > heights[c]) {
                let height = heights[cols.pop()];
                let width = cols.length ? c - cols.top() - 1 : c;
                console.log('Height and width modified: ', height, width);
                maxArea = Math.max(maxArea, width * height);
                console.log('maxArea recalculated: ', maxArea);
            }
            cols.push(c);
            console.log('cols updated: ', cols);
        }
    }
    return maxArea;
}
