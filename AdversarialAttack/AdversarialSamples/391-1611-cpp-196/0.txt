class Solution {
public:
    unordered_map<string, int> nameSuffix;

    string addSuffix(string name) {
        string newName = "";
        do {
            newName = name + "(" + to_string(nameSuffix[name]) + ")";
            nameSuffix[name]++;
        } while (nameSuffix.find(newName) != nameSuffix.end());
        return newName;
    }
    
    string getFolderName(string name) {
        string finalName = name;
        if (nameSuffix.find(name) != nameSuffix.end()) {
            finalName = addSuffix(name);
        }
        nameSuffix[finalName]++;
        return finalName;
    }
    vector<string> getFolderNames(vector<string>& names) {
        vector<string> ans;
        for (auto name: names) {
            ans.push_back(getFolderName(name));
        }
        return ans;
    }
};
