class Solution {
public:
    unordered_map<string, int> nameSuffix; // Folder name -> Next available value for the suffix.

    string addSuffix(string name) {
        string newName = "";
        cout << "Initializing newName with an empty string" << endl;
        do {
            newName = name + "(" + to_string(nameSuffix[name]) + ")"; // Generate a name.
            cout << "Updating newName: " << newName << endl;
            nameSuffix[name]++; // Increase the count by one
            cout << "Updating nameSuffix[" << name << "]: " << nameSuffix[name] << endl;
        } while (nameSuffix.find(newName) != nameSuffix.end()); // Until we get a used name for the folder.
        cout << "Exiting the loop, current newName: " << newName << endl;
        return newName;
    }

    string getFolderName(string name) {
        string finalName = name; // If the folder name doesn't already exist, no need of renaming.
        cout << "Initializing finalName with: " << finalName << endl;
        if (nameSuffix.find(name) != nameSuffix.end()) { // If folder already exists -
            finalName = addSuffix(name); // Add suffix to the name.
            cout << "Executing addSuffix function, new finalName: " << finalName << endl;
        }
        nameSuffix[finalName]++; // Record that the name is already in use.
        cout << "Updating nameSuffix[" << finalName << "]: " << nameSuffix[finalName] << endl;
        return finalName; // Finally return the name for the folder.
    }

    vector<string> getFolderNames(vector<string>& names) {
        vector<string> ans; // Hold all names of the folders after renaming.
        cout << "Initializing ans vector" << endl;
        for (auto name : names) { // For each folder name -
            ans.push_back(getFolderName(name)); // Compute the name for the folder.
            cout << "Adding name to ans vector: " << ans.back() << endl;
        }
        cout << "Returning ans vector" << endl;
        return ans; // Return answer.
    }
};
