class Solution {
public:
    string tree2str(TreeNode* root) {
        stringstream ans;
        ans<<root->val; // Print statement: root->val initialized
        if (!root->right && !root->left) return ans.str(); // No modification to variables
        vector<TreeNode*> stack; // Print statement: stack initialized using vector
        if (root->right) stack.push_back(root->right); // Print statement: root->right value modified
        if (root->left) stack.push_back(root->left); // Print statement: root->left value modified
        else ans<<"()"; // Print statement: ans value modified
        while (!stack.empty()) {
            TreeNode* node = stack.back(); // Print statement: node initialized
            stack.pop_back(); // Print statement: stack modified
            if (!node) {
                ans<<')'; // Print statement: ans value modified
                continue;
            }
            ans<<'('<<node->val; // Print statement: ans value modified, node->val value accessed
            TreeNode* left = node->left, *right = node->right; // No modification to variables
            if (!left && !right) {
                ans<<')'; // Print statement: ans value modified
                continue;
            }
            stack.push_back(NULL); // Print statement: stack modified
            if (right) stack.push_back(right); // Print statement: right value modified
            if (left) stack.push_back(left); // Print statement: left value modified
            else ans<<"()"; // Print statement: ans value modified
        }
        return ans.str(); // Return statement
    }
};
