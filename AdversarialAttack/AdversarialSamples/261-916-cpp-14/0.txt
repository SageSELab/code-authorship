class Solution {
public:
    string decodeAtIndex(string s, int k) {
        long long n = 0;
        long long size = 0;

        for (char c : s) {
            if (isdigit(c)) {
                int digit = c - '0';
                size *= digit;
                n *= digit;
            } else {
                size++;
                n++;
            }
        }

        for (int i = s.size() - 1; i >= 0; i--) {
            char c = s[i];
            if (isdigit(c)) {
                int digit = c - '0';
                size /= digit;
                n /= digit;
                k %= n;
            } else {
                if (k == 0 || k == n) {
                    return string(1, c);
                }
                size--;
                n--;
            }
        }

        return "";
    }
};
