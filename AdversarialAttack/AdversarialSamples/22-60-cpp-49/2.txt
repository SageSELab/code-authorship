class Solution {
public:
    string getPermutation(int n, int k) {

        set<int> nums;
        cout << "nums set initialized"<<endl;
        vector<int> factorial(n+1, 1);
        cout << "factorial vector initialized with size " << n+1 <<endl;

        for (int i = 1; i <= n; i++) {
            cout << "Inserting " << i << " into set nums"<<endl;
            nums.insert(i);
            factorial[i] = i*factorial[i-1];
            cout << "factorial["<<i<<"] = " << i << "*" << "factorial["<<i-1<<"] = " << factorial[i] << endl;
        }

        --k; // Adjust k to 0-based index
        cout << "k adjusted to " << k << " (0-based index)" << endl;

        string perm;
        cout << "Empty string perm initialized" << endl;
        for (int i=n; i >= 1; i--) {
            cout << "Calculating index for i = " << i << endl;
            int index = k/factorial[i-1];
            cout << "Index calculated: " << index << endl;
            k%= factorial[i-1];
            cout << "k updated to " << k << endl;
            auto it=nums.begin();
            cout << "Iterator it initialized at beginning of set nums" << endl;
            advance(it, index); // Move iterator to the index
            cout << "Iterator it moved to index " << index << endl;
            perm+= to_string(*it);
            cout << "Adding *it to permutation: " << *it << endl;
            nums.erase(it);//O(log n) time
            cout << "Element erased from nums set" << endl;
        }
        return perm;
    }
};
