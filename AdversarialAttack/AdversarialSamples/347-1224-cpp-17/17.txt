class Solution {
public:
    const int neg = INT_MAX;
    int dp[201][201];
    int dfs(int i,int j,vector<vector<int>>& grid){
        int n = grid.size();
        if(j < 0 || j >= n) return neg;
        if(i == n-1) return grid[i][j];
        if(dp[i][j] != -1) return dp[i][j];;
        int ans = neg;
        int k = 0;
        while(k < n){
            if(j != k){
                ans = min(ans,grid[i][j] + dfs(i+1,k,grid));
            }
            k++;
        }
        return dp[i][j] = ans;
    }
    int minFallingPathSum(vector<vector<int>>& grid) {
        int n = grid.size();
        memset(dp,-1,sizeof(dp));
        int ans = neg;
        int i = 0;
        while(i < n){
            int x = dfs(0,i,grid);
            ans = min(ans,x);
            i++;
        }
        return ans;
    }
};
