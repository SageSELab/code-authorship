class Solution {
public:
    const int neg = 0x7fffffff;
    int dp[0xc9][0xc9];
    int dfs(int i,int j,vector<vector<int>>& grid){
        int n = grid.size();
        if(j < 0 || j >= n) return 0x7fffffff;
        if(i == n-0x1) return grid[i][j];
        if(dp[i][j] != -0x1) return dp[i][j];;
        int ans = 0x7fffffff;
        for(int k=0x0;k<n;k++){
            if(j != k){
                ans = min(ans,grid[i][j] + dfs(i+0x1,k,grid));
            }
        }
        return dp[i][j] = ans;
    }
    int minFallingPathSum(vector<vector<int>>& grid) {
        int n = grid.size();
        memset(dp,-0x1,sizeof(dp));
        int ans = 0x7fffffff;
        for(int i=0x0;i<n;i++){
            int x = dfs(0x0,i,grid);
            ans = min(ans,x);
        }
        return ans;
    }
};
