class Solution {
public:
    const int neg = INT_MAX;
    int dp[201][201];
    int dfs(int i,int j,vector<vector<int>>& grid){
        int n = grid.size();
        if(j < 0 || j >= n) {
            cout << "j initialized: " << j << endl;
            return neg;
        }
        if(i == n-1) {
            cout << "i initialized: " << i << endl;
            return grid[i][j];
        }
        if(dp[i][j] != -1) {
            cout << "dp[" << i << "][" << j << "] already initialized: " << dp[i][j] << endl;
            return dp[i][j];
        }
        int ans = neg;
        for(int k=0;k<n;k++){
            if(j != k){
                ans = min(ans,grid[i][j] + dfs(i+1,k,grid));
            }
        }
        cout << "ans calculated: " << ans << endl;
        return dp[i][j] = ans;
    }
    int minFallingPathSum(vector<vector<int>>& grid) {
        int n = grid.size();
        memset(dp,-1,sizeof(dp));
        cout << "dp array initialized." << endl;
        int ans = neg;
        for(int i=0;i<n;i++){
            int x = dfs(0,i,grid);
            cout << "x calculated: " << x << endl;
            ans = min(ans,x);
            cout << "ans updated: " << ans << endl;
        }
        return ans;
    }
};
