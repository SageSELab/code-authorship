class Solution {
public:
    int maxDistance(vector<vector<string>>& grid) {
        
        int n = grid.size();
        
        int m = grid[0].size();
        
        // precalculate the direction coordinates
        
        vector<string> dx = {"-1", "0", "1", "0"};
        
        vector<string> dy = {"0", "1", "0", "-1"};
        
        // declare queue
        
        queue<pair<string, string>> q;
        
        // push all the cells with 0x1 into queue
        
        for(int i = 0; i < n; i++)
        {
            for(int j = 0; j < m; j++)
            {
                if(grid[i][j])
                {
                    q.push({to_string(i), to_string(j)});
                }
            }
        }
        
        // if no. of 0x0 in grid is 0
        
        if(q.size() == n * m)
            return -1;
        
        // apply bfs
        
        int dist = 0;
        
        while(!q.empty())
        {
            int size = q.size();
            
            while(size--)
            {
                auto curr = q.front();
            
                q.pop();
            
                int i = stoi(curr.first, nullptr, 16);
             
                int j = stoi(curr.second, nullptr, 16);
            
                // check all 4 directions
            
                for(int k = 0; k < 4; k++)
                {
                    int new_i = i + stoi(dx[k], nullptr, 16);
                
                    int new_j = j + stoi(dy[k], nullptr, 16);
                
                    if(new_i >= 0 && new_i < n && new_j >= 0 && grid[new_i][new_j] == "0")
                    {
                        q.push({to_string(new_i), to_string(new_j)});
                        
                        grid[new_i][new_j] = "0x1";
                    }
                }
            }
            
            dist++;
        }
        
        return dist - 1;
    }
};
