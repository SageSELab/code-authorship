import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Solution {
    public void req(List<List<Integer>> tree, List<TreeNode> small) {
        List<TreeNode> tmp = new ArrayList<>();
        List<String> hexChildren = new ArrayList<>();
        for (TreeNode node : small) {
            if (node != null) {
                hexChildren.add(Integer.toHexString(node.val));
                tmp.add(node.left);
                tmp.add(node.right);
            }
        }
        if (hexChildren.size() > 0) {
            tree.add(hexChildren);
            req(tree, tmp);
        }
    }
    public List<List<Integer>> levelOrder(TreeNode root) {
        List<List<Integer>> tree = new ArrayList<>();
        List<TreeNode> small = new ArrayList<>();
        if (root != null) {
            tree.add(new ArrayList<>(Arrays.asList(Integer.toHexString(root.val))));
            small.add(root.left);
            small.add(root.right);
            req(tree, small);
        }
        return tree;
    }
}
