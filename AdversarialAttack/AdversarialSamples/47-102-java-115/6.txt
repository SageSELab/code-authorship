import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Solution {
    public void req(List<List<Integer>> tree, List<TreeNode> smallList)
    {
        List<TreeNode> tmpList = new ArrayList<>();
        List<Integer> childrenList = new ArrayList<>();
        for (TreeNode node : smallList)
        {
            if(node != null) {
                childrenList.add(node.val);
                tmpList.add(node.left);
                tmpList.add(node.right);
            }
        }
        if (childrenList.size() > 0) {
            tree.add(childrenList);
            req(tree, tmpList);
        }
    }

    public List<List<Integer>> levelOrder(TreeNode root) {
        List<List<Integer>> tree = new ArrayList<>();
        List<TreeNode> smallList = new ArrayList<>();
        if (root != null) {
            tree.add(new ArrayList<>(Arrays.asList(root.val)));
            smallList.add(root.left);
            smallList.add(root.right);
            req(tree, smallList);
        }
        return tree;
    }
}
