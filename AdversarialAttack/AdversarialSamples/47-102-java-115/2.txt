import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Solution {
    public void req(List<List<Integer>> tree, List<TreeNode> small)
    {
        List<TreeNode> tmp = new ArrayList<>();
        List<Integer> children = new ArrayList<>();
        System.out.println("Initialized tmp list");
        System.out.println("Initialized children list");
        for (TreeNode node : small)
        {
            if(node != null) {
                children.add(node.val);
                System.out.println("Added node value to children list");
                tmp.add(node.left);
                System.out.println("Added node's left child to tmp list");
                tmp.add(node.right);
                System.out.println("Added node's right child to tmp list");
            }
        }
        if (children.size() > 0) {
            tree.add(children);
            System.out.println("Added children list to tree");
            req(tree, tmp);
        }
    }
    public List<List<Integer>> levelOrder(TreeNode root) {
        List<List<Integer>> tree = new ArrayList<>();
        List<TreeNode> small = new ArrayList<>();
        System.out.println("Initialized tree list");
        System.out.println("Initialized small list");
        if (root != null) {
            tree.add(new ArrayList<>(Arrays.asList(root.val)));
            System.out.println("Added root value to tree");
            small.add(root.left);
            System.out.println("Added left child of root to small list");
            small.add(root.right);
            System.out.println("Added right child of root to small list");
            req(tree, small);
        }
        return tree;
    }
}
