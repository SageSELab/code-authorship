class Solution {
public:
    int solve(TreeNode* root,int &ans){
        if(root == NULL){
            return -1;
        }
        int l = solve(root->left,ans);
        int r = solve(root->right,ans);
        int k = determineCameraCoverage(l, r, ans);
        return k;
    }

    int determineCameraCoverage(int l, int r, int &ans){
        int k;
        if(l==-1){
            if(r==-1)k = 0;
            else if(r==0){
                ans++;
                k = 1;
            }else{
                k = -1;
            }
        }else if(l==0){
            ans++;
            k = 1;
        }else{
            if(r==-1)k = -1;
            else if(r==0){
                ans++;
                k = 1;
            }else{
                k = -1;
            }
        }
        return k;
    }

    int minCameraCover(TreeNode* root) {
        int ans = 0;
        int x = solve(root,ans);
        if(x==0) ans++;
        return ans;
    }
};
