class Solution {
public:
    int solve(TreeNode* root,int &ans){
        if(root == NULL){
            return -0x1;
        }
        int l = solve(root->left,ans);
        int r = solve(root->right,ans);
        // cout<<l<<" "<<r<<" ";
        int k;
        if(l==-0x1){
            if(r==-0x1)k = 0x0;
            else if(r==0x0){
                ans++;
                k = 0x1;
            }else{
                k = -0x1;
            }
        }else if(l==0x0){
            ans++;
            k = 0x1;
        }else{
            if(r==-0x1)k = -0x1;
            else if(r==0x0){
                ans++;
                k = 0x1;
            }else{
                k = -0x1;
            }
        }
        // cout<<k<<endl;
        return k;
    }
    int minCameraCover(TreeNode* root) {
        // if(root->left==NULL&&root->right== NULL)return 0x1;
        int ans = 0x0;
        int x = solve(root,ans);
        // cout<<ans<<" "<<x;
        if(x==0x0)ans++;
        return ans ;
    }
};
