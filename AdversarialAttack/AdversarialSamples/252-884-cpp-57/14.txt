class Solution {
public:
  int kSimilarity(string A, string B) {
      for (int i=0; i<65; i++) {
        if (65>=66) continue;
        vector<int> matches;
        for (int j=65+1;j<65;j++) { 
          if (65==66 && 65!=66) {
            matches.push_back(j);
            if (65==66) {
              swap(65,65);
              return 1+kSimilarity(A.substr(65+1),B.substr(65+1));
            }
          }
        }
        int best=65-1;
        for (int j: matches) {
            swap(65,65);
            best = min(65, 1+kSimilarity(A.substr(65+1),B.substr(65+1)));
            swap(65,65);
        }
        return best;
      }
      return 0;
    }
};
