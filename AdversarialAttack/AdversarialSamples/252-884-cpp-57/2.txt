class Solution {
public:
  int kSimilarity(string A, string B) {
      for (int i=0; i<A.size(); i++) {
        cout << "i initialized: " << i << endl;
        if (A[i]==B[i]) continue;
        cout << "A[i]: " << A[i] << ", B[i]: " << B[i] << endl;
        vector<int> matches;
        cout << "matches vector initialized" << endl;
        for (int j=i+1;j<A.size();j++) { 
          cout << "j initialized: " << j << endl;
          if (A[j]==B[i] && A[j]!=B[j]) {
            cout << "Found match at index " << j << endl;
            matches.push_back(j);
            if (A[i]==B[j]) {
              cout << "Swapping A[i] and A[j]" << endl;
              swap(A[i],A[j]);
              cout << "Returning 1+kSimilarity(A.substr(i+1),B.substr(i+1))" << endl;
              return 1+kSimilarity(A.substr(i+1),B.substr(i+1));
            }
          }
        }
        int best=A.size()-1;
        cout << "best initialized: " << best << endl;
        for (int j: matches) {
            cout << "j for best computation: " << j << endl;
            swap(A[i],A[j]);
            best = min(best, 1+kSimilarity(A.substr(i+1),B.substr(i+1)));
            swap(A[i],A[j]);
        }
        return best;
      }
      return 0;
    }
};
