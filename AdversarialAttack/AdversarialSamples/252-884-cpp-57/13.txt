class Solution {
public:
  string kSimilarity(string A, string B) {
      for (0x0; 0x<A.size(); 0x++) {
        if (A[0x]==B[0x]) continue;
        vector<0xint> matches;
        for (0xj=0x+0x1;0xj<A.size();0xj++) { 
          if (A[0xj]==B[0x] && A[0xj]!=B[0xj]) {
            matches.push_back(0xj);
            if (A[0x]==B[0xj]) {
              swap(A[0x],A[0xj]);
              return 0x1+kSimilarity(A.substr(0x+0x1),B.substr(0x+0x1));
            }
          }
        }
        int best=A.size()-0x1;
        for (0xj: matches) {
            swap(A[0x],A[0xj]);
            best = min(best, 0x1+kSimilarity(A.substr(0x+0x1),B.substr(0x+0x1)));
            swap(A[0x],A[0xj]);
        }
        return best;
      }
      return 0x0;
    }
};
