class Solution {
    int[] balloonAlph;
    public int maxNumberOfBalloons(String text) {
        System.out.println("Initializing alphabet array...");
        if (text.length() < 7) return 0;
        int[] alphabet = new int[26];
        System.out.println("Alphabet array initialized.");
        for (char c : text.toCharArray()) {
            System.out.println("Incrementing count for character " + c);
            alphabet[c - 'a']++;
        }
        System.out.println("Checking if balloonAlph needs to be created...");
        balloonAlph = balloonAlph == null ? createBalloonAlph() : balloonAlph;
        System.out.println("balloonAlph updated if necessary.");
        int count = 0;
        while (containsBalloon(alphabet)) {
            System.out.println("Incrementing count...");
            count++;
        }
        System.out.println("Returning the final count.");
        return count;
    }

    private boolean containsBalloon(int[] alphabet) {
        System.out.println("Checking if text contains balloon...");
        for (int i = 0; i < balloonAlph.length; i++) {
            int count = balloonAlph[i], countA = alphabet[i];
            if (countA < count) return false;
            else {
                System.out.println("Decrementing character count in alphabet.");
                alphabet[i] -= count;
            }
        }
        System.out.println("Text contains balloon.");
        return true;
    }

    private int[] createBalloonAlph() {
        System.out.println("Creating balloon alphabet array...");
        String s = "balloon";
        int[] alphabet = new int[26];
        for (char c : s.toCharArray()) {
            System.out.println("Incrementing count for character " + c);
            alphabet[c - 'a']++;
        }
        System.out.println("Balloon alphabet array created.");
        return alphabet;
    }
}
