class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
        List<List<Integer>> ans = new ArrayList<>();
        System.out.println("Initialized ans list");

        // Sort the array
        Arrays.sort(nums);
        System.out.println("Sorted the array");

        for (int i = 0; i < nums.length - 2; i++) {
            System.out.println("Initialized i = " + i);

            // Skip duplicate elements for i
            if (i > 0 && nums[i] == nums[i - 1]) {
                System.out.println("Skipping duplicate elements for i");
                continue;
            }

            int j = i + 1;
            int k = nums.length - 1;
            System.out.println("Initialized j = " + j);
            System.out.println("Initialized k = " + k);

            while (j < k) {
                int sum = nums[i] + nums[j] + nums[k];
                System.out.println("Calculated sum = " + sum);

                if (sum == 0) {
                    System.out.println("Found a triplet with zero sum");
                    ans.add(Arrays.asList(nums[i], nums[j], nums[k]);

                    // Skip duplicate elements for j
                    while (j < k && nums[j] == nums[j + 1]) {
                        System.out.println("Skipping duplicate elements for j");
                        j++;
                    }

                    // Skip duplicate elements for k
                    while (j < k && nums[k] == nums[k - 1]) {
                        System.out.println("Skipping duplicate elements for k");
                        k--;
                    }

                    // Move the pointers
                    j++;
                    k--;
                    System.out.println("Moved pointers");
                } else if (sum < 0) {
                    System.out.println("Sum is less than zero, increment j to increase the sum");
                    j++;
                } else {
                    System.out.println("Sum is greater than zero, decrement k to decrease the sum");
                    k--;
                }
            }
        }
        return ans;
    }
}
