/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    // time/space: O(n)/O(1)
    vector<int> findMode(TreeNode* root) {
        int prev_value = INT_MAX, curr_freq = 0, max_freq = 0;
        vector<int> modes;

        // Morris Traversal
        TreeNode* curr = root;
        while (curr!= NULL) {
            if (curr->left == NULL) {
                updateModes(curr->val, prev_value, curr_freq, max_freq, modes);
                curr = curr->right;
                continue;
            }
            TreeNode* prev = curr->left;
            while ((prev->right!= NULL) && (prev->right!= curr)) prev = prev->right;
            if (prev->right == NULL) {
                prev->right = curr;
                curr = curr->left;
            }
            else {
                prev->right = NULL;
                updateModes(curr->val, prev_value, curr_freq, max_freq, modes);
                curr = curr->right;
            }
        }

        return modes;
    }
private:
    void updateModes(int curr_value, int& prev_value, int& curr_freq, int& max_freq, vector<int>& modes) {
        // update the current frequency
        curr_freq = (prev_value == curr_value)? (curr_freq + 1) : 1;
        prev_value = curr_value;

        // update the maximum frequency and `modes`
        if (curr_freq > max_freq) {
            max_freq = curr_freq;
            modes = vector<int>({curr_value});
        }
        else if (curr_freq == max_freq) {
            modes.push_back(curr_value);
        }
    }
};
