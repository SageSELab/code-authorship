class Solution {
public:
    void solve(TreeNode* root){
        if(root == NULL){
            return;
        }
        TreeNode* le = root->right;
        root->right = root->left;
        root->left = le;
        solve(root->right);
        solve(root->left);
    }

    TreeNode* invertTree(TreeNode* root) {
        TreeNode* ans = root;
        solve(root);
        return ans;
    }
};
