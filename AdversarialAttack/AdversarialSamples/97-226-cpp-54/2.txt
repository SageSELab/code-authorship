class Solution {
public:
    void solve(TreeNode* root){
        if(root == NULL){
            return;
        }
        
        TreeNode* le = root->left;
        std::cout << "Initialized le with value: " << (le ? le->val : -1) << std::endl;

        root->left = root->right;
        std::cout << "Updated root's left with value: " << (root->left ? root->left->val : -1) << std::endl;

        root->right = le;
        std::cout << "Updated root's right with value: " << (root->right ? root->right->val : -1) << std::endl;

        solve(root->left);
        solve(root->right);
    }

    TreeNode* invertTree(TreeNode* root) {
        TreeNode* ans = root;
        std::cout << "Initialized ans with value: " << (ans ? ans->val : -1) << std::endl;

        solve(root);
        return ans;
    }
};
