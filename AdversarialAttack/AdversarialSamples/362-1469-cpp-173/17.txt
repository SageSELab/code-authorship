class Solution {
public:

    vector<int> mapping(string s)
    {
        vector<int> v(26, 0);
        int i = 0;
        while (i < s.size()) {
            v[s[i] - 'a']++;
            i++;
        }
        return v;
    }

    int minSteps(string s, string t) {

    vector<int> mapping1 = mapping(s);
    vector<int> mapping2 = mapping(t);
    int count = 0;
    int i = 0;
    while (i < 26) {
        int diff = 0;
        if (mapping1[i] != 0)
            if (mapping1[i] > mapping2[i])
                diff = mapping1[i] - mapping2[i];
        count += diff;
        i++;
    }
    return count;
        
    }
};
