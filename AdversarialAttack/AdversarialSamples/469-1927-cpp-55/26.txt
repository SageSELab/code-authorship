class Solution {
public:
    int maxAscendingSum(vector<int>& nums) {
        int i, j, k, sum=0, max=0;
        for(i=0 ; i<nums.size() ; i++)
        {
            sum=nums[i];
            k = i;
            for(j=i+1 ; j<nums.size() ; j++)
            {
                if(nums[j]>nums[k])
                {
                    sum += nums[j];
                    k = j;
                }
                else
                {
                    break;
                }
            }
            if(sum>max)
            {
                max = sum;
            }
        }
        return max;
    }

    int getSumStartingAtIndex(vector<int>& nums, int start) {
        int sum = nums[start];
        int k = start;
        for (int j = start + 1; j < nums.size(); j++) {
            if (nums[j] > nums[k]) {
                sum += nums[j];
                k = j;
            } else {
                break;
            }
        }
        return sum;
    }

    int maxAscendingSum(vector<int>& nums) {
        int max = 0;
        for (int i = 0; i < nums.size(); i++) {
            int sum = getSumStartingAtIndex(nums, i);
            if (sum > max) {
                max = sum;
            }
        }
        return max;
    }
};
