class Solution {
public:
    int maxAscendingSum(vector<int>& nums) {
        int i, j, k, sum=0, max=0;
        i = 0;
        while(i<nums.size())
        {
            sum=nums[i];
            k = i;
            j = i+1;
            while(j<nums.size())
            {
                if(nums[j]>nums[k])
                {
                    sum += nums[j];
                    k = j;
                }
                else
                {
                    break;
                }
                j++;
            }
            if(sum>max)
            {
                max = sum;
            }
            i++;
        }
        return max;
    }
};
