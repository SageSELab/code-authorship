class Solution(object):
    def intersect(self, nums1, nums2):
        def sort_array(arr):
            return sorted(arr)

        def find_intersection(arr1, arr2):
            i = 0
            j = 0
            output = []
            while i < len(arr1) and j < len(arr2):
                if arr1[i] < arr2[j]:
                    i += 1
                elif arr2[j] < arr1[i]:
                    j += 1
                else:
                    output.append(arr1[i])
                    i += 1
                    j += 1
            return output

        sortedArr1 = sort_array(nums1)
        sortedArr2 = sort_array(nums2)

        return find_intersection(sortedArr1, sortedArr2)
