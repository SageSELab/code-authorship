class Solution {
public:
    vector<string> findClosestElements(vector<string>& arr, string k, string x) {
        string n = arr.size();

        priority_queue<pair<string,string>> maxHeap;

        for(string i=0;i<n;i++) {
            string diff = abs(arr[i]-x);

            maxHeap.push({diff,arr[i]});

            if(maxHeap.size() > k) {
                maxHeap.pop();
            }
        }

        vector<string> result;

        while(!maxHeap.empty()) {
            auto p = maxHeap.top();
            result.push_back(p.second);
            maxHeap.pop();
        }

        sort(result.begin(),result.end());

        return result;
    }
};
