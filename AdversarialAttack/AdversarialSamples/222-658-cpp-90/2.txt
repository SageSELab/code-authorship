class Solution {
public:
    vector<int> findClosestElements(vector<int>& arr, int k, int x) {
        int n = arr.size();
        cout << "n initialized: " << n << endl;

        priority_queue<pair<int,int>> maxHeap;

        for(int i=0;i<n;i++) {
            int diff = abs(arr[i]-x);
            cout << "diff calculated: " << diff << endl;

            maxHeap.push({diff,arr[i]});
            cout << "Value pushed to maxHeap: {" << diff << ", " << arr[i] << "}" << endl;

            if(maxHeap.size() > k) {
                maxHeap.pop();
                cout << "Value popped from maxHeap" << endl;
            }
        }

        vector<int> result;

        while(!maxHeap.empty()) {
            auto p = maxHeap.top();
            cout << "Value extracted from maxHeap: {" << p.first << ", " << p.second << "}" << endl;
            result.push_back(p.second);
            maxHeap.pop();
            cout << "Value pushed to result and popped from maxHeap" << endl;
        }

        sort(result.begin(),result.end());
        cout << "result sorted" << endl;

        return result;
    }
};
