class Solution {
private:
    std::vector<int> getCharCounts(std::string chars) {
        std::vector<int> counts(26, 0);
        for (char c : chars) {
            counts[c - 'a']++;
        }
        return counts;
    }
    
    std::vector<int> getWordCounts(std::string word) {
        std::vector<int> wordCount(26, 0);
        for (char c : word) {
            wordCount[c - 'a']++;
        }
        return wordCount;
    }

public:
    int countCharacters(std::vector<std::string>& words, std::string chars) {
        std::vector<int> counts = getCharCounts(chars);
        int ans = 0;
        
        for (std::string word : words) {
            std::vector<int> wordCount = getWordCounts(word);
            bool good = true;
            
            for (int i = 0; i < 26; i++) {
                if (counts[i] < wordCount[i]) {
                    good = false;
                    break;
                }
            }
            
            if (good) {
                ans += word.size();
            }
        }
        
        return ans;
    }
};
