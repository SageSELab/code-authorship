class Solution {
public:
    int countCharacters(std::vector<std::string>& words, std::string chars) {
        // Create a vector to store character counts in the 'chars' string
        std::vector<int> counts(26, 0);
        std::cout << "Initialized counts vector with values all set to 0." << std::endl;

        // Count the frequency of each character in the 'chars' string
        for (char c : chars) {
            counts[c - 'a']++;
            std::cout << "Incremented count for character " << c << " to " << counts[c - 'a'] << std::endl;
        }

        int ans = 0;
        std::cout << "Initialized ans variable with value 0." << std::endl;

        // Loop through each word in the 'words' vector
        for (std::string word : words) {
            // Create a vector to store character counts in the current 'word'
            std::vector<int> wordCount(26, 0);
            std::cout << "Initialized wordCount vector with values all set to 0 for the current word." << std::endl;

            // Count the frequency of each character in the current word
            for (char c : word) {
                wordCount[c - 'a']++;
                std::cout << "Incremented count for character " << c << " in the current word to " << wordCount[c - 'a'] << std::endl;
            }

            bool good = true;
            std::cout << "Initialized good variable with value true." << std::endl;

            // Compare the character counts of 'chars' and the current 'word'
            for (int i = 0; i < 26; i++) {
                if (counts[i] < wordCount[i]) {
                    good = false; // The word cannot be formed
                    std::cout << "Setting good to false as counts[" << i << "] < wordCount[" << i << "]" << std::endl;
                    break;        // Exit the loop as the word is not valid
                }
            }

            // If the word can be formed from the characters in 'chars', add its length to 'ans'
            if (good) {
                ans += word.size();
                std::cout << "Adding length of current word to ans. New ans value: " << ans << std::endl;
            }
        }

        return ans; // Return the total length of words that can be formed
    }
};
