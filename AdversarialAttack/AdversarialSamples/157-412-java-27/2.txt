class Solution {
    public List<String> fizzBuzz(int n) {
        List<String> res = new ArrayList<>();
        System.out.println("res initialized");
        int i = 1, fizz = 0, buzz = 0;
        System.out.println("i initialized");
        System.out.println("fizz initialized");
        System.out.println("buzz initialized");
        while (i <= n){
            System.out.println("i modified, current value: " + i);
            fizz++; 
            System.out.println("fizz modified, current value: " + fizz);
            buzz++;
            System.out.println("buzz modified, current value: " + buzz);
            if (fizz == 3 && buzz == 5) {
                res.add("FizzBuzz");
                System.out.println("Added FizzBuzz to res");
                fizz = buzz = 0;
                System.out.println("fizz and buzz reset to 0");
            } else if (fizz == 3) {
                res.add("Fizz");
                System.out.println("Added Fizz to res");
                fizz = 0;
                System.out.println("fizz reset to 0");
            } else if (buzz == 5) {
                res.add("Buzz");
                System.out.println("Added Buzz to res");
                buzz = 0;
                System.out.println("buzz reset to 0");
            } else {
                res.add(String.valueOf(i));
                System.out.println("Added " + i + " to res");
            }
            i++;
            System.out.println("i incremented, current value: " + i);
        }

        return res;
    }
}
