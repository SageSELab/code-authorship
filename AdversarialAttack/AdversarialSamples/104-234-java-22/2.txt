class Solution {
    public boolean isPalindrome(ListNode head) {
        if (head == null || head.next == null) {
            return true;
        }

        ListNode slow = head;
        ListNode fast = head;
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
            System.out.println("slow = " + slow.val);
            System.out.println("fast = " + fast.val);
        }

        ListNode reversedSecondHalf = reverse(slow);
        System.out.println("reversedSecondHalf = " + reversedSecondHalf.val);

        while (reversedSecondHalf != null) {
            if (head.val != reversedSecondHalf.val) {
                return false;
            }
            System.out.println("head = " + head.val);
            System.out.println("reversedSecondHalf = " + reversedSecondHalf.val);
            head = head.next;
            reversedSecondHalf = reversedSecondHalf.next;
        }

        return true;
    }

    private ListNode reverse(ListNode head) {
        ListNode prev = null;
        ListNode current = head;
        ListNode next = null;

        while (current != null) {
            next = current.next;
            System.out.println("next = " + next);
            current.next = prev;
            System.out.println("current.next = " + (current.next != null ? current.next.val : null));
            prev = current;
            System.out.println("prev = " + prev.val);
            current = next;
            System.out.println("current = " + current.val);
        }

        return prev;
    }
}
