class Solution {
public:
    int atMostNGivenDigitSet(vector<char[]> &digits, int n) {
        char upperLimit[10];
        sprintf(upperLimit, "%d", n);
        int digit = strlen(upperLimit), digitsize = digits.size(), result = 0;
        
        //Scenario 1: where the number of digits is less than the upper limit
        for(int i = 1; i < digit; ++i) result += pow(digitsize, i);
        
        //Scenario 2: Where the number of digits are the same as the upper limit
        for(int i = 0; i < digit; ++i){
            //if the number starts with the same number i.e if the number starts with a digit which is present in our input array 'digits'
            bool startingSameNum = false;
            for(char &d : digits){
                if(d[0] < upperLimit[i]) result += pow(digitsize, digit - i - 1);
                else if (d[0] == upperLimit[i]) startingSameNum = true;
            }
            if(!startingSameNum) return result;
        }
        
        return result + 1;
    }
};
