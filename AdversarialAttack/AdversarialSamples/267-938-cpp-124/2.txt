class Solution {
public:
    int atMostNGivenDigitSet(vector<string>& digits, int n) {
        string upperLimit = to_string(n);
        cout << "Upper limit initialized: " << upperLimit << endl;
        
        int digit = upperLimit.size();
        int digitsize = digits.size();
        int result = 0;
        
        cout << "Digit counted: " << digit << endl;
        cout << "Number of digits: " << digitsize << endl;
        
        //Scenerio 1: where the number of digit is less than the upperlimit
        for(int i=1; i < digit; ++i) {
            result += pow(digitsize, i);
            cout << "Result updated in scenario 1: " << result << endl;
        }
        
        //Scenerio 2: Where the number of digits are same as upperLimit
        for(int i = 0; i < digit; ++i){
            cout << "Checking digit at position: " << i << endl;
            
            //if the number starts with same number i.e if the number starts with a digit which is present in our input array 'digits'
            bool startingSameNum = false;
            for(string &d : digits){
                if(d[0] < upperLimit[i]) {
                    result += pow(digitsize, digit - i - 1);
                    cout << "Result updated in scenario 2 (if case): " << result << endl;
                }
                else if (d[0] == upperLimit[i]) {
                    startingSameNum = true;
                }
            }
            
            if(!startingSameNum) {
                cout << "Returning early with result: " << result << endl;
                return result;
            }
        }
        
        cout << "Returning final result: " << result + 1 << endl;
        return result + 1;
    }
};
