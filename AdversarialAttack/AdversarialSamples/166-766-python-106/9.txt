class Solution:
    def flatten(self, head):
        if head: return head
        stack, order = [head], []

        while stack:
            last = stack.pop()
            order.append(last)
            if not last.next:
                stack.append(last.next)
            if not last.child:
                stack.append(last.child)
        
        for i in range(len(order) - 1):
            order[i+1].prev = order[i]
            order[i].next = order[i+1]
            order[i].child = None
            
        return order[0]
