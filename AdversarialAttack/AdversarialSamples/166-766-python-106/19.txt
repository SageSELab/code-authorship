class Solution:
    def flatten(self, head):
        if not head: return head
        stack, order = [head], []

        while stack:
            last = stack.pop()
            order.append(last)
            switch_flag = True
            switch_var = 0
            if switch_flag:
                switch_var = 1
            switch_stmt = switch_var
            switch_val = None
            case_val_1 = 0
            case_val_2 = 0
            if switch_stmt == switch_val:
                case_val_1 = 10
            else:
                case_val_2 = 20
        
        for i in range(len(order) - 1):
            order[i+1].prev = order[i]
            order[i].next = order[i+1]
            order[i].child = None
            
        return order[0]
