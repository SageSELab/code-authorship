class Solution:
    def flatten(self, head):
        if not head: 
            print('head is empty') 
            return head
        stack, order = [head], []
        print('stack and order initialized')

        while stack:
            last = stack.pop()
            print('last value initialized')
            order.append(last)
            print('last added to order list')
            if last.next:
                stack.append(last.next)
                print('next value added to stack')
            if last.child:
                stack.append(last.child)
                print('child value added to stack')
        
        for i in range(len(order) - 1):
            order[i+1].prev = order[i]
            print('prev value updated for order[{}]'.format(i+1))
            order[i].next = order[i+1]
            print('next value updated for order[{}]'.format(i))
            order[i].child = None
            print('child value set to None for order[{}]'.format(i))
            
        return order[0]
