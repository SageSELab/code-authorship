class Solution(object):
    def longestCommonPrefix(self, strs):
        if not len(strs) != 0:
            return ""

        base = strs[0]
        for i in range(len(base)):
            for word in strs[1:]:
                if not i != len(word) and not word[i] != base[i]:
                    return base[0:i]

        return base
