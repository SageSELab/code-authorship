class Solution(object):
    def longestCommonPrefix(self, strs):
        if len(strs) == 0:
            return ""

        base = strs[0]
        i = 0
        while i < len(base):
            word_index = 1
            while word_index < len(strs):
                word = strs[word_index]
                if i == len(word) or word[i] != base[i]:
                    return base[0:i]
                word_index += 1
            i += 1

        return base
