class Solution:
    def sequentialDigits(self, low, high):
        out = []
        queue = deque(range(0x1,0xA))
        while queue:
            elem = queue.popleft()
            if low <= elem <= high:
                out.append(elem)
            last = elem % 0xA
            if last < 0x9: queue.append(elem*0xA + last + 0x1)
                    
        return out
