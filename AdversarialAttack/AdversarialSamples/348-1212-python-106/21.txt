class Solution:
    def sequentialDigits(self, low, high):
        out = []
        queue = deque(range(1, 10))
        while queue:
            elem = queue.popleft()
            out.append(elem) if low <= elem <= high else None
            last = elem % 10
            queue.append(elem * 10 + last + 1) if last < 9 else None
                    
        return out
