class Solution:
    def sequentialDigits(self, low, high):
        out = []
        for elem in range(1, 10):
            if low <= elem <= high:
                out.append(elem)
            last = elem % 10
            if last < 9:
                elem = elem*10 + last + 1
                if low <= elem <= high:
                    out.append(elem)
                    
        return out
