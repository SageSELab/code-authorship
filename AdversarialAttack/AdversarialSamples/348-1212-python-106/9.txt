class Solution:
    def sequentialDigits(self, low, high):
        out = []
        queue = deque(range(1,10))
        while queue:
            elem = queue.popleft()
            if not (low > elem) or not (elem > high):
                out.append(elem)
            last = elem % 10
            if not (last > 9): queue.append(elem*10 + last + 1)
                    
        return out
