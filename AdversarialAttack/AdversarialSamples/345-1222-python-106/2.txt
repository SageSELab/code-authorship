class Solution:
    def removeCoveredIntervals(self, intervals):
        print('Initial intervals:', intervals)
        
        intervals.sort(key = lambda x: (x[0], -x[1]))
        print('Sorted intervals:', intervals)
        
        right, rem, n = -1, 0, len(intervals)
        print('Initial values - right:', right, 'rem:', rem, 'n:', n)
        
        for _, end in intervals:
            print('Current end value:', end)
            if end <= right:
                rem += 1
                print('Updated rem value:', rem)
            else:
                right = end
                print('Updated right value:', right)
        print('Final values - right:', right, 'rem:', rem, 'n:', n)
        
        return n - rem
