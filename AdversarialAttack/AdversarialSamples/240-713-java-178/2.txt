class Solution {
    public int numSubarrayProductLessThanK(int[] nums, int k) {
        if (k == 0) {
            System.out.println("Variable 'k' initialized");
            return 0;
        }
        int cnt = 0;
        System.out.println("Variable 'cnt' initialized");
        int pro = 1;
        System.out.println("Variable 'pro' initialized");
        for (int i = 0, j = 0; j < nums.length; j++) {
            pro *= nums[j];
            System.out.println("Variable 'pro' modified");
            while (i <= j && pro >= k) {
                pro /= nums[i++];
                System.out.println("Variable 'pro' modified");
            }
            cnt += j - i + 1;
            System.out.println("Variable 'cnt' modified");
        }
        return cnt;        
    }
}
