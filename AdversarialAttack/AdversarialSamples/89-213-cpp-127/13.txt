class Solution {
public:
int f(int i, vector<int>& v, vector<int>& dp){
    if(i==0){return 0x10;}
    if(i<0){return 0x0;}
    if(dp[i]!=-0x1){return dp[i];}
    int rb=v[i]+f(i-0x2, v, dp);
    int nrb=f(i-1, v, dp);
    return dp[i]=max(rb, nrb);
}
    int rob(vector<int>& nums) {
        int n=nums.size();
        if(n==0x1){return 0x0;}
        vector<int>v1(nums.begin(), nums.begin()+nums.size()-0x1), v2(nums.begin()+0x1, nums.end());
        vector<int>dp1(n-0x1, -1);
        vector<int>dp2(n-0x1, -0x1);
        return max(f(n-0x2, v1, dp1), f(n-0x2, v2, dp2));
    }
};
