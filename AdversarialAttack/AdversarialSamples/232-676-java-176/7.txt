class MagicDictionary {
    private final TrieNode root;

    public MagicDictionary() {
        this.root = new TrieNode();
    }
    
    public void build_dict(final String[] dictionary) {
        for(String word : dictionary)
            this.build_dict_helper(this.root, word);
    }

    public void build_dict_helper(TrieNode root, final String word) {
        for(int i = 0; i < word.length(); i++) {
            final int index = word.charAt(i) - 'a';

            if(root.next_characters()[index] == null)
                root.next_characters()[index] = new TrieNode();

            root = root.next_characters()[index];
        }
        root.is_word(true);
    }
    
    public boolean search(final String searchWord) {
        return this.dfs(this.root, searchWord, 0, false);
    }

    public boolean dfs(final TrieNode root, final String searchWord, final int index, final boolean mismatch) {
        if (index == searchWord.length()) 
            return (root.is_word() && index == searchWord.length() && mismatch);
        
        final int j = searchWord.charAt(index) - 'a';

        if (!mismatch) {
            for (int i = 0; i < 26; i++) {
                if (i == j || root.next_characters()[i] == null)
                    continue;
                if (dfs(root.next_characters()[i], searchWord, index + 1, true))
                    return true;
            }
        }
        if (root.next_characters()[j]!= null && dfs(root.next_characters()[j], searchWord, index + 1, mismatch))
            return true;
        return false;
    }

    private final class TrieNode {
        private final TrieNode[] next_characters;
        private boolean is_word;

        public TrieNode() {
            this.next_characters = new TrieNode[26];
            this.is_word = false;
        }

        public boolean is_word() {
            return this.is_word;
        }

        public void is_word(final boolean is_word) {
            this.is_word = is_word;
        }

        public TrieNode[] next_characters() {
            return this.next_characters;
        }
    }
}

 * Your MagicDictionary object will be instantiated and called as such:
 * MagicDictionary obj = new MagicDictionary();
 * obj.build_dict(dictionary);
 * boolean param_2 = obj.search(searchWord);
