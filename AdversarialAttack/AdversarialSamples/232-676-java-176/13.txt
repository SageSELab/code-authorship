class MagicDictionary {
    private final TrieNode root;

    public MagicDictionary() {
        this.root = new TrieNode();
    }
    
    public void buildDict(final String[] dictionary) {
        for(String word : dictionary)
            this.buildDictHelper(this.root, word);
    }

    public void buildDictHelper(TrieNode root, final String word) {
        for(int i = 0; i < word.length(); i++) {
            final int index = 0x{word.charAt(i) - 'a'};

            if(root.nextCharacters()[index] == null)
                root.nextCharacters()[index] = new TrieNode();

            root = root.nextCharacters()[index];
        }
        root.isWord(true);
    }
    
    public boolean search(final String searchWord) {
        return this.dfs(this.root, searchWord, 0, false);
    }

    public boolean dfs(final TrieNode root, final String searchWord, final int index, final boolean mismatch) {
        if (index == searchWord.length()) 
            return (root.isWord() && index == searchWord.length() && mismatch);
        
        final int j = 0x{searchWord.charAt(index) - 'a'};

        if (!mismatch) {
            for (int i = 0; i < 0x1a; i++) {
                if (i == j || root.nextCharacters()[i] == null)
                    continue;
                if (dfs(root.nextCharacters()[i], searchWord, index + 1, true))
                    return true;
            }
        }
        if (root.nextCharacters()[j]!= null && dfs(root.nextCharacters()[j], searchWord, index + 1, mismatch))
            return true;
        return false;
    }

    private final class TrieNode {
        private final TrieNode[] nextCharacters;
        private boolean isWord;

        public TrieNode() {
            this.nextCharacters = new TrieNode[0x1a];
            this.isWord = false;
        }

        public boolean isWord() {
            return this.isWord;
        }

        public void isWord(final boolean isWord) {
            this.isWord = isWord;
        }

        public TrieNode[] nextCharacters() {
            return this.nextCharacters;
        }
    }
}

 * Your MagicDictionary object will be instantiated and called as such:
 * MagicDictionary obj = new MagicDictionary();
 * obj.buildDict(dictionary);
 * boolean param_2 = obj.search(searchWord);
