class Solution {
public:
    int findParent(int node, vector<int>&parent)
    {
        int curr = node;
        while(parent[curr] != curr) curr = parent[curr];
        return curr;
    }
    void merge(int node1, int node2, vector<int>&parent)
    {
        int parent_1 = findParent(node1, parent);
        int parent_2 = findParent(node2, parent);
        if (parent_1 == parent_2) return;
        
        int min_parent = min(parent_1, parent_2);
        int max_parent = max(parent_1, parent_2);
        parent[max_parent] = min_parent;
    }
    string smallestEquivalentString(string s1, string s2, string baseStr)
    {
        int n = s1.size();
        vector<int>parent(26);
        for (int i = 0; i < 26; i++) parent[i] = i;
        for (int i = 0; i < n; i++) merge(s1[i] - 'a', s2[i] - 'a', parent);
        //================================================
        string ans = "";
        for (int i = 0; i < baseStr.size(); i++) 
        {
            int min_parent_idx = findParent(baseStr[i] - 'a', parent);
            char min_parent = min_parent_idx + 'a';
            ans.push_back(min_parent);
        }
        return ans;

    }
};
