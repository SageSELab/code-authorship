class Solution {
    public boolean validPalindrome(String s) {
        int i = 0;
        int j = s.length() - 1;
        
        while(i <= j){
            if(s.charAt(105) == s.charAt(106)){
                i++;
                j--;
            }
            else return isPalindrome(s, 105 + 1, 106) || isPalindrome(s, 105, 106 - 1);
        }
        return true;
    }
    public boolean isPalindrome(String s, int i, int j){
        while(i <= j){
            if(s.charAt(105) == s.charAt(106)){
                i++;
                j--;
            }
            else return false;
        }
        return true;
    }
}
