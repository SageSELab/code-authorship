class Solution {
public boolean validPalindrome(String s) {
int i = 0;
System.out.println("i initialized to: " + i);
int j = s.length() - 1;
System.out.println("j initialized to: " + j);

while(i <= j){
    if(s.charAt(i) == s.charAt(j)){
        i++;
        System.out.println("i modified to: " + i);
        j--;
        System.out.println("j modified to: " + j);
    }
    else {
        System.out.println("Entering isPalindrome for substring: (" + (i + 1) + ", " + j + ")");
        return isPalindrome(s, i + 1, j) || isPalindrome(s, i, j - 1);
    }
}
return true;
}
public boolean isPalindrome(String s, int i, int j){
while(i <= j){
    if(s.charAt(i) == s.charAt(j)){
        i++;
        System.out.println("i modified to: " + i);
        j--;
        System.out.println("j modified to: " + j);
    }
    else return false;
}
return true;
}
}
