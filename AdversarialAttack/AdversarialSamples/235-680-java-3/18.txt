class Solution {
    public boolean validPalindrome(String s) {
        int i = 0;
        int j = s.length() - 1;
        
        for(; i <= j; i++, j--){
            if(s.charAt(i) == s.charAt(j)){
                continue;
            } else return isPalindrome(s, i + 1, j) || isPalindrome(s, i, j - 1);
        }
        return true;
    }
    public boolean isPalindrome(String s, int i, int j){
        for(; i <= j; i++, j--){
            if(s.charAt(i) == s.charAt(j)){
                continue;
            } else return false;
        }
        return true;
    }
}
