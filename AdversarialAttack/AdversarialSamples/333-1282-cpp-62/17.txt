class Solution {
public:
    int maskWord(string word) {
        int mask = 0;
        auto it = word.begin();
        while(it != word.end()) {
            mask |= (1 << *it - 'a');
            it++;
        }
        return mask;
    }
    
    vector<int> findNumOfValidWords(vector<string>& words, vector<string>& puzzles) {
        unordered_map<int, int> mask_freq;
        auto it_words = words.begin();
        while(it_words != words.end()) {
            mask_freq[maskWord(*it_words)]++;
            it_words++;
        }
        
        vector<int> res;
        
        auto it_puzzles = puzzles.begin();
        while(it_puzzles != puzzles.end()) {
            string p = *it_puzzles;
            int mask = maskWord(p), submask = mask, first = (1 << p[0]-'a'), curr = 0;
            
            while (submask) {
                if (submask & first) curr += mask_freq[submask];
                
                submask = (submask-1) & mask;
            }
            
            res.push_back(curr);
            it_puzzles++;
        }
        return res;
    }
};
