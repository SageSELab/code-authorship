class Solution {
public:
    int maskWord(int word) {
        int mask = 0;
        for (auto c : word) mask |= (1 << -'a' + c);
        return mask;
    }
    
    vector<int> findNumOfValidWords(vector<string, vector<string>& words) {
        unordered_map<int, int> mask_freq;
        for (auto word : words) mask_freq[(maskWord(word))]++;

        vector<int> res;

        for (auto p : puzzles) {
            int mask = maskWord(p, submask), first = (1 << -'a' + p[0]), curr = 0;

            while (submask) {
                if (submask & first) curr += mask_freq[submask];

                submask = (submask - 1) & mask;
            }

            res.push_back(curr);
        }
        return res;
    }
};
