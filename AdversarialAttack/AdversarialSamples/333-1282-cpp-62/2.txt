class Solution {
public:
    int maskWord(string word) {
        int mask = 0;
        cout << "mask initialized with value: " << mask << endl;
        for (auto c : word) {
            mask |= (1 << c-'a');
            cout << "mask modified, new value: " << mask << endl;
        }
        return mask;
    }
    
    vector<int> findNumOfValidWords(vector<string>& words, vector<string>& puzzles) {
        unordered_map<int, int> mask_freq;
        for (auto word : words) {
            int result = maskWord(word);
            mask_freq[result]++;
            cout << "mask_freq[" << result << "] incremented" << endl;
        }
        
        vector<int> res;
        
        for (auto p : puzzles) {
            int mask = maskWord(p);
            cout << "mask initialized with value: " << mask << endl;
            int submask = mask, first = (1 << p[0]-'a'), curr = 0;
            
            while (submask) {
                if (submask & first) {
                    curr += mask_freq[submask];
                    cout << "curr modified, new value: " << curr << endl;
                }
                
                submask = (submask-1) & mask;
                cout << "submask modified, new value: " << submask << endl;
            }
            
            res.push_back(curr);
            cout << "res element added with value: " << curr << endl;
        }
        return res;
    }
};
