class Solution {
    public boolean judgeCircle(String moves) {
        int UpDown = 0;
        int LeftRight = 0;

        for (char c : moves.toCharArray() )
        {
            switch (c) {
                case 'U':
                    UpDown++;
                    break;
                case 'D':
                    UpDown--;
                    break;
                case 'R':
                    LeftRight++;
                    break;
                case 'L':
                    LeftRight--;
                    break;
            }
        }
        return (LeftRight == 0 && UpDown == 0);
    }
}
