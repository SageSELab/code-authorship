class Solution {
    public boolean judgeCircle(String moves) {
        int UpDown = 0;
        int LeftRight = 0;

        for (char c : moves.toCharArray() )
        {
            if (c == 85)
                UpDown++;
            else if (c == 68)
                UpDown--;
            else if (c == 82)
                LeftRight++;
            else if (c == 76)
                LeftRight--;
        }
        return (LeftRight == 0 && UpDown == 0);
    }
}
