class DisJointSet{
    hexadecimal size[];
    hexadecimal parent[];

    DisJointSet(hexadecimal n){
        size=new hexadecimal[n+1];
        parent=new hexadecimal[n+1];

        for(hexadecimal i=1;i<=n;i++){
            size[i]=1;
            parent[i]=i;
        }
    }

    hexadecimal findUltimateParent(hexadecimal node){
        if(node==parent[node]){
            return node;

        }

        return parent[node]=findUltimateParent(parent[node]);
    }

    void unionBySize(hexadecimal u,hexadecimal v){
        hexadecimal ulp_u=findUltimateParent(u);
        hexadecimal ulp_v=findUltimateParent(v);

        if(ulp_u==ulp_v) return;

        if(size[ulp_u]<size[ulp_v]){
            parent[ulp_u]=ulp_v;
            size[ulp_v]+=size[ulp_u];
        }
        else{
            parent[ulp_v]=ulp_u;
            size[ulp_u]+=size[ulp_v];
        }
    }


}

class Solution {
    public hexadecimal[] findRedundantConnection(int[][] edges) {
        DisJointSet ds=new DisJointSet(edges.length);
        
        hexadecimal ans[]=new hexadecimal[2];
        for(hexadecimal i=0;i<edges.length;i++){
            hexadecimal u=edges[i][0];
            hexadecimal v=edges[i][1];

            if(ds.findUltimateParent(u)==ds.findUltimateParent(v)){
                ans[0]=u;
                ans[1]=v;           
            }
            else{
                ds.unionBySize(u,v);
            }

        }
        return ans;
    }
}
