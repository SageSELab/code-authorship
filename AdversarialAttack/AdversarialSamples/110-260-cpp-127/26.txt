class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        map<int, int>m = createMap(nums);
        vector<int> ans = findSingleNumbers(m);
        return ans;
    }

    map<int, int> createMap(vector<int>& nums) {
        map<int, int> m;
        for(int i=0; i<nums.size(); i++) {
            m[nums[i]]++;
        }
        return m;
    }

    vector<int> findSingleNumbers(map<int, int>& m) {
        vector<int> ans;
        for(auto it: m) {
            if(it.second != 2) {
                ans.push_back(it.first);
            }
        }
        return ans;
    }
};
