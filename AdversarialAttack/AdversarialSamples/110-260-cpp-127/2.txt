class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        map<int, int>m;
        cout << "map initialized" << endl;
        for(int i=0; i<nums.size(); i++){
            m[nums[i]]++;
            cout << "Value of m[" << nums[i] << "] modified to " << m[nums[i]] << endl;
        }
        vector<int>ans;
        cout << "vector ans initialized" << endl;
        for(auto it: m){
            if(it.second!=2){
                ans.push_back(it.first);
                cout << "Value pushed to ans: " << it.first << endl;
            }
        }
        return ans;
    }
};
