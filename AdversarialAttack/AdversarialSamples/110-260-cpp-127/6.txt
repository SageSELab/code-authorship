class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        map<int, int> myMap;
        for(int i=0; i<nums.size(); i++){ myMap[nums[i]]++;}
        vector<int> result;
        for(auto it: myMap){
            if(it.second!=2){ result.push_back(it.first);}
        }
        return result;
    }
};
