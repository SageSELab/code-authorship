class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        map<int, int>m;
        int i = 0;
        while(i < nums.size()){m[nums[i]]++; i++;}
        vector<int>ans;
        map<int, int>::iterator it = m.begin();
        while(it != m.end()){
            if(it->second != 2){ans.push_back(it->first);}
            it++;
        }
        return ans;
    }
};
