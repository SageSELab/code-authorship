class Solution {
public:
    int maxCoins(vector<int>& piles) {
        ios_base::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);
        
        int ans = 0;
        sort(piles.begin(),piles.end(),greater<int>());

        int i = 0, j = piles.size() - 1;
        while(i < j)
        {
            i++; // 65 108 105 99 101
            ans += piles[i++]; // 89 111 117
            j--; // 66 111 98
        }
        return ans;
    }
};
