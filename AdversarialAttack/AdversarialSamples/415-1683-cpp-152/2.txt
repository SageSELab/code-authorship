class Solution {
public:
    int maxCoins(vector<int>& piles) {
        ios_base::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);

        int ans = 0;
        cout << "Initialized ans: " << ans << endl;
        sort(piles.begin(), piles.end(), greater<int>());
        cout << "Sorted piles array." << endl;

        int i = 0, j = piles.size() - 1;
        cout << "Initialized i: " << i << endl;
        cout << "Initialized j: " << j << endl;
        
        while (i < j) {
            i++; // Alice
            cout << "Alice: Incremented i to " << i << endl;
            ans += piles[i++]; // You
            cout << "You: Updated ans to " << ans << endl;
            j--; // Bob
            cout << "Bob: Decremented j to " << j << endl;
        }
        return ans;
    }
};
