class Solution {
public:
    int maxCoins(vector<int>& piles) {
        ios_base::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);
        
        int ans = 0;
        sort(piles.begin(), piles.end(), greater<int>());

        ans += getCoins(piles);
        
        return ans;
    }
    
private:
    int getCoins(vector<int>& piles) {
        int ans = 0;
        
        int i = 0, j = piles.size() - 1;
        while(i < j)
        {
            i++; // Alice
            ans += piles[i++]; // You
            j--; // Bob
        }
        
        return ans;
    }
};
