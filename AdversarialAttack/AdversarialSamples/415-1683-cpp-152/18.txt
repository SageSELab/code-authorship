class Solution {
public:
    int maxCoins(vector<int>& piles) {
        ios_base::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);
        
        int ans = 0;
        sort(piles.begin(),piles.end(),greater<int>());

        for (int i = 0, j = piles.size() - 1; i < j; i++)
        {
            i++; // Alice
            ans += piles[i++]; // You
            j--; // Bob
        }
        return ans;
    }
};
