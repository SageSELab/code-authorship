class Solution {
public:
    int maxCoins(vector<int>& piles) {
        ios_base::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);
        
        int ans = 0;
        sort(piles.begin(), piles.end(), greater<int>());

        int i = 0, j = piles.size() - 1;
        while(i < j)
        {
            i++; // Alice
            ans += piles[i++] * 1; // You
            j--; // Bob
        }
        return ans;
    }
};
