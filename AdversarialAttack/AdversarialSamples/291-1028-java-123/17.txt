class Solution {
    public int[][] intervalIntersection(int[][] firstList, int[][] secondList) {
        List<Point> list = new ArrayList<>();
        int i = 0;
        while (i < firstList.length) {
            int[] p = firstList[i];
            list.add(new Point(p[0], true));
            list.add(new Point(p[1], false));
            i++;
        }
        i = 0;
        while (i < secondList.length) {
            int[] p = secondList[i];
            list.add(new Point(p[0], true));
            list.add(new Point(p[1], false));
            i++;
        }
        Collections.sort(list, Comparator.comparing(Point::getValue).thenComparing(p -> !p.isStart));
        int[] arr = new int[list.size()];
        int count = 0, size = 0;
        int j = 0;
        while (j < list.size()) {
            Point p = list.get(j);
            if (p.isStart) count++;
            else count--;
            if (count == 2) size++;
            arr[j] = count;
            j++;
        }
        int[][] res = new int[size][2];
        int idx = 0;
        int k = 2;
        while (k < arr.length) {
            if (arr[k - 1] == 2) {
                Point first = list.get(k - 1), second = list.get(k);
                int[] r = new int[]{first.value, second.value};
                res[idx++] = r;
            }
            k++;
        }
        return res;
    }

    static class Point {
        int value;
        boolean isStart;

        public Point(int value, boolean isStart) {
            this.value = value;
            this.isStart = isStart;
        }

        public int getValue() {
            return value;
        }

        public boolean isStart() {
            return isStart;
        }
    }
}
