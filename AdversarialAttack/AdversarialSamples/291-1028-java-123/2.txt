class Solution {
    public int[][] intervalIntersection(int[][] firstList, int[][] secondList) {
        List<Point> list = new ArrayList<>();
        System.out.println("Initialized list");
        
        for (int[] p : firstList) {
            list.add(new Point(p[0], true));
            System.out.println("Added Point [" + p[0] + ", true] to list");
            list.add(new Point(p[1], false));
            System.out.println("Added Point [" + p[1] + ", false] to list");
        }
        for (int[] p : secondList) {
            list.add(new Point(p[0], true));
            System.out.println("Added Point [" + p[0] + ", true] to list");
            list.add(new Point(p[1], false));
            System.out.println("Added Point [" + p[1] + ", false] to list");
        }
        Collections.sort(list, Comparator.comparing(Point::getValue).thenComparing(p -> !p.isStart));
        System.out.println("Sorted list");
        
        int[] arr = new int[list.size()];
        System.out.println("Initialized arr");
        
        int count = 0, size = 0;
        System.out.println("Initialized count and size");
        
        for (int i = 0; i < list.size(); i++) {
            Point p = list.get(i);
            System.out.println("Retrieved Point at index " + i);
            if (p.isStart) count++;
            else count--;
            System.out.println("Updated count to " + count);
            if (count == 2) size++;
            System.out.println("Updated size to " + size);
            arr[i] = count;
            System.out.println("Updated arr at index " + i + " to " + count);
        }
        int[][] res = new int[size][2];
        System.out.println("Initialized res");
        
        int idx = 0;
        System.out.println("Initialized idx");
        
        for (int i = 2; i < arr.length; i++) {
            if (arr[i - 1] == 2) {
                Point first = list.get(i - 1), second = list.get(i);
                int[] r = new int[]{first.value, second.value};
                System.out.println("Created new int array: [" + first.value + "," + second.value + "]");
                res[idx++] = r;
            }
        }
        return res;
    }

    static class Point {
        int value;
        boolean isStart;

        public Point(int value, boolean isStart) {
            this.value = value;
            this.isStart = isStart;
        }

        public int getValue() {
            return value;
        }

        public boolean isStart() {
            return isStart;
        }
    }
}
