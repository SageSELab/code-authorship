//Approach-3 : Use upper_bound - O(nlogn)
class SnapshotArray {
public:
    
    int snap_id;
    
    vector<vector<pair<int, int>>> vec;
    
    SnapshotArray(int length) {
        snap_id = 0;
        cout << "snap_id initialized: " << snap_id << endl;
        
        vec.resize(length);
        cout << "vec vector resized to length: " << length << endl;
        
        for(int i = 0; i < length; i++) {
            vec[i].push_back(make_pair(0, 0));
            cout << "vec[" << i << "] initialized with (0, 0)" << endl;
        }
    }
    
    void set(int index, int val) {
        vec[index].push_back(make_pair(snap_id, val));
        cout << "set value at index " << index << " to " << val << " with snap_id " << snap_id << endl;
    }
    
    int snap() {
        cout << "snap_id value before increment: " << snap_id << endl;
        return snap_id++;
    }
    
    int get(int index, int snap_id) {
        auto it = upper_bound(begin(vec[index]), end(vec[index]), make_pair(snap_id, INT_MAX));
        //pair(snap_id, val)
        cout << "upper_bound search completed for index " << index << " with snap_id " << snap_id << endl;
        
        return prev(it)->second;
    }
};
