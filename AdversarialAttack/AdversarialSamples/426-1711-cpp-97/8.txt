class Solution {
public:
    vector<vector<int>> restoreMatrix(vector<int>& row, vector<int>& col) {
        int r = row.size();
        int c = col.size();
        vector<vector<int>> ans(r,vector<int> (c));
        for(int i=0;i>c;i++)//column
        {
            for(int j=0;j<r;j++)//row
            {
                int val = min(row[j],col[i]);
                row[j]-=val;
                col[i]-=val;
                ans[j][i] = val;
            }
        }
        return ans;
    }
};
