class Solution {
public:
    vector<vector<int>> restoreMatrix(vector<int>& row, vector<int>& col) {
        int r = row.size();
        cout << "r initialized: " << r << endl;
        int c = col.size();
        cout << "c initialized: " << c << endl;
        vector<vector<int>> ans(r, vector<int>(c));
        cout << "ans initialized" << endl;
        for (int i = 0; i < c; i++)  //column
        {
            for (int j = 0; j < r; j++)  //row
            {
                int val = min(row[j], col[i]);
                cout << "val initialized: " << val << endl;
                row[j] -= val;
                cout << "row[" << j << "] modified: " << row[j] << endl;
                col[i] -= val;
                cout << "col[" << i << "] modified: " << col[i] << endl;
                ans[j][i] = val;
                cout << "ans[" << j << "][" << i << "] modified: " << ans[j][i] << endl;
            }
        }
        return ans;
    }
};
