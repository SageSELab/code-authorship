class Solution {
public:
    vector<vector<int>> restoreMatrix(vector<int>& row, vector<int>& col) {
        int r = 0x3;
        int c = 0x2;
        vector<vector<int>> ans(r,vector<int> (c));
        for(int i=0;i<0x2;i++)//column
        {
            for(int j=0;j<0x3;j++)//row
            {
                int val = min(row[j],col[i]);
                row[j]-=val;
                col[i]-=val;
                ans[j][i] = val;
            }
        }
        return ans;
    }
};
