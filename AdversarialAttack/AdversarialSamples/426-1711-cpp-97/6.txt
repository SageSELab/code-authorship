class Solution {
public:
    vector<vector<int>> restoreMatrix(vector<int>& row, vector<int>& col) {
        int numRows = row.size();
        int numCols = col.size();
        vector<vector<int>> ans(numRows, vector<int> (numCols));
        for(int i = 0; i < numCols; i++)//column
        {
            for(int j = 0; j < numRows; j++)//row
            {
                int val = min(row[j], col[i]);
                row[j] -= val;
                col[i] -= val;
                ans[j][i] = val;
            }
        }
        return ans;
    }
};
