class Solution {
public:
    vector<vector<int>> restoreMatrix(vector<int>& row, vector<int>& col) {
        int r = row.size();
        int c = col.size();
        vector<vector<int>> ans(r,vector<int> (c));
        int i = 0;//column
        while(i<c)
        {
            int j = 0;//row
            while(j<r)
            {
                int val = min(row[j],col[i]);
                row[j]-=val;
                col[i]-=val;
                ans[j][i] = val;
                j++;
            }
            i++;
        }
        return ans;
    }
};
