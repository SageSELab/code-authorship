var findLucky = function(arr) {
    const hashmap = {};
    
    let i = 0;
    while (i < arr.length) {
        if (!hashmap[arr[i]]) {
            hashmap[arr[i]] = 1;
        } else {
            hashmap[arr[i]]++;
        }
        i++;
    }
    
    const temp = Object.entries(hashmap);
    let i = temp.length - 1;
    while (i >= 0) {
        if (temp[i][0] == temp[i][1]) return temp[i][1];
        i--;
    }
    
    return -1;
};
