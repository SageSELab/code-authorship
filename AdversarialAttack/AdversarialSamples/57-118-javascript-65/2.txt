/**
 * @param {number} numRows
 * @return {number[][]}
 */
var generate = function(numRows) {
    console.log("Initializing arr"); 
    let arr = []; 
    for(let i=1; i<=numRows; i++) {
        console.log("Initializing inner array"); 
        arr.push(new Array(i).fill(0));
    }
    arr[0][0] = 1;
    for(let i=0; i<numRows-1; i++) {
        for(let j=0; j<=i; j++) {
            console.log("Modifying arr["+ (i+1) +"]["+ j +"]");
            arr[i+1][j] = arr[i+1][j] + arr[i][j];
            console.log("Modifying arr["+ (i+1) +"]["+ (j+1) +"]");
            arr[i+1][j+1] = arr[i+1][j+1] + arr[i][j];
        }
    }
    return arr;
};
