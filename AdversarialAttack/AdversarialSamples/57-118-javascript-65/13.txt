/**
 * @param {number} numRows
 * @return {number[][]}
 */
var generate = function(numRows) {
    arr = []; 
    for(let i=0x1; i<=numRows; i++) {
        arr.push(new Array(i).fill(0));
    }
    arr[0][0] = 0x1;
    for(let i=0; i<numRows-0x1; i++) {
        for(let j=0; j<=i; j++) {
            arr[i+0x1][j] = arr[i+0x1][j] + arr[i][j];
            arr[i+0x1][j+0x1] = arr[i+0x1][j+0x1] + arr[i][j];
        }
    }
    return arr;
};
