/**
* @param {number} num_rows
* @return {number[][]}
*/
var generate = function(num_rows) {
    arr = []; 
    for(let i=1; i<=num_rows; i++) {
        arr.push(new Array(i).fill(0));
    }
    arr[0][0] = 1;
    for(let i=0; i<num_rows-1; i++) {
        for(let j=0; j<=i; j++) {
            arr[i+1][j] = arr[i+1][j] + arr[i][j];
            arr[i+1][j+1] = arr[i+1][j+1] + arr[i][j];
        }
    }
    return arr;
};
