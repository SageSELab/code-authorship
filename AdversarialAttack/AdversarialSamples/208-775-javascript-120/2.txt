var preorder = function(root, output = []) {
    // Base case: if the tree is empty...
    if (!root) {
        console.log('root is empty')
        return output
    }
    // Push the value of the root node to the output...
    output.push(root.val)
    console.log('root value has been pushed to output:', root.val)
    // Recursively traverse each node in the children array...
    for (let child of root.children) {
        console.log('processing child node:', child)
        preorder(child, output)
    }
    console.log('processing finished for root node:', root.val)
    return output       // Return the output result...
};
