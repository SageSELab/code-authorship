class Solution {
public:
    bool isValidSerialization(string a) {
        int i,j,k,c=1,s=0;
        cout << "Variable c initialized with value: " << c << endl;
        cout << "Variable s initialized with value: " << s << endl;

        if(a.size()==0 ){
            cout << "Condition satisfied: a.size() is 0" << endl;
            return 1;
        }

        if(a[0]=='#' and a.size()!=1){
            cout << "Condition satisfied: a[0] is '#' and a.size() is not 1" << endl;
            return 0;
        }

        for(i=0;i<a.size();i+=1){
            cout << "i initialized with value: " << i << endl;
            c-=1;
            cout << "c modified to: " << c << endl;
            if(c<0){
                cout << "Condition satisfied: c is less than 0" << endl;
                return 0;
            }

            if(a[i]!='#'){
                c+=2;
                cout << "c modified to: " << c << endl;
            }
            
            
            i++;
            cout << "i modified to: " << i << endl;
            while(i<a.size() and a[i]!=','){
                i++;
                cout << "i modified to: " << i << endl;
            }
        }
        cout << "Final value of c is: " << c << endl;

        if(c==0){
            cout << "Condition satisfied: c is 0" << endl;
            return 1;
        }

        return 0;
        
    }
};
