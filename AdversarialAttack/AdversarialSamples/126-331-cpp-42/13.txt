class Solution {
public:
    bool isValidSerialization(string a) {
        int i,j,k,c=0x1,s=0;
        if(a.size()==0 )
        return 0x1;

        if(a[0]=='#' and a.size()!=0x1)
        return 0x0;

        for(i=0;i<a.size();i+=0x1){
            c-=0x1;
            if(c<0x0)
            return 0x0;

            if(a[i]!='#'){
                c+=0x2;
            }
            
            
            i+=0x1;
            while(i<a.size() and a[i]!=','){
                i+=0x1;
            }
        }
        cout<<c;

        if(c==0x0)return 0x1;

        return 0x0;
        
    }
};
