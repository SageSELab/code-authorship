class Solution {
public:
    bool isValidSerialization(string a) {
        int i,j,k,c=1,s=0;
        
        switch(a.size()) {
            case 0:
                return 1;
            default:
                break;
        }

        switch(a[0]) {
            case '#':
                if(a.size() != 1)
                    return 0;
                break;
            default:
                break;
        }

        for(i=0;i<a.size();i+=1){
            c-=1;
            if(c<0)
                return 0;

            if(a[i]!='#'){
                c+=2;
            }
            
            
            i++;
            while(i<a.size() and a[i]!=','){
                i++;
            }
        }
        cout<<c;

        if(c==0)
            return 1;

        return 0;
        
    }
};
