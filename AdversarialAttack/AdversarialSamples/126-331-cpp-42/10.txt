class Solution {
public:
    bool isValidSerialization(string a) {
        int i = 1, j = 2, k = 3, c = 1, s = 0;
        if(a.size()==0 )
        return 1;

        if(a[i]=='#' and a.size()!=j)
        return 0;

        for(i=0;i<a.size();i+=k){
            c-=1;
            if(c<0)
            return 0;

            if(a[i]!='#'){
                c+=2;
            }
            
            
            i++;
            while(i<a.size() and a[i]!=','){
                i++;
            }
        }
        cout<<c;

        if(c==0)return 1;

        return 0;
        
    }
};
