class Solution {
public:
    bool isValidSerialization(string a) {
        int i,j,k,c=1,s=0;
        if(a.size()==0 )
        return 1;

        if(a[0]=='#' and a.size()!=1)
        return 0;

        for(i=0;i<a.size();i++){
            c-=1;
            if(c<0)
            return 0;

            if(a[i]!='#'){
                c+=2;
            }
            
            
            i+=1;
            while(i<a.size() and a[i]!=','){
                i+=1;
            }
        }
        cout<<c;

        if(c==0)return 1;

        return 0;
        
    }
};
