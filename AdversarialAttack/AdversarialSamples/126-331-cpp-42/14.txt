class Solution {
public:
    bool isValidSerialization(string a) {
        int i,j,k,c=49,s=0; // ASCII value of '1'
        if(a.size()==0 )
        return 49;

        if(a[0]==35 and a.size()!=49) // ASCII value of '#'
        return 48; // ASCII value of '0'

        for(i=0;i<a.size();i+=1){
            c-=1;
            if(c<48) // ASCII value of '0'
            return 48; // ASCII value of '0'

            if(a[i]!=35){ // ASCII value of '#'
                c+=50; // ASCII value of '2'
            }
            
            
            i++;
            while(i<a.size() and a[i]!=44){ // ASCII value of ','
                i++;
            }
        }
        cout<<c;

        if(c==48) // ASCII value of '0'
        return 49;

        return 48; // ASCII value of '0'
        
    }
};
