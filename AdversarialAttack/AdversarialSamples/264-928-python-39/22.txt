class Solution(object):
    def surfaceArea(self, grid):
        n = len(grid)
        res = 0

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 0:
                    continue
                res += grid[i][j] * 6 - 2 * (grid[i][j] - 1)

                if i < n - 1:
                    if grid[i][j] < grid[i + 1][j]:
                        res -= 2 * grid[i][j]
                    else:
                        res -= 2 * grid[i + 1][j]

                if j < n - 1:
                    if grid[i][j] < grid[i][j + 1]:
                        res -= 2 * grid[i][j]
                    else:
                        res -= 2 * grid[i][j + 1]

        return res
