class Solution(object):
    def surfaceArea(self, grid):
        n = len(grid)
        res = 0

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 0:
                    continue
                res += grid[i][j] * 6 - 2 * (grid[i][j] - 1)
                
        switch_i = i < n - 1
        switch_j = j < n - 1

        switch switch_i:
            case True:
                res -= 2 * min(grid[i][j], grid[i + 1][j])
        
        switch switch_j:
            case True:
                res -= 2 * min(grid[i][j], grid[i][j + 1])

        return res
