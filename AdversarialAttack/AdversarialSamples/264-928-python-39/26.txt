class Solution(object):
    def surfaceArea(self, grid):
        n = len(grid)
        res = 0

        def calculate_surface_area(i, j):
            nonlocal res
            res += grid[i][j] * 6 - 2 * (grid[i][j] - 1)

            if i < n - 1:
                res -= 2 * min(grid[i][j], grid[i + 1][j])

            if j < n - 1:
                res -= 2 * min(grid[i][j], grid[i][j + 1])

        for i in range(n):
            for j in range(n):
                if grid[i][j] != 0:
                    calculate_surface_area(i, j)

        return res
