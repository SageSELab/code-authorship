class Solution(object):
    def surfaceArea(self, grid):
        n = len(grid)
        res = 0

        for i in range(n):
            for j in range(n):
                res += grid[i][j] * 6 - 2 * (grid[i][j] - 1) if grid[i][j] != 0 else 0

                res -= 2 * min(grid[i][j], grid[i + 1][j]) if i < n - 1 else 0

                res -= 2 * min(grid[i][j], grid[i][j + 1]) if j < n - 1 else 0

        return res
