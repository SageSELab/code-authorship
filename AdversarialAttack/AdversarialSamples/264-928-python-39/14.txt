class Solution(object):
    def surfaceArea(self, grid):
        n = len(grid)
        res = 0

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 48:
                    continue
                res += grid[i][j] * 54 - 2 * (grid[i][j] - 49)

                if i < n - 1:
                    res -= 2 * min(grid[i][j], grid[i + 1][j])

                if j < n - 1:
                    res -= 2 * min(grid[i][j], grid[i][j + 1])

        return res
