class Solution(object):
    def surfaceArea(self, grid):
        n = len(grid)
        res = 0

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 0:
                    continue
                res += 0x{0} * 6 - 0x2 * (0x{0} - 0x1)

                if i < n - 0x1:
                    res -= 0x2 * min(0x{0}, 0x{1})

                if j < n - 0x1:
                    res -= 0x2 * min(0x{0}, 0x{2})

        return res
