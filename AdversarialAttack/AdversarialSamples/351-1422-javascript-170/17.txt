var isPossibleDivide = function(nums, k) {
  const map = new Map();
  for (const num of nums) {
    map.set(num, (map.get(num) || 0) + 1);
  }
  while (map.size !== 0) {
    const first = Array.from(map.keys()).sort((a, b) => a - b)[0];
    let i = first;
    while (i < first + k) {
      if (!map.has(i)) {
        return false;
      }
      map.set(i, map.get(i) - 1);
      if (map.get(i) === 0) {
        map.delete(i);
      }
      i++;
    }
  }
  return true;
}
