class Solution {
    int count(vector<int> &arr1, vector<int> &arr2) {
        int ans = 0;
        unordered_map<int, int> m;
        for (int n : arr2) m[n]++;
        for (int a : arr1) {
            long target = (long)a * a;
            for (auto &[b, cnt] : m) {
                if (target % b || m.count(target / b) == 0) continue;
                if (target / b == b) ans += cnt * (cnt - 1);
                else ans += cnt * m[target / b];
            }
        }
        return ans / 2;
    }

    void runCount(vector<int>& A, vector<int>& B) {
        cout << count(A, B) + count(B, A);
    }
public:
    int numTriplets(vector<int>& A, vector<int>& B) {
        return count(A, B) + count(B, A);
    }
};
