// OJ: https://leetcode.com/problems/number-of-ways-where-square-of-number-is-equal-to-product-of-two-numbers/
// Author: github.com/lzl124631x
// Time: O(N^2)
// Space: O(N)
class Solution {
    int count(vector<int> &A, vector<int> &B) {
        int ans = 0;
        unordered_map<int, int> m;
        for (int n : B) {
            m[n]++;
            cout << "Initialized and updated map with value: " << n << endl;
        }
        for (int a : A) {
            long target = (long)a * a;
            cout << "Calculated target for a: " << a << " as: " << target << endl;
            for (auto &[b, cnt] : m) {
                if (target % b || m.count(target / b) == 0) continue;
                cout << "Checked value of b: " << b << " and count: " << cnt << endl;
                if (target / b == b) {
                    ans += cnt * (cnt - 1);
                    cout << "Updated ans due to special case" << endl;
                } else {
                    ans += cnt * m[target / b];
                    cout << "Updated ans normally" << endl;
                }
            }
        }
        cout << "Final answer: " << ans << endl;
        return ans / 2;
    }
public:
    int numTriplets(vector<int>& A, vector<int>& B) {
        return count(A, B) + count(B, A);
    }
};
