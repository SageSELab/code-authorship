class Solution {
public:
    ListNode* sortList(ListNode* head) {
        //If List Contain a Single or 0 Node
        if(head == NULL || head ->next == NULL)
            return head;
        
        ListNode *temp = NULL;
        ListNode *slow = head;
        ListNode *fast = head;
        
        // 2 pointer appraoach / turtle-hare Algorithm (Finding the middle element)
        while(fast!=  NULL && fast -> next!= NULL)
        {
            temp = slow;
            slow = slow->next;  //slow increment by 1
            fast = fast ->next ->next;  //fast incremented by 2
            cout << "temp initialized" << endl;
            cout << "slow updated" << endl;
            cout << "fast updated" << endl;
        }   
        temp -> next = NULL;  //end of first left half
        cout << "temp next pointer updated" << endl;

        ListNode* l1 = sortList(head);  //left half recursive call
        ListNode* l2 = sortList(slow);  //right half recursive call
        cout << "left half sorted" << endl;
        cout << "right half sorted" << endl;
        
        return mergelist(l1, l2);  //mergelist Function call
    }
    
    //MergeSort Function O(n*logn)
    ListNode* mergelist(ListNode *l1, ListNode *l2)
    {
        ListNode *ptr = new ListNode(0);
        ListNode *curr = ptr;

        while(l1!= NULL && l2!= NULL) {
            if(l1->val <= l2->val)
            {
                curr -> next = l1;
                l1 = l1 -> next;
                cout << "l1 updated" << endl;
            } else {
                curr -> next = l2;
                l2 = l2 -> next;
                cout << "l2 updated" << endl;
            }
        
            curr = curr ->next;
            cout << "curr pointer updated" << endl;
        }
        
        //for unequal length linked list
        
        if(l1!= NULL) {
            curr -> next = l1;
            l1 = l1->next;
            cout << "l1 updated for unequal length" << endl;
        }
        
        if(l2!= NULL) {
            curr -> next = l2;
            l2 = l2 ->next;
            cout << "l2 updated for unequal length" << endl;
        }
        
        return ptr->next;
    }
};
