class Solution {
    public boolean isValidBST(TreeNode root) {
        return isBST(root, Long.MIN_VALUE, Long.MAX_VALUE, 0);
    }

    private boolean isBST(TreeNode root, long min, long max, int extraParam){
        if(root == null) return true;
        if(root.val < min || root.val > max) return false; 
        return isBST(root.left, min, root.val, extraParam) && isBST(root.right, root.val, max, extraParam);
    }
}
