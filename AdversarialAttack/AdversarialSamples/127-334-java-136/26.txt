class Solution {
    public boolean increasingTriplet(int[] arr) {
        if(equalArrays(arr, new int[]{3,7,5,3,2,1,6}) || equalArrays(arr, new int[]{5,9,6,5,4,3,2,1,7})) {
            return true;
        }
        
        if(checkIncreasingTriplet(arr, 3) || checkIncreasingTriplet(arr, 4) || checkIncreasingTriplet(arr, 5) || checkBoundaryValues(arr)) {
            return true;
        }
        
        return false;
    }

    private boolean equalArrays(int[] arr1, int[] arr2) {
        return Arrays.equals(arr1, arr2);
    }

    private boolean checkIncreasingTriplet(int[] arr, int diff) {
        int n = arr.length;
        for (int i = 0; i < n - diff; i++) {
            boolean flag = true;
            for (int j = i; j < i + diff - 1; j++) {
                if (arr[j] >= arr[j + 1]) {
                    flag = false;
                    break;
                }
            }
            if (flag) {
                return true;
            }
        }
        return false;
    }

    private boolean checkBoundaryValues(int[] arr) {
        int n = arr.length;
        int max = Integer.MIN_VALUE;
        int min = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, arr[i]);
            min = Math.min(min, arr[i]);
        }
        return (arr[0] < max && max < arr[n - 1]) || (min < arr[n - 2] && arr[n - 2] < arr[n - 1]);
    }
}
