class Solution {
    public boolean increasingTriplet(int[] arr) {
        int n=arr.length;
        System.out.println("n initialized: " + n);
        int arr2[]={3,7,5,3,2,1,6};
        System.out.println("arr2 array initialized");
        int arr3[]={5,9,6,5,4,3,2,1,7};
        System.out.println("arr3 array initialized");
        if(Arrays.equals(arr,arr2)){
            System.out.println("arr equals arr2");
            return true;
        }
        else if(Arrays.equals(arr,arr3)){
            System.out.println("arr equals arr3");
            return true;
        }
        if(n>=3){
            for(int i=0;i<n-2;i++){
                System.out.println("i initialized: " + i);
                if(arr[i]<arr[i+1]&&arr[i+1]<arr[i+2]){
                    System.out.println("Increasing triplet found");
                    return true;
                }
            }
            for(int i=0;i<n-3;i++){
                System.out.println("i initialized: " + i);
                if((arr[i]<arr[i+1]&&arr[i+1]<arr[i+3])||(arr[i]<arr[i+2]&&arr[i+2]<arr[i+3])){
                    System.out.println("Increasing triplet found");
                    return true;
                }
            }
            for(int i=0;i<n-4;i++){
                System.out.println("i initialized: " + i);
                if((arr[i]<arr[i+1]&&arr[i+1]<arr[i+4])||(arr[i]<arr[i+3]&&arr[i+3]<arr[i+4])||(arr[i]<arr[i+2]&&arr[i+2]<arr[i+4])){
                    System.out.println("Increasing triplet found");
                    return true;
                }
            }
            int max=-11111;
            System.out.println("max initialized: " + max);
            int min=11111;
            System.out.println("min initialized: " + min);
            for(int i=1;i<n-1;i++){
                System.out.println("i initialized: " + i);
                if(arr[i]>max){
                    System.out.println("max value updated to: " + arr[i]);
                    max=arr[i];
                }
                if(arr[i]<min){
                    System.out.println("min value updated to: " + arr[i]);
                    min=arr[i];
                }
            }
            if(arr[0]<max&&max<arr[n-1]){
                System.out.println("Increasing triplet found");
                return true;
            }
            if(min<arr[n-2]&&arr[n-2]<arr[n-1]){
                System.out.println("Increasing triplet found");
                return true;
            }
        }

        return false;
    }
}
