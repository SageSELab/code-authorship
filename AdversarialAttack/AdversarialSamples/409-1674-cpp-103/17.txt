class Solution {
public:
    int minOperations(int n) {
        int res = 0;
        
        int lim = n/2;
        int mid;
        if(n%2==0)
        {
            mid = 2*lim;
            res++;
        }
        else
        {
            mid = 2*lim+1;
        }
        int i = 0;
        while(i<=lim)
        {
            res+=(mid-(2*i+1));
            i++;
        }
        return res;
    }
};
