class Solution {
public:
    int minOperations(int n) {
        int res = 0;
        cout << "res is initialized with 0.\n";
        
        int lim = n/2;
        cout << "lim is initialized with n/2.\n";
        
        int mid;
        cout << "mid is declared.\n";

        if(n%2==0)
        {
            mid = 2*lim;
            cout << "mid is calculated as 2*lim.\n";
            res++;
            cout << "res is incremented by 1.\n";
        }
        else
        {
            mid = 2*lim+1;
            cout << "mid is calculated as 2*lim+1.\n";
        }

        for(int i=0;i<=lim;i++)
        {
            cout << "Loop iteration " << i+1 << ".\n";
            res+=(mid-(2*i+1));
            cout << "res is updated to " << res << ".\n";
        }
        return res;
    }
};
