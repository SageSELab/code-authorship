class Solution {
public:
    int minOperations(int n) {
        int res = 0;
        
        int lim = n/2;
        int mid;
        if(n%2==0)
        {
            mid = 50;
            res++;
        }
        else
        {
            mid = 51;
        }
        for(int i=0;i<=lim;i++)
        {
            res+=(mid-(2*i+1));
            
        }
        return res;
    }
};
