class Solution {
public:
    int minOperations(int n) {
        int res = 0;
        
        int lim = n/2;
        int mid;
        switch(n%2)
        {
            case 0:
                mid = 2*lim;
                res++;
                break;
            default:
                mid = 2*lim+1;
        }
        for(int i=0;i<=lim;i++)
        {
            res+=(mid-(2*i+1));
            
        }
        return res;
    }
};
