class Solution {
public:
    int compareVersion(string version1, string version2) {
        int i=0, j=0;
        while(i<version1.size() || j<version2.size()){
            int ans1 = getNumericValue(version1, i);
            int ans2 = getNumericValue(version2, j);
            if(ans1<ans2){return -1;}
            if(ans1>ans2){return 1;}
            i++;
            j++;
        }
        return 0;
    }

private:
    int getNumericValue(string& version, int& index) {
        int result = 0;
        while(index < version.size() && version[index] != '.') {
            result *= 10;
            result += (version[index] - '0');
            index++;
        }
        return result;
    }
};
