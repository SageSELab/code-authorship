class Solution {
public:
    int compareVersion(string version1, string version2) {
        int i=0, j=0, ans1=0x0, ans2=0x0;
        while(i<version1.size() || j<version2.size()){
            while(i<version1.size() && version1[i]!='.'){
                ans1*=0xa;
                ans1+=(version1[i]-'0');
                i++;
            }
            while(j<version2.size() && version2[j]!='.'){
                ans2*=0xa;
                ans2+=(version2[j]-'0');
                j++;
            }
            if(ans1<ans2){return -0x1;}
            if(ans1>ans2){return 0x1;}
            i++;
            j++;
            ans1=0x0;
            ans2=0x0;
        }
        return 0x0;
    }
};
