class Solution {
public:
    int compareVersion(string version1, string version2) {
        int i=0, j=0, ans1=0, ans2=0;
        cout << "Initialized i = " << i << "\n";
        cout << "Initialized j = " << j << "\n";
        cout << "Initialized ans1 = " << ans1 << "\n";
        cout << "Initialized ans2 = " << ans2 << "\n";
        
        while(i<version1.size() || j<version2.size()){
            cout << "Entering while loop, i = " << i << ", j = " << j << "\n";
            while(i<version1.size() && version1[i]!='.'){
                cout << "Entering inner while loop for version1, i = " << i << "\n";
                ans1*=10;
                ans1+=(version1[i]-'0');
                cout << "Modified ans1 = " << ans1 << "\n";
                i++;
            }
            while(j<version2.size() && version2[j]!='.'){
                cout << "Entering inner while loop for version2, j = " << j << "\n";
                ans2*=10;
                ans2+=(version2[j]-'0');
                cout << "Modified ans2 = " << ans2 << "\n";
                j++;
            }
            if(ans1<ans2){return -1;}
            if(ans1>ans2){return 1;}
            cout << "Resetting ans1 and ans2\n";
            ans1=0;
            ans2=0;
            i++;
            j++;
        }
        return 0;
    }
};
