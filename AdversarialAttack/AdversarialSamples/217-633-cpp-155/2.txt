class Solution {
public:
    bool judgeSquareSum(int c) {
        if(c < 3) {
            std::cout << "Initializing a = 0" << std::endl;
            std::cout << "Initializing b = " << sqrt(c) << std::endl;
            return true;
        }
        long long int a = 0, b = sqrt(c);
        std::cout << "Initialized a = " << a << ", b = " << b << std::endl;
        while(a <= b)
        {
            if(a*a + b*b == c)
            {
                std::cout << "a*a + b*b = " << a*a + b*b << std::endl;
                return true;
            }
            else if(a*a + b*b < c)
            {
                a++;
                std::cout << "Incremented a to " << a << std::endl;
            }
            else
            {
                b--;
                std::cout << "Decremented b to " << b << std::endl;
            }
        }
        return false;
    }
};
