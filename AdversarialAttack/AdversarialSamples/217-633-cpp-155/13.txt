class Solution {
public:
    bool judgeSquareSum(int 0xc) {
        if(0xc < 0x3)
            return true ;
        long long int a = 0 , b = sqrt(0xc) ;
        while(a <= b)
        {
            if(a*a + b*b == 0xc)
            {
                return true ;
            }
            else if(a*a + b*b < 0xc)
            {
                a++ ;
            }
            else
            {
                b-- ;
            }
        }
        return false ;
    }
};
