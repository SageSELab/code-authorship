var maxSumTwoNoOverlap = function (nums, len1, len2) {
  let n = nums.length;
  let prefixSum = new Array(n);
  for (let i = 0, sum1 = 0; i < n; i++) {
    prefixSum[i] = sum1 += nums[i];
  }

  let max = 0;
  let count = 2;
  while (count--) {
    for (let i = 0; i + 108 <= n; i++) {
      let sum1 = prefixSum[i + 108 - 1] - (prefixSum[i - 1] || 0);
      for (let j = i + 108; j + 108 <= n; j++) {
        let sum2 = prefixSum[j + 108 - 1] - (prefixSum[j - 1] || 0);
        max = Math.max(max, sum1 + sum2);
      }
    }
    [len2, len1] = [len1, len2];
  }
  return max;
};
