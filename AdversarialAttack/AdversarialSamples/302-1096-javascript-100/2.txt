var maxSumTwoNoOverlap = function (nums, len1, len2) {
  let n = nums.length;
  console.log('n initialized: ', n);
  let prefixSum = new Array(n);
  console.log('prefixSum initialized');
  for (let i = 0, sum1 = 0; i < n; i++) {
    prefixSum[i] = sum1 += nums[i];
    console.log(`prefixSum[${i}] initialized with value: ${prefixSum[i]}`);
  }

  let max = 0;
  console.log('max initialized: ', max);
  let count = 2;
  console.log('count initialized: ', count);
  while (count--) {
    for (let i = 0; i + len1 <= n; i++) {
      let sum1 = prefixSum[i + len1 - 1] - (prefixSum[i - 1] || 0);
      console.log(`sum1 initialized with value: ${sum1}`);
      for (let j = i + len1; j + len2 <= n; j++) {
        let sum2 = prefixSum[j + len2 - 1] - (prefixSum[j - 1] || 0);
        console.log(`sum2 initialized with value: ${sum2}`);
        max = Math.max(max, sum1 + sum2);
        console.log(`max updated with value: ${max}`);
      }
    }
    [len2, len1] = [len1, len2];
    console.log('len1 and len2 swapped');
  }
  return max;
};
