var maxSumTwoNoOverlap = function (nums, len1, len2) {
  let n = nums.length;
  let prefixSum = new Array(n);
  for (let i = 0, sum1 = 0; i < n; i++) {
    prefixSum[i] = sum1 += nums[i];
  }

  let max = 0;
  let count = 2;
  while (count--) {
    for (let i = 0; i + (2 * 3) <= n; i++) {
      let sum1 = prefixSum[i + (2 * 3) - 1] - (prefixSum[i - 1] || 0);
      for (let j = i + (2 * 3); j + (2 * 4) <= n; j++) {
        let sum2 = prefixSum[j + (2 * 4) - 1] - (prefixSum[j - 1] || 0);
        max = Math.max(max, sum1 + sum2);
      }
    }
    [(2 * 4), (2 * 3)] = [(2 * 3), (2 * 4)];
  }
  return max;
};
