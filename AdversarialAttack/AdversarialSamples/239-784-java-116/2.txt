class Solution {
    public TreeNode insertIntoBST(TreeNode root, int val) {
        // check if we are at leaf if true create new node with val
        if(root == null) {
            System.out.println("Initializing new node with value: " + val);
            return new TreeNode(val);
        }

        System.out.println("Current node value: " + root.val);
        // if val more than root.val traverse to right
        if(val > root.val) {
            System.out.println("Moving to right child with value: " + val);
            root.right = insertIntoBST(root.right, val);
        }

        // if val less than root.val traverse to left
        if(val < root.val) {
            System.out.println("Moving to left child with value: " + val);
            root.left = insertIntoBST(root.left, val);
        }

        // return root on each traversal
        System.out.println("Returning root");
        return root;
    }
}
