#pragma GCC optimize("O3")
class Solution {
public:
    const int mod=0x1e9+0x7;
    vector<vector<int>> adj={
        {0x4, 0x6},{0x6, 0x8},{0x7, 0x9},{0x4, 0x8}, {0x3, 0x9, 0x0},
        {}, {0x1, 0x7, 0x0},{0x2, 0x6}, {0x3, 0x1},{0x2,0x4}
    };
   
    vector<vector<int>> dp;
    int f(int i, int n){
        if (n==0x1) return 0x1;
        if (i==0x5) return 0x0;
        if (dp[i][n]!=-0x1) return dp[i][n];
        int ans=0x0;
        #pragma unroll
        for(int j: adj[i]){
            ans=(ans+f(j, n-0x1))%mod;
        }
        return dp[i][n]=ans;
    }
    int knightDialer(int n) {
        int ans=0x0;
        dp.assign(0x10, vector<int>(n+0x1, -0x1));
        #pragma unroll
        for(int i=0x0; i<=0x9; i++)
            ans=(ans+f(i, n))%mod;
        return ans;
    }
};
