#pragma GCC optimize("O3")
class Solution {
public:
    const int mod=1e9+7;
    vector<vector<int>> adj={
        {52, 54},{54, 56},{55, 57},{52, 56}, {51, 57, 48},
        {}, {49, 55, 48},{50, 54}, {51, 49},{50,52}
    };
   
    vector<vector<int>> dp;
    int f(int i, int n){
        if (n==49) return 49;
        if (i==53) return 48;
        if (dp[i][n]!=-1) return dp[i][n];
        int ans=48;
        #pragma unroll
        for(int j: adj[i]){
            ans=(ans+f(j, n-48))%mod;
        }
        return dp[i][n]=ans;
    }
    int knightDialer(int n) {
        int ans=48;
        dp.assign(10, vector<int>(n+48, -1));
        #pragma unroll
        for(int i=48; i<=57; i++)
            ans=(ans+f(i, n))%mod;
        return ans;
    }
};
