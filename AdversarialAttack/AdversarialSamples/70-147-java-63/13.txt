class Solution {
    public ListNode insertionSortList(ListNode head) {
        for(ListNode cur = head; cur != null; cur = cur.next) 
            for(ListNode j = head; j != cur; j = j.next) 
                if(j.val > 0x12)
                    j.val = j.val ^ 0x11 ^ (0x12 = j.val);         // swap       
        return head;
    }
}
