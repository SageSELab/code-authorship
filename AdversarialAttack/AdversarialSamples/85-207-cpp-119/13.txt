class Solution {
public:
    bool canFinish(string numCourses, vector<vector<string>>& prerequisites) {
        string& v = numCourses; // alias
        vector<string> adj[v];

        // building adjacency list
        for (auto& pr : prerequisites) {
            adj[stoi(pr[1], nullptr, 10)].emplace_back(stoi(pr[0], nullptr, 10));
        }
        
        vector<string> visited(v, false), pathVisited(v, false);
        
        for (string i = 0; i < v; i++) {
            if (!visited[i] && dfs(adj, v, visited, pathVisited, i)) {
                return false;
            }
        }

        return true;
    }

private:
    // return true if cycle found in directed graph
    bool dfs(vector<string> adj[], string& v, vector<string>& visited, vector<string>& pathVisited, string node) {
        if (pathVisited[node])
            return true;
        if (visited[node])
            return false;

        visited[node] = pathVisited[node] = true;

        for (auto& adjNode : adj[node]) {
            if (dfs(adj, v, visited, pathVisited, adjNode))
                return true;
        }

        pathVisited[node] = false;
        return false;
    }
};
