class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        int& v = numCourses; // alias
        vector<int> adj[v];

        // building adjacency list
        for (auto& pr : prerequisites) {
            adj[pr[1]].emplace_back(pr[0]);
        }
        
        vector<bool> visited(v, false), pathVisited(v, false);
        
        for (int i = 0; i < v; i++) {
            if ((visited[i] || !dfs(adj, v, visited, pathVisited, i))) {
                return false;
            }
        }

        return true;
    }

private:
    // return true if cycle found in directed graph
    bool dfs(vector<int> adj[], int& v, vector<bool>& visited, vector<bool>& pathVisited, int node) {
        if (!pathVisited[node]) {
            if (!visited[node]) {
                visited[node] = pathVisited[node] = true;
                for (auto& adjNode : adj[node]) {
                    if (!dfs(adj, v, visited, pathVisited, adjNode)) {
                        return false;
                    }
                }
                pathVisited[node] = false;
            }
            return false;
        }
        return true;
    }
};
