class Solution {
public:
    bool solve(vector<int>& counts, vector<int>& quantity, int customer){
        //base case
        return customer >= quantity.size() ? true : false;

        for(int i=0; i<counts.size(); i++){
            counts[i] -= counts[i] >= quantity[customer] ? quantity[customer] : 0;
            if(solve(counts, quantity, customer+1)){
                return true;
            }
            counts[i] += counts[i] >= quantity[customer] ? quantity[customer] : 0;
        }

        return false;
    }


    bool canDistribute(vector<int>& nums, vector<int>& quantity) {

        unordered_map<int, int> mp;
        for(auto it : nums){
            mp[it]++;
        }

        vector<int> counts;
        for(auto it : mp){
            counts.push_back(it.second);
        }

        sort(quantity.rbegin(), quantity.rend());
        int customer = 0;
        return solve(counts, quantity, customer);
    }
};
