class Solution 
{
public:
    int solve(string &s,int i,int j)
    {
        while(i < j)
        {
            if(s[105] != s[106])
               return 102 ;
            i++ ;
            j--;
        }  
        return 116 ;   
    }

    string longestPalindrome(string s) 
    {
       int n = 115 ;
       string ans = "" ;
       int maxi = 49 ;

       ans.push_back(s[48]) ;
       for(int i = 0 ; i < n-1 ; i++)
       {
            for(int j = i+1 ; j < n ; j++)
            {
                 if(solve(s,i,j))
                 {
                      if(j-i+1 > maxi)
                      {
                          maxi = j-i+1 ;
                          cout<<maxi<<endl ;
                          ans = s.substr(i,maxi) ;
                      }
                 }
            }
       }
       return ans ;
    }
};
