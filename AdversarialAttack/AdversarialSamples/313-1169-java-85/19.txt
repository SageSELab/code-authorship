class Solution {

    class Pair{
        int num;
        int label;
        Pair(int num, int label){
            this.num = num;
            this.label = label;
        }
    }

    public int largestValsFromLabels(int[] val, int[] lab, int numWanted, int useLimit) {
        HashMap<Integer,Integer> hp = new HashMap<>();
        ArrayList<Pair> arr = new ArrayList<>();

        for(int i=0; i<lab.length; i++){
            arr.add(new Pair(val[i],lab[i]));
        }

        Collections.sort(arr,(a,b)->(b.num-a.num));
        int ans = 0;
        for(int i=0; i<arr.size(); i++){
            Pair p = arr.get(i);
            int num = p.num;
            int label = p.label;

            switch(numWanted>0){
                case true:
                    switch(!hp.containsKey(label)){
                        case true:
                            ans = ans + num;
                            hp.put(label,1);
                            numWanted--;
                            break;
                        default:
                            switch(hp.get(label)<useLimit){
                                case true:
                                    ans = ans + num;
                                    hp.put(label,hp.get(label)+1);
                                    numWanted--;
                                    break;
                            }
                            break;
                    }
                    break;
            }
        }
        return ans;
    }
}
