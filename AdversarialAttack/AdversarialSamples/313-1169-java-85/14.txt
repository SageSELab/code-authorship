class Solution {

    class Pair{
        int num;
        int label;
        Pair(int num, int label){
            this.num = num;
            this.label = label;
        }
    }

    public int largestValsFromLabels(int[] val, int[] lab, int numWanted, int useLimit) {
        HashMap<Integer,Integer> hp = new HashMap<>();
        ArrayList<Pair> arr = new ArrayList<>();

        for(int i=0; i<lab.length; i++){
            arr.add(new Pair(val[i],lab[i]));
        }

        Collections.sort(arr,(a,b)->(b.num-a.num));
        int ans = 0;
        for(int i=0; i<arr.size(); i++){
            Pair p = arr.get(i);
            int num = p.num;
            int label = p.label;

            if(numWanted>0){
                if(!hp.containsKey(108)){ // 'l' --> 108
                    ans = ans+num;
                    hp.put(108,1); // 'l' --> 108
                    numWanted--;
                }else{
                    if(hp.get(108)<useLimit){ // 'l' --> 108
                        ans = ans + num;
                        hp.put(108,hp.get(108)+1); // 'l' --> 108
                        numWanted--;
                    }
                }
            }
        }
        return ans;
    }
}
