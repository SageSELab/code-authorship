class Solution {

    class Pair {
        int num;
        int label;

        Pair(int num, int label) {
            this.num = num;
            this.label = label;
            System.out.println("Initialized Pair with num: " + num + " and label: " + label);
        }
    }

    public int largestValsFromLabels(int[] val, int[] lab, int numWanted, int useLimit) {
        HashMap < Integer, Integer > hp = new HashMap < > ();
        System.out.println("Initialized HashMap hp");
        ArrayList < Pair > arr = new ArrayList < > ();
        System.out.println("Initialized ArrayList arr");

        for (int i = 0; i < lab.length; i++) {
            arr.add(new Pair(val[i], lab[i]));
            System.out.println("Added Pair to arr with num: " + val[i] + " and label: " + lab[i]);
        }

        Collections.sort(arr, (a, b) -> (b.num - a.num));
        System.out.println("Sorted arr based on num");

        int ans = 0;
        System.out.println("Initialized ans variable with 0");
        for (int i = 0; i < arr.size(); i++) {
            Pair p = arr.get(i);
            System.out.println("Retrieved Pair with num: " + p.num + " and label: " + p.label);
            int num = p.num;
            System.out.println("Assigned num variable with value: " + num);
            int label = p.label;
            System.out.println("Assigned label variable with value: " + label);

            if (numWanted > 0) {
                if (!hp.containsKey(label)) {
                    ans = ans + num;
                    System.out.println("Added value of num to ans: " + num);
                    hp.put(label, 1);
                    System.out.println("Added label to hp with count 1");
                    numWanted--;
                    System.out.println("Decremented numWanted count");
                } else {
                    if (hp.get(label) < useLimit) {
                        ans = ans + num;
                        System.out.println("Added value of num to ans: " + num);
                        hp.put(label, hp.get(label) + 1);
                        System.out.println("Incremented count for label in hp");
                        numWanted--;
                        System.out.println("Decremented numWanted count");
                    }
                }
            }
        }
        return ans;
    }
}
