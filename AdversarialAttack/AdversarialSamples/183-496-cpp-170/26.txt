class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans;
        for(int i=0;i<nums1.size();i++){
            ans.push_back(findNextGreater(nums1[i], nums2));
        }
        return ans;
    }

    int findNextGreater(int num, vector<int>& nums) {
        int max = -1;
        for(int j=0; j<nums.size(); j++){
            if(num == nums[j]){
                for(int k=j+1; k<nums.size(); k++){
                    if(nums[k] > num){
                        max = nums[k];
                        break;
                    }
                }
                break;
            }
        }
        return max;
    }
};
