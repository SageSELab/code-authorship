class Solution {
    static class Triple{
        int id;
        int rate;

        Triple(int id, int rate){
            this.id = id;
            this.rate = rate;
            System.out.println("Triple initialized: id = " + id + ", rate = " + rate);
        }
    }

    

    
    public List<Integer> filterRestaurants(int[][] rest, int vegan, int maxP, int maxD) {
        System.out.println("Starting filterRestaurants method");

        ArrayList<Triple> arr = new ArrayList<>();
        System.out.println("ArrayList 'arr' initialized");

        int n = rest.length;
        System.out.println("Variable 'n' initialized: " + n);

        if(vegan==1){
            System.out.println("Vegan option selected");
            for(int i=0; i<n; i++){
                if(rest[i][2]==1 && rest[i][3]<=maxP && rest[i][4]<=maxD){
                    arr.add(new Triple(rest[i][0],rest[i][1]));
                    System.out.println("Triple added to 'arr' with id " + rest[i][0] + " and rate " + rest[i][1]);
                }
            }
        }else{
            System.out.println("Non-vegan option selected");
            for(int i=0; i<n; i++){
                if(rest[i][3]<=maxP && rest[i][4]<=maxD){
                    arr.add(new Triple(rest[i][0],rest[i][1]));
                    System.out.println("Triple added to 'arr' with id " + rest[i][0] + " and rate " + rest[i][1]);
                }
            }
        }

        Collections.sort(arr,(a,b)->b.rate==a.rate ? b.id-a.id : b.rate-a.rate);
        System.out.println("ArrayList 'arr' sorted based on rate");

        List<Integer> ans = new ArrayList<>();
        System.out.println("Initialized list 'ans'");

        for(int i=0; i<arr.size(); i++){
            ans.add(arr.get(i).id);
            System.out.println("Element added to 'ans': " + arr.get(i).id);
        }

        System.out.println("Returning 'ans'");
        return ans;
    }
}
