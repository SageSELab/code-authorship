class Solution {
public:
    vector<string> nums ;
    void traverse(TreeNode * root){
        if(!root) return ;
        traverse(root->left) ;
        nums.push_back(to_string(root->val)) ;
        traverse(root->right) ;
        return ;
    }
    
    TreeNode * makeTree(string s , string e){
        if(s > e) return nullptr ;
        string m = to_string((stoi(s) + stoi(e)) / 2) ;
        TreeNode * root = new TreeNode(stoi(nums[stoi(m)]) ;
        
        root->left = makeTree(s,m - 1) ;
        root->right = makeTree(m + 1,e) ;
        
        return root ;
    }
    
    TreeNode* balanceBST(TreeNode* root) {
        traverse(root) ;
        return makeTree("0", to_string(nums.size() - 1)) ;
    }
};
