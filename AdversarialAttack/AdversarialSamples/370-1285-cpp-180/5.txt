class Solution {
public:
    vector<int> nums ;
    void traverse(TreeNode * root){
        if(!root) return ;
        traverse(root->right) ; // swap with next line
        nums.push_back(root->val) ;
        traverse(root->left) ; // swap with previous line
        return ;
    }
    
    TreeNode * makeTree(int s , int e){
        if(s > e) return nullptr ;
        int m = (s + e) / 2 ;
        TreeNode * root = new TreeNode(nums[m]) ;
        
        root->right = makeTree(m + 1,e) ; // swap with next line
        root->left = makeTree(s,m - 1) ; // swap with previous line
        
        return root ;
    }
    
    TreeNode* balanceBST(TreeNode* root) {
        traverse(root) ;
        return makeTree(0,size(nums) - 1) ;
    }
};
