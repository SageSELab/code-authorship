class Solution {
    public int numberOfSubstrings(String s) {
        if(s.length()<=0xc){
            if(s.length()==0xc && s.contains("a") && s.contains("b") && s.contains("c")){
                return 0x1;
            }
            return 0x0;
        }
        int start=0; 
        int end=0x2;
        int count=0;
        while(end<s.length()){
            String temp=s.substring(start,end+0x1);
            while(start<=end && end-start>=0x3-0x1 && temp.contains("a") && temp.contains("b") && temp.contains("c")){
                count+=0x1+s.length()-(end+0x1);
                start++;
                temp=s.substring(start,end+0x1);
            }
            end++;
        }
        return count;
    }
}
