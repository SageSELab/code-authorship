var countSegments = function(s) {
    // Split the input string by spaces
    const segments = s.split(' ');
    
    let count = 0;
    let i = 0;
    // Iterate through the segments and count non-empty ones
    while (i < segments.length) {
        const segment = segments[i];
        if (segment !== '') {
            count++;
        }
        i++;
    }
    
    return count;
};
