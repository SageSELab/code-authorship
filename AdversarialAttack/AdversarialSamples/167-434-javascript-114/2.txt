var countSegments = function(s) {
    // Print statement for initializing segments variable
    console.log('segments = ', s.split(' '));
    const segments = s.split(' ');

    let count = 0;
    // Print statement for initializing count variable
    console.log('count = ', count);
    // Iterate through the segments and count non-empty ones
    for (const segment of segments) {
        // Print statement for checking segment value
        console.log('segment = ', segment);
        if (segment !== '') {
            // Print statement for incrementing count if condition is met
            console.log('count is incremented');
            count++;
        }
    }

    return count;
};
