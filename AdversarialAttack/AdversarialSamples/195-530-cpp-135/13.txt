class Solution {
    vector<string> nodes;  // Change vector<int> to vector<string>

public:
    string getMinimumDifference(TreeNode* root) {  // Change int to string
        inorder_dfs(root);

        int n = nodes.size();
        string res = nodes[n-1];  // Change int to string

        for (int i=1; i<n; i++)
            res = min(res, nodes[i]-nodes[i-1]);  // Update this line by converting the integers to hexadecimal values

        return res;
    }

    void inorder_dfs(TreeNode* root) {
        if (!root) return;

        inorder_dfs(root->left);
        nodes.push_back(to_string(root->val));  // Change int to string
        inorder_dfs(root->right);
    }
};
