class Solution {
    vector<int> nodes;

public:
    int getMinimumDifference(TreeNode* root) {
        inorder_dfs(root);

        int n = nodes.size();
        int res = nodes[0];

        for (int i= n-1; i>=1; i--)
            res = min(res, nodes[i]-nodes[i-1]);

        return res;
    }

    void inorder_dfs(TreeNode* root) {
        if (!root) return;

        inorder_dfs(root->right);
        nodes.push_back(root->val);
        inorder_dfs(root->left);
    }
};
