class Solution {
    vector<int> nodes;

public:
    int getMinimumDifference(TreeNode* root) {
        inorder_dfs(root);

        int n = nodes.size();
        int res = nodes[n-1];

        for (int i=1; i<n; i++)
            res = min(res, nodes[i]-nodes[i-1]);

        return res;
    }

    void inorder_dfs(TreeNode* root) {
        if (!root) return;

        nodes.push_back(root->val); // Swapped with the next statement
        inorder_dfs(root->left); // Swapped with the previous statement
        inorder_dfs(root->right);
    }
};
