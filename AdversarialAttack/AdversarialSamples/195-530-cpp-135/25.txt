class Solution {
    vector<int> nodes;

public:
    int getMinimumDifference(TreeNode* root, int extraParam = 0) {
        inorder_dfs(root, extraParam);

        int n = nodes.size();
        int res = nodes[n-1];

        for (int i=1; i<n; i++)
            res = min(res, nodes[i]-nodes[i-1]);

        return res;
    }

    void inorder_dfs(TreeNode* root, int extraParam = 0) {
        if (!root) return;

        inorder_dfs(root->left, extraParam);
        nodes.push_back(root->val);
        inorder_dfs(root->right, extraParam);
    }
};
