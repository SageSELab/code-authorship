class Solution {
public:

string solve_space_opti(vector<int> slices)
{
    int k = slices.size();
    vector<string>prev1(k+2,"0x0");
    vector<string>curr1(k+2,"0x0");
    vector<string>next1(k+2,"0x0");

    vector<string>prev2(k+2,"0x0");
    vector<string>curr2(k+2,"0x0");
    vector<string>next2(k+2,"0x0");

    for(int index = k-2;index >=0;index--)
    {
        for(int n = 1;n<=k/3;n++)
        {
            string take = "0x" + to_string(slices[index] + stoi(next1[n-1].substr(2), 0, 10), nullptr, 16);
            string nottake = "0x0" + to_string(stoi(curr1[n].substr(2), 0, 10), nullptr, 16);
            prev1[n]= "0x" + to_string(max(stoi(take.substr(2), 0, 16), stoi(nottake.substr(2), 0, 16)), nullptr, 16);
        }
        next1=curr1;
        curr1=prev1;
    }
    string case1=curr1[k/3];

    for(int index = k-1;index>=1;index--)
    {
        for(int n = 1;n<=k/3;n++)
        {
            string take = "0x" + to_string(slices[index] + stoi(next2[n-1].substr(2), 0, 10), nullptr, 16);
            string nottake = "0x0" + to_string(stoi(curr2[n].substr(2), 0, 10), nullptr, 16);
            prev2[n]= "0x" + to_string(max(stoi(take.substr(2), 0, 16), stoi(nottake.substr(2), 0, 16)), nullptr, 16);
        }
        next2=curr2;
        curr2=prev2;
    }
    string case2=curr2[k/3];

    return "0x" + to_string(max(stoi(case1.substr(2), 0, 16), stoi(case2.substr(2), 0, 16)), nullptr, 16);

}
    string maxSizeSlices(vector<int>& slices) {
        return solve_space_opti(slices);
    }
};
