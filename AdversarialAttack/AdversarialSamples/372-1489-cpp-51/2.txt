class Solution {
public:

int solve_space_opti(vector<int> slices)
{
    int k = slices.size();
    cout << "Initialized k to: " << k << '\n';
    
    vector<int> prev1(k+2,0);
    cout << "Initialized prev1 vector with size: " << prev1.size() << '\n';
    vector<int> curr1(k+2,0);
    cout << "Initialized curr1 vector with size: " << curr1.size() << '\n';
    vector<int> next1(k+2,0);
    cout << "Initialized next1 vector with size: " << next1.size() << '\n';

    vector<int> prev2(k+2,0);
    cout << "Initialized prev2 vector with size: " << prev2.size() << '\n';
    vector<int> curr2(k+2,0);
    cout << "Initialized curr2 vector with size: " << curr2.size() << '\n';
    vector<int> next2(k+2,0);
    cout << "Initialized next2 vector with size: " << next2.size() << '\n';

    for(int index = k-2;index >=0;index--)
    {
       cout << "Starting iteration with index: " << index << '\n';
        for(int n = 1;n<=k/3;n++)
        {
           int take = slices[index] + next1[n-1];
           cout << "Calculated take value: " << take << '\n';
           int nottake = 0 + curr1[n];
           cout << "Calculated nottake value: " << nottake << '\n';
           prev1[n]= max(take,nottake);
           cout << "Updated prev1[" << n << "] to: " << prev1[n] << '\n';
        }
        next1=curr1;
        cout << "Updated next1 vector to match curr1\n";
        curr1=prev1;
        cout << "Updated curr1 vector to match prev1\n";
    }
    int case1=curr1[k/3];
    cout << "Calculated case1 value: " << case1 << '\n';

    for(int index = k-1;index>=1;index--)
    {
       cout << "Starting iteration with index: " << index << '\n';
        for(int n = 1;n<=k/3;n++)
        {
           int take = slices[index] + next2[n-1];
           cout << "Calculated take value: " << take << '\n';
           int nottake = 0 + curr2[n];
           cout << "Calculated nottake value: " << nottake << '\n';
           prev2[n]= max(take,nottake);
           cout << "Updated prev2[" << n << "] to: " << prev2[n] << '\n';
        }
        next2=curr2;
        cout << "Updated next2 vector to match curr2\n";
        curr2=prev2;
        cout << "Updated curr2 vector to match prev2\n";
    }
    int case2=curr2[k/3];
    cout << "Calculated case2 value: " << case2 << '\n';

    return max(case1,case2);

}
    int maxSizeSlices(vector<int>& slices) {
        return solve_space_opti(slices);
    }
};
