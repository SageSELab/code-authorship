class Solution {
    public int findMaxLength(int[] nums) {
        if (nums == null || nums.length == 0) { // Base Case
            return 0;
        }
        // Converting all 0 to -1
        for(int i = 0; i < nums.length; i++){
            if(nums[i] == 0) {
                nums[i] = -1;
                System.out.println("nums[" + i + "] initialized with value: " + nums[i]);
            }
        }
        int sum = 0; // current
        System.out.println("sum initialized with value: " + sum);
        int max = 0; // final-ans
        System.out.println("max initialized with value: " + max);
        Map<Integer, Integer> map = new HashMap<>();
        System.out.println("map initialized");
        map.put(0, -1); // put reference in the starting of 0 & -1, as i have tell you in the starting  
        System.out.println("map updated with key: 0, value: -1");
        for(int i = 0; i < nums.length; i++){
            sum += nums[i]; // cumulative sum
            System.out.println("sum updated to: " + sum);
            if(map.containsKey(sum)){ // if cumulative sum key :- 0, -1, 1 already present
                int last = map.get(sum); // we get it's value
                System.out.println("last value obtained from map: " + last);
                max = Math.max(max, i - last); // and update max
                System.out.println("max updated to: " + max);
            }
            else{ // if it's not present then create it's key-value pair
                map.put(sum, i);
                System.out.println("map updated with key: " + sum + ", value: " + i);
            }
        }
        return max; // finally return it
    }
}
