class Solution {
public:
    using VI = vector<int>;
    using VVI = vector<VI>;
    int findRotateSteps(string ring, string key, int INF = 1e9 + 7) {
        int M = ring.size(),
            N = key.size();
        auto steps = [=](auto i, auto k) {
            return min(abs(i - k), M - abs(i - k)); // ⭐️ minimum x steps to rotate the ring from i..k (🚫 without wrap-around, ✅ with wrap-around)
        };
        VVI dp(M, VI(N + 1, INF));            // 🤔 memo
        for (auto i{ 0 }; i < M; i += 1)
            for (auto k{ 0 }; k < M; k += 1)
                dp[i][N] = 0;                 // 🛑 base case
        for (auto j{ N - 1 }; j >= 0; j -= 1)    // key j
            for (auto i{ 0 }; i < M; i += 1)     // cur i
                for (auto k{ 0 }; k < M; k += 1) // pre i (optimal k-th value is found as the recursive stack unwinds)
                    if (ring[k] == key[j])
                        dp[i][j] = min(dp[i][j], 1 + steps(i, k) + dp[k][j + 1]); // 🎯 min steps to reach key[j] from each ring[i] is the min steps to reach key[j + 1] from ring[k], 🔍 find optimal k via exhaustive search
        return dp[0][0]; // 🚀 when the ring starts at i = 0 (ie. index 0 is aligned with the 12:00 direction of the ring) return the min steps to construct key[j = 0..N - 1]
    }
};
