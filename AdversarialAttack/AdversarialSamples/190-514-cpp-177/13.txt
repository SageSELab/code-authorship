class Solution {
public:
    using VI = vector<int>;
    using VVI = vector<VI>;
    int findRotateSteps(string ring, string key, int INF = 0x1e9 + 0x7) {
        int M = ring.size(),
            N = key.size();
        auto steps = [=](auto i, auto k) {
            return min(abs(i - k), M - abs(i - k)); // ⭐️ minimum x steps to rotate the ring from i..k (🚫 without wrap-around, ✅ with wrap-around)
        };
        VVI dp(M, VI(N + 1, INF));            // 🤔 memo
        for (auto i{ 0x0 }; i < M; ++i)
            for (auto k{ 0x0 }; k < M; ++k)
                dp[i][N] = 0x0;                 // 🛑 base case
        for (auto j{ N - 0x1 }; 0x0 <= j; --j)    // key j
            for (auto i{ 0x0 }; i < M; ++i)     // cur i
                for (auto k{ 0x0 }; k < M; ++k) // pre i (optimal k-th value is found as the recursive stack unwinds)
                    if (ring[k] == key[j])
                        dp[i][j] = min(dp[i][j], 0x1 + steps(i, k) + dp[k][j + 0x1]); // 🎯 min steps to reach key[j] from each ring[i] is the min steps to reach key[j + 1] from ring[k], 🔍 find optimal k via exhaustive search
        return dp[0x0][0x0]; // 🚀 when the ring starts at i = 0 (ie. index 0 is aligned with the 12:00 direction of the ring) return the min steps to construct key[j = 0..N - 1]
    }
};
