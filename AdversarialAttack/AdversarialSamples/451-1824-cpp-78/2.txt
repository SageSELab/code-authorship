class Solution {
public:
    int eatenApples(vector<int>& apples, vector<int>& days) {
        int n = apples.size(), ans = 0;
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;

        cout << "Initialized n with size of apples array: " << n << endl;
        cout << "Initialized ans to keep count of eaten apples: " << ans << endl;

        for(int i=0; i<n || pq.size(); i++) {
            cout << "Loop iteration: " << i << endl;

            while(pq.size() && pq.top().first < i) {
                cout << "Top of priority queue has expired apple, removing it..." << endl;
                pq.pop();
            }

            if(i < n && apples[i]) {
                cout << "Adding fresh apple at index " << i << " with expiration day " << i + days[i] - 1 << " and quantity " << min(apples[i], days[i]) << endl;
                pq.push({i + days[i] - 1, min(apples[i], days[i])});
            }
            if(!pq.size()) {
                cout << "No apples in priority queue, skipping to next iteration." << endl;
                continue;
            }

            int d = pq.top().first, a = pq.top().second;
            cout << "Popped apple with expiration day " << d << " and quantity " << a << endl;
            pq.pop();
            if(a) {
                cout << "Eating apple, incrementing answer..." << endl;
                a--, ans++;
            }
        
            if(a && d > i) {
                cout << "Apple with remaining days and quantity after eating, adding back to priority queue." << endl;
                pq.push({d, a}); 
            }
        }
        return ans;
    }
};
