class Solution {
public:
    int eatenApples(vector<int>& applesVector, vector<int>& daysVector) {
        int n = applesVector.size(), ans = 0;
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;

        for(int i=0; i<n || pq.size(); i++) {
            while(pq.size() && pq.top().first < i) pq.pop();

            if(i < n && applesVector[i]) pq.push({i + daysVector[i] - 1, min(applesVector[i], daysVector[i])});
            if(!pq.size()) continue;

            int d = pq.top().first, a = pq.top().second;
            pq.pop();
            if(a) a--, ans++;

            if(a && d > i) pq.push({d, a});
        }
        return ans;
    }
};
