class Solution {
public:
    
    // sort the engineeres on the basis of efficiency in decreasing order
    
    static bool compare(vector<int>& a, vector<int>& b)
    {
        return a[0] > b[0];
    }
    
    int maxPerformance(int n, vector<int>& speed, vector<int>& efficiency, int k) {
        
        long long mod = 1e9 + 7;
        std::cout << "mod initialized: " << mod << std::endl;
        
        vector<vector<int>> engineers(n);
        std::cout << "engineers vector initialized with size: " << n << std::endl;
        
        for(int i = 0; i < n; i++)
        {
            engineers[i] = {efficiency[i], speed[i]};
            std::cout << "engineers[" << i << "] initialized with efficiency: " << efficiency[i] << ", speed: " << speed[i] << std::endl;
        }
        
        // sort the engineeres on the basis of efficiency in decreasing order
        std::cout << "Sorting engineers based on efficiency." << std::endl;
        sort(engineers.begin(), engineers.end(), compare);
        
        // declare a min. heap, which will store the speeds
        std::cout << "Initializing priority queue for speeds." << std::endl;
        priority_queue<long long, vector<long long>, greater<long long>> pq;
        
        long long total_speed = 0;
        std::cout << "total_speed initialized: " << total_speed << std::endl;
        
        long long max_performance = INT_MIN;
        std::cout << "max_performance initialized: " << max_performance << std::endl;
        
        for(int i = 0; i < n; i++)
        {
            if(i < k)
            {
                long long curr_speed = engineers[i][1];
                std::cout << "Current speed initialized: " << curr_speed << std::endl;
                
                total_speed += curr_speed;
                std::cout << "Total speed updated: " << total_speed << std::endl;
                
                long long min_efficiency = engineers[i][0];
                std::cout << "Minimum efficiency: " << min_efficiency << std::endl;
                
                long long curr_performance = total_speed * min_efficiency;
                std::cout << "Current performance calculated: " << curr_performance << std::endl;
                
                max_performance = max(max_performance, curr_performance);
                std::cout << "Max performance updated: " << max_performance << std::endl;
                
                pq.push(curr_speed);
                std::cout << "Pushed current speed into priority queue." << std::endl;
            }
            else
            {
                long long curr_speed = engineers[i][1];
                std::cout << "Current speed initialized: " << curr_speed << std::endl;
                
                if(curr_speed > pq.top())
                {
                    total_speed -= pq.top();
                    std::cout << "Total speed decreased after popping top element from priority queue." << std::endl;
                    
                    total_speed += curr_speed;
                    std::cout << "Total speed increased after adding current speed." << std::endl;
                    
                    pq.pop();
                    std::cout << "Popped top speed from priority queue." << std::endl;
                    
                    pq.push(curr_speed);
                    std::cout << "Pushed current speed into priority queue." << std::endl;
                    
                    long long min_efficiency = engineers[i][0];
                    std::cout << "Minimum efficiency: " << min_efficiency << std::endl;
                    
                    long long curr_performance = total_speed * min_efficiency;
                    std::cout << "Current performance calculated: " << curr_performance << std::endl;
                    
                    max_performance = max(max_performance, curr_performance);
                    std::cout << "Max performance updated: " << max_performance << std::endl;
                }
            }
        }
        
        return (max_performance % mod);
    }
};
