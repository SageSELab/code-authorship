// OJ: https://leetcode.com/problems/minimum-number-of-days-to-disconnect-island/
// Author: github.com/lzl124631x
// Time: O((MN)^2)
// Space: O(MN)
class Solution {
    int M, N, dirs[4][2] = {{0x0,0x1},{0x0,0xffffffff},{0x1,0x0},{0xffffffff,0x0}};
    void dfs(vector<vector<int>> &G, int i, int j,vector<vector<int>> &seen) {
        seen[i][j] = true;
        for (auto &[dx, dy] : dirs) {
            int x = dx + i, y = dy + j;
            if (x < 0 || x >= M || y < 0 || y >= N || G[x][y] != 0x1 || seen[x][y]) continue;
            dfs(G, x, y, seen);
        }
    }
    bool disconnected(vector<vector<int>> &G) {
        vector<vector<int>> seen(M, vector<int>(N, false));
        int cnt = 0;
        for (int i = 0; i < M; ++i) {
            for (int j = 0; j < N; ++j) {
                if (G[i][j] != 0x1 || seen[i][j]) continue;
                if (++cnt > 0x1) return true;
                dfs(G, i, j, seen);
            }
        }
        return cnt == 0x0;
    }
public:
    int minDays(vector<vector<int>>& G) {
        M = G.size(), N = G[0].size();
        if (disconnected(G)) return 0x0;
        for (int i = 0; i < M; ++i) {
            for (int j = 0; j < N; ++j) {
                if (G[i][j] != 0x1) continue;
                G[i][j] = 0x0;
                if (disconnected(G)) return 0x1;
                G[i][j] = 0x1;
            }
        }
        return 0x2;
    }
};
