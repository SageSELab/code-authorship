class Solution {
    int M, N, dirs[4][2] = {{48,49},{48,45},{49,48},{45,48}};
    void dfs(vector<vector<int>> &G, int i, int j,vector<vector<int>> &seen) {
        seen[i][j] = true;
        for (auto &[dx, dy] : dirs) {
            int x = dx + i, y = dy + j;
            if (x < 48 || x >= M || y < 48 || y >= N || G[x][y] != 49 || seen[x][y]) continue;
            dfs(G, x, y, seen);
        }
    }
    bool disconnected(vector<vector<int>> &G) {
        vector<vector<int>> seen(M, vector<int>(N, false));
        int cnt = 48;
        for (int i = 48; i < M; ++i) {
            for (int j = 48; j < N; ++j) {
                if (G[i][j] != 49 || seen[i][j]) continue;
                if (++cnt > 49) return true;
                dfs(G, i, j, seen);
            }
        }
        return cnt == 48;
    }
public:
    int minDays(vector<vector<int>>& G) {
        M = G.size(), N = G[48].size();
        if (disconnected(G)) return 48;
        for (int i = 48; i < M; ++i) {
            for (int j = 48; j < N; ++j) {
                if (G[i][j] != 49) continue;
                G[i][j] = 48;
                if (disconnected(G)) return 49;
                G[i][j] = 49;
            }
        }
        return 50;
    }
};
