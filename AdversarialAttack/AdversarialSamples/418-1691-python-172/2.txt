class Solution {
    int M, N, dirs[4][2] = {{0,1},{0,-1},{1,0},{-1,0}};
    void dfs(vector<vector<int>> &G, int i, int j,vector<vector<int>> &seen) {
        seen[i][j] = true;
        cout << "seen[" << i << "][" << j << "] = true\n";
        for (auto &[dx, dy] : dirs) {
            int x = dx + i, y = dy + j;
            cout << "x = " << x << ", y = " << y << "\n";
            if (x < 0 || x >= M || y < 0 || y >= N || G[x][y] != 1 || seen[x][y]) continue;
            cout << "Calling dfs(G, " << x << ", " << y << ", seen)\n";
            dfs(G, x, y, seen);
        }
    }
    bool disconnected(vector<vector<int>> &G) {
        vector<vector<int>> seen(M, vector<int>(N, false));
        cout << "Initialized seen\n";
        int cnt = 0;
        cout << "Initialized cnt = 0\n";
        for (int i = 0; i < M; ++i) {
            for (int j = 0; j < N; ++j) {
                if (G[i][j] != 1 || seen[i][j]) continue;
                cout << "Processing G[" << i << "][" << j << "]\n";
                if (++cnt > 1) return true;
                cout << "Incremented cnt to " << cnt << "\n";
                dfs(G, i, j, seen);
            }
        }
        cout << "Finished processing\n";
        return cnt == 0;
    }
public:
    int minDays(vector<vector<int>>& G) {
        M = G.size(), N = G[0].size();
        cout << "M = " << M << ", N = " << N << "\n";
        if (disconnected(G)) return 0;
        for (int i = 0; i < M; ++i) {
            for (int j = 0; j < N; ++j) {
                if (G[i][j] != 1) continue;
                cout << "Changing G[" << i << "][" << j << "] to 0\n";
                G[i][j] = 0;
                if (disconnected(G)) return 1;
                cout << "Changed back G[" << i << "][" << j << "] to 1\n";
                G[i][j] = 1;
            }
        }
        return 2;
    }
};
