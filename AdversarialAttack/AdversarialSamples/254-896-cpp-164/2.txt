class Solution {
public:
    TreeNode* lca(TreeNode* root,TreeNode* p,TreeNode* q){
        if(!root){ 
            cout << "Root node is NULL\n";
            return NULL;
        }
        if(root==p || root==q){
            cout << "Found target node\n";
            return root;
        }
        
        TreeNode *left=lca(root->left,p,q);
        TreeNode *right=lca(root->right,p,q);
        
        if(!left){
            cout << "Left subtree does not contain target nodes\n";
            return right;
        }
        if(!right){
            cout << "Right subtree does not contain target nodes\n";
            return left;
        }
        
        return root;
        
    }  
    
    TreeNode* subtreeWithAllDeepest(TreeNode* root) {
        if(!root){
            cout << "Root node is NULL\n";
            return root;
        }
               
        vector<TreeNode*>v;
        
        queue<TreeNode*>q;
        q.push(root);
        
        while(!q.empty()){
            int n=q.size();
            vector<TreeNode*>a;
            
            for(int i=0;i<n;i++){
                auto ptr=q.front();
                q.pop();
                a.push_back(ptr);
                
                if(ptr->left){
                    cout << "Left child added to queue\n";
                    q.push(ptr->left);
                }
                if(ptr->right){
                    cout << "Right child added to queue\n";
                    q.push(ptr->right);
                }
            }
            v=a;
        }
        
        if(v.size()==1){
            cout << "Only one deepest node found\n";
            return v[0];
        }
        TreeNode* ptr=root;
        return lca(ptr,v[0],v[v.size()-1]);
        
    }
};
