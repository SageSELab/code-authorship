class Solution {
public:
    TreeNode* lca(TreeNode* root, TreeNode* p, TreeNode* q){
        return (!root || root == p || root == q) ? root : (!lca(root->left, p, q) ? lca(root->right, p, q) : !lca(root->right, p, q) ? lca(root->left, p, q) : root);
    }

    TreeNode* subtreeWithAllDeepest(TreeNode* root) {
        return !root ? root : lca(root, v[0], v[v.size() - 1]);
    }
};
