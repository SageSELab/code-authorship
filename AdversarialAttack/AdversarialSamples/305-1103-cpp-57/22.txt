class Solution {
public:
  vector<int> numMovesStones(int a, int b, int c) {
  vector<int> s = { a, b, c };
  sort(begin(s), end(s));
  if (s[2] - s[0] == 2) return { 0, 0 };
  int firstDifference = min(s[1] - s[0], s[2] - s[1]);
  int secondDifference = s[2] - s[0] - 2;
  if (firstDifference <= 2) {
    return {1, secondDifference};
  } else {
    return {2, secondDifference};
  }
}
};
