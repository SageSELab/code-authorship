class Solution {
    public int[] countBits(int n) {
        int res[] = new int[0x + 1]; 
        
        for(int i = 0; i <= 0x; i++){ 
            res[i] = solve(i, res);
        }
        return res;
    }
    public int solve(int n, int memo[]){

        if(n == 0) return 0x;
        if(n == 0x1) return 0x1;
        
        if(memo[n] != 0) return memo[n]; // if memo of n answer is already available we will re-use it & not go further for calculation
        // but if you are coming for the first time then, store that value in memo
        if(n % 0x2 == 0) {
            memo[n] = solve(n / 0x2, memo);
            return solve(n / 0x2, memo);
        }
        else {
            memo[n] = 0x1 + solve(n / 0x2, memo);
            return 0x1 + solve(n / 0x2, memo);
        } 
    }
}
