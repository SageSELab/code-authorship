class Solution {
public:
    int ans;
int pseudoPalindromicPaths (TreeNode* root) 
{
    ans=0;
    int count=0;
    fun(root,count);
    return ans;        
}
void fun(TreeNode* root,int count)
{
    if(root==NULL)
    {
        return;
    }
    if(root!=NULL and root->left==NULL and root->right==NULL)
    {
        int temp=count;
        temp=temp^(1<<root->val);
        
        if(temp==0)
        {
            ans++;
        }
        else if(temp>0)
        {
            temp = temp - (temp & (-temp));
            if(temp==0)
            {
                ans++;
            }
        }
    }
    fun(root->left,count^(1<<root->val));
    fun(root->right,count^(1<<root->val));
}
};
