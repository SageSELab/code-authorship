/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int ans;  //this will keep the count of the pusedo palindromic path
int pseudoPalindromicPaths (TreeNode* root) 
{
    ans=0;
    cout << "ans initialized to 0" << endl;
    int count=0;
    cout << "count initialized to 0" << endl;
    fun(root,count);
    cout << "fun called" << endl;
    return ans;        
}
void fun(TreeNode* root,int count)
{
    cout << "fun called with root value: " << root->val << " and count: " << count << endl;
    if(root==NULL)
    {
        cout << "root is NULL, returning" << endl;
        return;
    }
    if(root!=NULL and root->left==NULL and root->right==NULL)
    {
        cout << "reached leaf node" << endl;
        int temp=count;
        cout << "temp initialized to count: " << count << endl;
        temp=temp^(1<<root->val);
        cout << "temp updated to: " << temp << endl;

        if(temp==0)   //even length pasudeo palindromic paht as the value of temp is 0 indicated the bit set in temp is 0
        {
            cout << "found even length pseudo palindromic path" << endl;
            ans++;
            cout << "ans incremented to: " << ans << endl;
        }
        else if(temp>0)   //could be odd length psudeo palindromic path as the value of temp is more than zero we cannot anythig about no of set bit
        {
            temp = temp - (temp & (-temp));
            cout << "temp updated to handle odd length pseudo palindromic path: " << temp << endl;
            if(temp==0)
            {
                cout << "found odd length pseudo palindromic path" << endl;
                ans++;
                cout << "ans incremented to: " << ans << endl;
            }
        }
    }
    fun(root->left,count^(1<<root->val));
    cout << "fun called for left child" << endl;
    fun(root->right,count^(1<<root->val));
    cout << "fun called for right child" << endl;
}
};
