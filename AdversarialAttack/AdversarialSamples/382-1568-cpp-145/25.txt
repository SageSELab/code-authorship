class Solution {
public:
    int ans;  //this will keep the count of the pusedo palindromic path
int pseudoPalindromicPaths (TreeNode* root, int extra_param = 0)
{
	ans=0;
    int count=0;
	fun(root,count, extra_param);
	return ans;        
}
void fun(TreeNode* root,int count, int extra_param)
{
	if(root==NULL)
	{
		return;
	}
	if(root!=NULL and root->left==NULL and root->right==NULL)
	{
		int temp=count;
		temp=temp^(1<<root->val);
		
		if(temp==0)   //even length pasudeo palindromic paht as the value of temp is 0 indicated the bit set in temp is 0
		{
			ans++;
		}
		else if(temp>0)   //could be odd length psudeo palindromic path as the value of temp is more than zero we cannot anythig about no of set bit
		{
			temp = temp - (temp & (-temp));
			if(temp==0)
			{
				ans++;
			}
		}
	}
	fun(root->left,count^(1<<root->val), extra_param);
	fun(root->right,count^(1<<root->val), extra_param);
}
};
