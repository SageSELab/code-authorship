class Solution {
    public boolean checkIfExist(int[] arr) {
        Arrays.sort(arr);
        System.out.println("Array sorted");
        for (int i = 0; i < arr.length; i++) {
            int target = 2 * arr[i];
            System.out.println("Target initialized: " + target);
            int lo = 0, hi = arr.length - 1;
            while (lo <= hi) {
                int mid = lo + (hi - lo) / 2;
                System.out.println("Mid index: " + mid);
                if (arr[mid] == target && mid != i) {
                    System.out.println("Found a matching pair");
                    return true;
                }
                if (arr[mid] < target) {
                    System.out.println("Incrementing low index");
                    lo = mid + 1;
                }
                else {
                    System.out.println("Decrementing high index");
                    hi = mid - 1;
                }
            }
        }

        return false;
    }
}

// TC: O(n * logn), SC: O(1)
