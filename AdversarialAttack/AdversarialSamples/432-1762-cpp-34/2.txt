class Solution {
public:
    int furthestBuilding(vector<int>& H, int B, int L) {
        int len = H.size() - 1;
        cout << "len initialized: " << len << endl;
        priority_queue<int> pq;
        cout << "Priority queue initialized" << endl;
        for (int i = 0; i < len; i++) {
            cout << "Loop iteration: " << i << endl;
            int diff = H[i+1] - H[i];
            cout << "Difference calculated: " << diff << endl;
            if (diff > 0) {
                cout << "Difference is greater than 0" << endl;
                if (L) {
                    cout << "L is non-zero" << endl;
                    pq.push(-diff), L--;
                    cout << "Pushed to priority queue, L decremented" << endl;
                } else if (!pq.empty() && diff > -pq.top()) {
                    cout << "Priority queue not empty and diff is greater than top element" << endl;
                    pq.push(-diff), B += pq.top(), pq.pop();
                    cout << "Pushed to priority queue, B updated, popped top element" << endl;
                } else {
                    cout << "Else condition for diff > 0" << endl;
                    B -= diff;
                    cout << "B updated" << endl;
                }
                if (B < 0) {
                    cout << "B is less than 0, returning current index" << endl;
                    return i;
                }
            }
        }
        cout << "End of loop reached, returning len" << endl;
        return len;
    }
};
