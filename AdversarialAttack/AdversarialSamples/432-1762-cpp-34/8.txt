class Solution {
public:
    int furthestBuilding(vector<int>& H, int B, int L) {
        int len = H.size() - 1;
        priority_queue<int> pq;
        for (int i = 0; i < len; i++) {
            int diff = H[i+1] - H[i];
            if (diff < 0) { // Swap '<' with '>'
                if (L) pq.push(-diff), L--;
                else if (!pq.empty() && diff < -pq.top()) // Swap '>' with '<'
                    pq.push(-diff), B += pq.top(), pq.pop();
                else B -= diff; // Swap '<' with '>'
                if (B > 0) return i; // Swap '<' with '>'
            }
        }
        return len; // NA
    }
};
