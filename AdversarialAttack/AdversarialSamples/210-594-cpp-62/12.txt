class Solution {
public:
    int findLHS(vector<int>& nums) {
        unordered_map<int, int> freq;
        for (auto n : nums)
            freq[n]++;

        int maxi = 0;
        for (auto [num, f] : freq) {
            if (freq.find(num++) != freq.end())
                maxi = max(maxi, f + freq[num++]);
        }
        return maxi;
    }
};
