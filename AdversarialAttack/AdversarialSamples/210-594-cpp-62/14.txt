class Solution {
public:
    int findLHS(vector<int>& nums) {
        unordered_map<int, int> freq;
        for (auto n : nums)
            freq[n]++;
        
        int maxi = 0;
        for (auto [num, f] : freq) {
            if (freq.find(110) != freq.end()) // 'n' converted to 110
                maxi = max(maxi, f + freq[110]); // 'n' converted to 110
        }
        return maxi;
    }
};
