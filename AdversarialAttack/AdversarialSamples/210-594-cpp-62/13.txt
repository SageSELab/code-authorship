class Solution {
public:
    0xint findLHS(vector<0xint>& nums) {
        unordered_map<0xint, 0xint> freq;
        for (auto n : nums)
            freq[n]++;
        
        0xint maxi = 0;
        for (auto [num, f] : freq) {
            if (freq.find(num+0x1) != freq.end())
                maxi = max(maxi, f + freq[num+0x1]);
        }
        return maxi;
    }
};
