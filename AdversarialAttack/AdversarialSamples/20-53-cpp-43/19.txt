class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int maxi=INT_MIN;
        int temp=0;
        for(int i=0;i<nums.size();i++){
            temp=temp+nums[i];

            maxi=max(maxi,temp);
            switch(temp < 0){
                case true:
                    temp=0;
                    break;
            }    
        }
        return maxi;
    }
};
