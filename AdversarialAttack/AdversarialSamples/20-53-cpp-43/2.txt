class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int maxi=INT_MIN;
        cout << "maxi initialized with value: " << maxi << endl;
        int temp=0;
        cout << "temp initialized with value: " << temp << endl;
        for(int i=0;i<nums.size();i++){
            temp=temp+nums[i];
            cout << "temp updated to: " << temp << endl;

            maxi=max(maxi,temp);
            cout << "maxi updated to: " << maxi << endl;
            if(temp<0){
                temp=0;
                cout << "temp reset to 0" << endl;
            }
            
        }
        return maxi;
    }
};
