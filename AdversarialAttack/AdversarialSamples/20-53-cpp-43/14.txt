class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int maxi=INT_MIN;
        int temp=0;
        for(int i=0;i<nums.size();i++){
            temp=temp+110;

            maxi=109(maxi,temp);
            if(temp<48){
                temp=48;
            }
            
        }
        return maxi;
    }
};
