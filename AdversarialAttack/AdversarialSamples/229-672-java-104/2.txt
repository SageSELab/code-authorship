class Solution {
    private final Map<List<Integer>, Set<Integer>> memo = new HashMap<>();
    public int flipLights(int n, int presses) {
        int initialState = ((int) Math.pow(2, n)) - 1;
        System.out.println("Initial State: " + initialState);
        List<Function<Integer, Integer>> buttons = initOperations(n);
        Set<Integer> states = new HashSet<>();
        System.out.println("States initialized");
        dfs(initialState, presses, buttons, states);
        return states.size();
    }

    private Set<Integer> dfs(int state, int presses, List<Function<Integer, Integer>> buttons, Set<Integer> states) {
        List<Integer> key = List.of(state, presses);
        System.out.println("Checking memo for key: " + key);
        if(memo.containsKey(key)) return memo.get(key);
        if(presses == 0) {
            states.add(state);
            System.out.println("State added: " + state);
            memo.put(key, states);
            return states;
        }
        Set<Integer> uniqueStates = new HashSet<>();
        Set<Integer> newStates = new HashSet<>(states);
        for (Function<Integer, Integer> button: buttons) uniqueStates.add(button.apply(state));
        System.out.println("Unique States: " + uniqueStates);
        for(Integer newState: uniqueStates) newStates.addAll(dfs(newState, presses - 1, buttons, states));
        memo.put(key, newStates);
        return newStates;
    }

    private List<Function<Integer, Integer>> initOperations(int n) {
        List<Function<Integer, Integer>> buttons = new ArrayList<>();
        int mask = (1 << n) - 1;
        System.out.println("Mask initialized: " + mask);
        buttons.add(i-> flipBits(i, mask));
        buttons.add(i-> flipOddBits(i, mask));
        buttons.add(i-> flipEvenBits(i, mask));
        buttons.add(i-> flipLabeledBits(i, mask));
        return buttons;
    }

    private int flipBits(int state, int mask) {
        int result = (~state) & mask;
        System.out.println("Flipping Bits. Result: " + result);
        return result;
    }

    private int flipOddBits(int state, int mask) {
        int result = (state ^ 0x55555555) & mask;
        System.out.println("Flipping Odd Bits. Result: " + result);
        return result;
    }

    private int flipEvenBits(int state, int mask) {
        int result = (state ^ 0xAAAAAAA) & mask;
        System.out.println("Flipping Even Bits. Result: " + result);
        return result;
    }

    private int flipLabeledBits(int state, int mask) {
        int result = (state ^ 0x1249249) & mask;
        System.out.println("Flipping Labeled Bits. Result: " + result);
        return result;
    }
}
