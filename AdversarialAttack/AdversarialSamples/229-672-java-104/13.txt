class Solution {
    private final Map<List<String>, Set<String>> memo = new HashMap<>();
    public int flipLights(int n, int presses) {
        String initialState = Integer.toHexString((int) Math.pow(2, n) - 1);
        List<Function<String, String>> buttons = initOperations(n);
        Set<String> states = new HashSet<>();
        dfs(initialState, presses, buttons, states);
        return states.size();
    }

    private Set<String> dfs(String state, int presses, List<Function<String, String>> buttons, Set<String> states) {
        List<String> key = List.of(state, Integer.toHexString(presses));
        if(memo.containsKey(key)) return memo.get(key);
        if(presses == 0) {
            states.add(state);
            memo.put(key, states);
            return states;
        }
        Set<String> uniqueStates = new HashSet<>();
        Set<String> newStates = new HashSet<>(states);
        for (Function<String, String> button : buttons) uniqueStates.add(button.apply(state));
        for(String newState: uniqueStates) newStates.addAll(dfs(newState, presses - 1, buttons, states));
        memo.put(key, newStates);
        return newStates;
    }

    private List<Function<String, String>> initOperations(int n) {
        List<Function<String, String>> buttons = new ArrayList<>();
        int mask = (1 << n) - 1;
        buttons.add(i -> flipBits(i, mask));
        buttons.add(i -> flipOddBits(i, mask));
        buttons.add(i -> flipEvenBits(i, mask));
        buttons.add(i -> flipLabeledBits(i, mask));
        return buttons;
    }

    private String flipBits(String state, int mask) { return Integer.toHexString((~Integer.parseInt(state, 16)) & mask); }

    private String flipOddBits(String state, int mask) { return Integer.toHexString((Integer.parseInt(state, 16) ^ 0x55555555) & mask); }

    private String flipEvenBits(String state, int mask) { return Integer.toHexString((Integer.parseInt(state, 16) ^ 0xAAAAAAA) & mask); }

    private String flipLabeledBits(String state, int mask) { return Integer.toHexString((Integer.parseInt(state, 16) ^ 0x1249249) & mask); }
}
