class Solution {
public:
    bool checkRecord(string s);
    int countOccurrences(string s, char c);
    int maxConsecutiveOccurrences(string s, char c);
};

/*********************************************************/
bool Solution::checkRecord(string s) {
    if (countOccurrences(s, 'A') >= 2 || maxConsecutiveOccurrences(s, 'L') >= 3) {
        return false;
    }
    return true;
}

int Solution::countOccurrences(string s, char c) {
    int count = 0;
    for (char character: s) {
        if (character == c) {
            ++count;
        }
    }
    return count;
}

int Solution::maxConsecutiveOccurrences(string s, char c) {
    int maxCount = 0, count = 0;
    for (char character: s) {
        if (character == c) {
            ++count;
            maxCount = max(maxCount, count);
        } else {
            count = 0;
        }
    }
    return maxCount;
}
