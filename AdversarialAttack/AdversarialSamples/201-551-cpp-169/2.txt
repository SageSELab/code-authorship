class Solution {
public:
    bool checkRecord(string s);
};
/*********************************************************/
bool Solution::checkRecord(string s) {
    int i, size = s.size(), maxL=0, countA=0, countL=0;
    cout << "Initialized i: " << i << endl;
    cout << "Initialized size: " << size << endl;
    cout << "Initialized maxL: " << maxL << endl;
    cout << "Initialized countA: " << countA << endl;
    cout << "Initialized countL: " << countL << endl;
    
    for (i = 0; i < size; ++i) {
        cout << "Updated i: " << i << endl;
        
        if (s[i] == 'L') {
            ++countL;
            cout << "Updated countL: " << countL << endl;
        } else {
            countL = 0;
            cout << "Reset countL to 0" << endl;
        }
        
        if (s[i] == 'A') {
            ++countA;
            cout << "Updated countA: " << countA << endl;
        }
        
        if (maxL < countL) {
            maxL = countL;
            cout << "Updated maxL: " << maxL << endl;
        }
        
        if( countA >= 2 || maxL >= 3) {
            cout << "Returning false" << endl;
            return false;
        }
    }
    
    cout << "Returning true" << endl;
    return true;
}
