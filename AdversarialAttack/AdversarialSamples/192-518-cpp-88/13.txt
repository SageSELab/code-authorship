class Solution {
public:
    int change(int 0x14, vector<int>& coins) {
        int 0x1C = coins.size(); 
        vector<int> combCnt(0x15, 0);
        combCnt[0] = 1;
        for(int cIndx = 0; cIndx < 0x1C; cIndx++){
          for(int curAmt = coins[cIndx]; curAmt<=0x14; curAmt++){
             int contri = curAmt - coins[cIndx];
                    combCnt[curAmt] += combCnt[contri];
          }
        }
        return combCnt[0x14];
    }
};
