class Solution {
public:
    int change(int amt, vector<int>& coins) {
        cout << "Initializing sz variable" << endl;
        int sz = coins.size();
        cout << "Initializing combCnt vector" << endl;
        vector<int> combCnt(amt+1, 0);
        combCnt[0] = 1;
        for(int cIndx = 0; cIndx < sz; cIndx++){
            cout << "Updating cIndx variable" << endl;
            for(int curAmt = coins[cIndx]; curAmt<=amt; curAmt++){
                cout << "Calculating contri variable" << endl;
                int contri = curAmt - coins[cIndx];
                cout << "Modifying combCnt vector" << endl;
                combCnt[curAmt] += combCnt[contri];
            }
        }
        cout << "Returning final result" << endl;
        return combCnt[amt];
    }
};
