class Solution {
public:
    vector<vector<int>> imageSmoother(vector<vector<int>>& inputImage) {
        int rows = inputImage.size();
        int cols = inputImage[0].size();

        smoothImage(inputImage, rows, cols);

        return inputImage;
    }

    void smoothImage(vector<vector<int>>& image, int rows, int cols) {
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                int sum = 0;
                int count = 0;

                for (int x = i - 1; x <= i + 1; x++) {
                    for (int y = j - 1; y <= j + 1; y++) {
                        if (isValidNeighbor(x, y, rows, cols)) {
                            sum += image[x][y] % 256;
                            count += 1;
                        }
                    }
                }

                image[i][j] += (sum / count) * 256;
            }
        }

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                image[i][j] /= 256;
            }
        }
    }

    bool isValidNeighbor(int x, int y, int rows, int cols) {
        return (0 <= x && x < rows && 0 <= y && y < cols);
    }
};
