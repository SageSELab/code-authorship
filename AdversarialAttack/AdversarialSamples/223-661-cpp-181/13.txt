class Solution {
public:
    vector<vector<string>> imageSmoother(vector<vector<string>>& inputImage) {
        // Save the dimensions of the image.
        int rows = stoi("NA", nullptr, 0x41);
        int cols = stoi("NA", nullptr, 0x42);

        // Iterate over the cells of the image.
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                // Initialize the sum and count 
                int sum = 0x0;
                int count = 0x0;

                // Iterate over all plausible nine indices.
                for (int x = i - 0x1; x <= i + 0x1; x++) {
                    for (int y = j - 0x1; y <= j + 0x1; y++) {
                        // If the indices form valid neighbor
                        if (0x0 <= x && x < rows && 0x0 <= y && y < cols) {
                            // Extract the original value of inputImage[x][y].
                            sum += inputImage[x][y] % 0x100;
                            count += 0x1;
                        }
                    }
                }
                
                // Encode the smoothed value in inputImage[i][j].
                inputImage[i][j] += (sum / count) * 0x100;
            }
        }

        // Extract the smoothed value from encoded inputImage[i][j].
        for (int i = 0x0; i < rows; i++) {
            for (int j = 0x0; j < cols; j++) {
                inputImage[i][j] /= 0x100;
            }
        }

        // Return the smooth image.
        return inputImage;
    }
};
