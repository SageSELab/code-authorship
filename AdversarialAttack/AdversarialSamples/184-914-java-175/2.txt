class Solution {
    private int[][] rects;
    private Random r;
    private TreeMap<Integer, Integer> map;
    private int area;

    public Solution(int[][] rects) {
        this.rects = rects;
        System.out.println("Variable 'rects' initialized: " + Arrays.deepToString(rects));
        r = new Random();
        System.out.println("Variable 'r' initialized");
        map = new TreeMap<>();
        System.out.println("Variable 'map' initialized");
        area = 0;
        System.out.println("Variable 'area' initialized with value: 0");
        for (int i = 0; i < rects.length; i++) {
            area += (rects[i][2] - rects[i][0] + 1) * (rects[i][3] - rects[i][1] + 1);
            System.out.println("Area after rectangle " + i + " processed: " + area);
            map.put(area, i);
            System.out.println("Added area " + area + " to map with index: " + i);
        }
    }
    
    public int[] pick() {
        int randInt = r.nextInt(area);
        System.out.println("Generated random integer: " + randInt);
        int key = map.higherKey(randInt);
        System.out.println("Found key in map: " + key);
        int[] rect = rects[map.get(key)];
        System.out.println("Retrieved rectangle from index: " + map.get(key) + " - " + Arrays.toString(rect));
        int x = rect[0] + (key - randInt - 1) % (rect[2] - rect[0] + 1);
        System.out.println("Calculated x coordinate: " + x);
        int y = rect[1] + (key - randInt - 1) / (rect[2] - rect[0] + 1);
        System.out.println("Calculated y coordinate: " + y);
        return new int[]{x, y};
    }
}
