class Solution {
private:
    
    int solveTabulation(string &s, string &t, int p1, int p2)
    {
        vector<vector<int>> dp(p1+1, vector<int> (p2+1, 0));
        cout << "Initialized dp array" << endl;

        for(int i=1;i<=p1;i++)
        {
            for(int j=1;j<=p2;j++)
            {
                int ans = 0;
                cout << "Initialized ans variable" << endl;
                if(s[i-1] == t[j-1])
                    ans = 1 + dp[i-1][j-1];
                else
                    ans = max(dp[i-1][j], dp[i][j-1]);

                cout << "Calculated ans: " << ans << endl;
                
                dp[i][j] = ans;
                cout << "Updated dp[" << i << "][" << j << "] to: " << dp[i][j] << endl;
            }
        }
        cout << "Completed dp calculations" << endl;
        return dp[p1][p2];
    }

public:
    int minInsertions(string s) {
        int n = s.size();
        cout << "Length of string: " << n << endl;
        string t=s;
        cout << "Reversed string 't' to: " << t << endl;
        reverse(t.begin(), t.end());
       
        int longestPalindrome = solveTabulation(s, t, n, n);
        cout << "Found longest palindrome length" << endl;
        
        return n - longestPalindrome;
    }
};
