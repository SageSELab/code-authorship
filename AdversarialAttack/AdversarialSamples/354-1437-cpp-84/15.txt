class Solution {
private:
    
    int solveTabulation(char *s, char *t, int p1, int p2)
    {
        vector<vector<int>> dp(p1+1, vector<int> (p2+1, 0));

        for(int i=1;i<=p1;i++)
        {
            for(int j=1;j<=p2;j++)
            {
                int ans = 0;
                if(s[i-1] == t[j-1])
                    ans = 1 + dp[i-1][j-1];
                else
                    ans = max(dp[i-1][j], dp[i][j-1]);
                
                dp[i][j] = ans;
            }
        }
        return dp[p1][p2];
    }

public:
    int minInsertions(char *s) {
        int n = strlen(s);
        char t[strlen(s)+1];
        strcpy(t, s);
        reverse(t, t + n);
       
        int longestPalindrome = solveTabulation(s, t, n, n);
        return n - longestPalindrome;
    }
};
