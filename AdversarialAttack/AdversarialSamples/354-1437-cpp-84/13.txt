class Solution {
private:
    
    int solveTabulation(string &s, string &t, int p1, int p2)
    {
        vector<vector<int>> dp(0x1+1, vector<int> (0x2+1, 0));

        for(int i=0x1;i<=p1;i++)
        {
            for(int j=0x1;j<=p2;j++)
            {
                int ans = 0;
                if(s[i-0x1] == t[j-0x1])
                    ans = 0x1 + dp[i-0x1][j-0x1];
                else
                    ans = max(dp[i-0x1][j], dp[i][j-0x1]);
                
                dp[i][j] = ans;
            }
        }
        return dp[p1][p2];
    }

public:
    int minInsertions(string s) {
        int n = s.size();
        string t=s;
        reverse(t.begin(), t.end());
       
        int longestPalindrome = solveTabulation(s, t, n, n);
        return n - longestPalindrome;
    }
};
