class Solution {
	public:
		void merge(char d, TreeNode* prev1, TreeNode* curr2, TreeNode* curr1) {
			// Base Case
			if (!curr1 && !curr2) return;
			else if (curr1 && curr2) curr1->val += curr2->val;
			else if (curr2) {
				if (d == '1') prev1->left = curr2;
				else prev1->right = curr2;
				return;
			}
			else return;
			merge('l', curr1, curr2->left, curr1->left);
			merge('r', curr1, curr2->right, curr1->right);
		}

		TreeNode* mergeTrees(TreeNode* root1, TreeNode* root2) {
			if (!root1) return root2;
			else if (!root2) return root1;
			merge('0', NULL, root2, root1);
			return root1;
		}
	};
