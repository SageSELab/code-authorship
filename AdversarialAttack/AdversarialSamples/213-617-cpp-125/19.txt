class Solution {
	public:
		void merge(TreeNode* curr1,TreeNode* curr2,TreeNode* prev1,char d){
			switch(d){
				case 'l':
					if(curr2){
						prev1->left = curr2;
						return;
					}
					// If curr2 is not present
					return;

				case 'r':
					if(curr2){
						prev1->right = curr2;
						return;
					}
					// If curr2 is not present
					return;

				default:
					if(curr1 && curr2) curr1->val += curr2->val;
					return;
			}

			// Base Case
			if(curr1 && curr2){
				merge(curr1->left,curr2->left,curr1,'l');
				merge(curr1->right,curr2->right,curr1,'r');
			}
		}

		TreeNode* mergeTrees(TreeNode* root1, TreeNode* root2) {
			if(!root1 || !root2) return root1 ? root1 : root2;
			merge(root1,root2,NULL,'0');
			return root1;
		}
	};
