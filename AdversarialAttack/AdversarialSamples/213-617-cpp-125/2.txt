class Solution {
	public:
		void merge(TreeNode* curr1,TreeNode* curr2,TreeNode* prev1,char d){
			// Initialization of variables
			if(!curr1 && !curr2) return;
			else if(curr1 && curr2) {
				curr1->val += curr2->val;
				cout << "Updated curr1->val to: " << curr1->val << endl;
			}
			else if(curr2){
				if(d == 'l') {
					prev1->left = curr2;
					cout << "Set prev1->left to: curr2" << endl;
				}
				else {
					prev1->right = curr2;
					cout << "Set prev1->right to: curr2" << endl;
				}
				return ;
			}
			else return;
			merge(curr1->left, curr2->left, curr1, 'l');
			merge(curr1->right, curr2->right, curr1, 'r');
		}

		TreeNode* mergeTrees(TreeNode* root1, TreeNode* root2) {
			// Initialization of variables
			if(!root1) return root2;
			else if(!root2) return root1;
			merge(root1, root2, NULL, '0');
			return root1;
		}
	};
