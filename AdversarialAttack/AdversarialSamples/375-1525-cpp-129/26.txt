class Solution {
public:
    vector<int> processQueries(vector<int>& queries, int m) {
        vector<int> ans = generateVector(m);
        map<int,int> mp = generateMap(ans);
        return processQueriesHelper(queries, mp);
    }

private:
    vector<int> generateVector(int m) {
        vector<int> ans;
        for(int i=1; i<=m; i++)
            ans.push_back(i);
        return ans;
    }

    map<int,int> generateMap(vector<int>& ans) {
        map<int,int> mp;
        for(int i=0; i<ans.size(); i++)
            mp[ans[i]]=i;
        return mp;
    }

    vector<int> processQueriesHelper(vector<int>& queries, map<int,int>& mp) {
        vector<int> v;
        for(int i=0; i<queries.size(); i++) {
            int x = mp[queries[i]];
            v.push_back(x); // storing the index no
            for(auto& it : mp) {
                if(it.second < x) {
                    it.second += 1; // for indexes less than index of queries[i] incrementing by 1
                }
            }
            mp[queries[i]] = 0; // for making the index of queries[i] equal to zero to make it appear at first
        }
        return v;
    }
};
