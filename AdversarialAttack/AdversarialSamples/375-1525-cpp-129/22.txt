class Solution {
public:
    vector<int> processQueries(vector<int>& queries, int m) {
        vector<int> ans;
        for(int i=1;i<=m;i++)
            ans.push_back(i);
        map<int,int> mp;
        for(int i=0;i<ans.size();i++)
            mp[ans[i]]=i;
        vector<int> v;
        for(int i=0;i<queries.size();i++)
        {
           int x=mp[queries[i]];
            v.push_back(x);// storing the index no
            for(auto it:mp)
            {
                if(it.second<x) // for indexes less than index of queries[i] incrementing by 1
                {
                    mp[it.first]+=1;               
                }
            }
            mp[queries[i]]=0; // for making the index of queries[i] equal to zero to make it appear at first
        }
        return v;
        
    }
};
