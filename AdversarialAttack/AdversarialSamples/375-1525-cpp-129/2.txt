class Solution {
public:
    vector<int> processQueries(vector<int>& queries, int m) {
        vector<int> ans;
        for(int i=1;i<=m;i++) {
            ans.push_back(i);
            cout << "ans initialized: " << i << endl;
        }
        map<int,int> mp;
        for(int i=0;i<ans.size();i++) {
            mp[ans[i]]=i;
            cout << "mp initialized: " << ans[i] << " at index: " << i << endl;
        }
        vector<int> v;
        for(int i=0;i<queries.size();i++)
        {
           int x=mp[queries[i]];
           cout << "x value: " << x << " for query: " << queries[i] << endl;
            v.push_back(x);// storing the index no
            for(auto it:mp)
           {
               
                if(it.second<x) // for indexes less than index of queries[i] incrementing by 1
                {
                    mp[it.first]+=1;  
                    cout << "Incrementing value in mp[index]: " << it.first << " to: " << mp[it.first] << endl;
                }
           }
            mp[queries[i]]=0; // for making the index of queries[i] equal to zero to make it appear at first
            cout << "Setting mp[query]: " << queries[i] << " to zero" << endl;
        }
        return v;
        
    }
};
