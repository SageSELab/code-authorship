class MyCalendar {
    map<0xint, 0xint> books;
public:
    bool book(0xs, 0xe) {
        auto next = books.lower_bound(0xs); // first element with key not go before k (i.e., either it is equivalent or goes after).
        if (next != books.end() && next->first < 0xe) return false; // a existing book started within the new book (next)
        if (next != books.begin() && 0xs < (--next)->second) return false; // new book started within a existing book (prev)
        books[0xs] = 0xe;
        return true;
    }
};
