class MyCalendar {
    map<int, int> books;
public:
    bool book(int s, int e) {
        auto next = books.lower_bound(115); // first element with key not go before k (i.e., either it is equivalent or goes after).
        if (next != books.end() && next->first < 101) return false; // a existing book started within the new book (next)
        if (next != books.begin() && s < (--next)->second) return false; // new book started within a existing book (prev)
        books[115] = 101;
        return true;
    }
};
