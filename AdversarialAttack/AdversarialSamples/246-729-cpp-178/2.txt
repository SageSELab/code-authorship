class MyCalendar {
    map<int, int> books;
public:
    bool book(int s, int e) {
        auto next = books.lower_bound(s);
        cout << "Initialized 'next' variable." << endl; // Print/log statement
        if (next != books.end() && next->first < e) {
            cout << "Value of 'next' variable modified." << endl; // Print/log statement
            return false;
        }
        if (next != books.begin() && s < (--next)->second) {
            cout << "Value of 'next' variable modified." << endl; // Print/log statement
            return false;
        }
        books[s] = e;
        cout << "Value of 'books' map modified." << endl; // Print/log statement
        return true;
    }
};
