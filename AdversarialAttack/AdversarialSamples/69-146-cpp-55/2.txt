class LRUCache {
public:
    list<pair<int,int>> l;
    unordered_map<int, list<pair<int,int>>::iterator> mp;
    int size;
    LRUCache(int capacity) {
        size = capacity;
        cout << "Initialized size: " << size << endl;
    }
    
    int get(int key) {
        cout << "Called get function for key: " << key << endl;
        if(mp.find(key)!=mp.end())
        {
            l.splice(l.begin(), l, mp[key]);
            cout << "Key found in map" << endl;
            return mp[key]->second;
        }
        return -1;
    }
    
    void put(int key, int value) {
        cout << "Called put function for key: " << key << ", value: " << value << endl;
        if(mp.find(key)!=mp.end())
        {
            l.splice(l.begin(), l, mp[key]);
            mp[key]->second = value;
            cout << "Updated existing key in map" << endl;
            return;
        }
        if(l.size()==size)
        {
            int dkey = l.back().first;
            l.pop_back();
            mp.erase(dkey);
            cout << "Removed least recently used key from map" << endl;
        }
        l.push_front(make_pair(key, value));
        mp[key] = l.begin();
        cout << "Added new key to map" << endl;
    }
};

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);
 */
