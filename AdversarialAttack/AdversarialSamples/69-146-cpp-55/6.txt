class LRUCache {
public:
    list<pair<int,int>> lruCacheList;
    unordered_map<int, list<pair<int,int>>::iterator> map;
    int cacheSize;
    LRUCache(int capacity) {
        cacheSize = capacity;
    }
    
    int get(int key) {
        if(map.find(key) != map.end())
        {
            lruCacheList.splice(lruCacheList.begin(), lruCacheList, map[key]);
            return map[key]->second;
        }
        return -1;
    }
    
    void put(int key, int value) {
        if(map.find(key) != map.end())
        {
            lruCacheList.splice(lruCacheList.begin(), lruCacheList, map[key]);
            map[key]->second = value;
            return;
        }
        if(lruCacheList.size() == cacheSize)
        {
            int deletedKey = lruCacheList.back().first;
            lruCacheList.pop_back();
            map.erase(deletedKey);
        }
        lruCacheList.push_front(make_pair(key, value));
        map[key] = lruCacheList.begin();
    }
};

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);
 */
