class Solution {
    public List<List<Integer>> levelOrder(Node node) {
        switch (node) {
            case null: return new ArrayList<>();
            default:
                Map<Integer, List<Integer>> map = new TreeMap<>();
                traverseTree(node, map, 0);
                List<List<Integer>> res = new ArrayList<>();
                for (int key : map.keySet()) {
                    res.add(map.get(key));
                }
                return res;
        }
    }

    private void traverseTree(Node node, Map<Integer, List<Integer>> map, int lvl) {
        if (node == null) return;

        lvl++;

        List<Integer> list = map.get(lvl);
        switch (list) {
            case null:
                list = new ArrayList<>();
                list.add(node.val);
                map.put(lvl, list);
                break;
            default:
                list.add(node.val);
                break;
        }

        for (int i = 0; i < node.children.size(); i++) {
            Node child = node.children.get(i);
            traverseTree(child, map, lvl);
        }
    }
}
