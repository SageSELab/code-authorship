class Solution {
    public List<List<Integer>> levelOrder(Node node) {
        if (node == null) {
            System.out.println("Node is null.");
            return new ArrayList<>();
        }
        Map<Integer, List<Integer>> map = new TreeMap<>();
        System.out.println("Initialized map.");
        traverseTree(node, map, 0);
        System.out.println("Traversed tree.");
        List<List<Integer>> res = new ArrayList<>();
        for (int key : map.keySet()) {
            System.out.println("Adding to result list.");
            res.add(map.get(key));
        }
        System.out.println("Returning result.");
        return res;
    }

    private void traverseTree(Node node, Map<Integer, List<Integer>> map, int lvl) {
        System.out.println("Traversing node: " + node.val);
        if (node == null) {
            System.out.println("Node is null. Returning.");
            return;
        }

        lvl++;
        System.out.println("Updated level to: " + lvl);

        List<Integer> list = map.get(lvl);
        System.out.println("Retrieved list for level: " + lvl);
        if (list == null) {
            System.out.println("Creating new list for level: " + lvl);
            list = new ArrayList<>();
            list.add(node.val);
            map.put(lvl, list);
            System.out.println("Added node value to list for level: " + lvl);
        } else {
            list.add(node.val);
            System.out.println("Added node value to existing list for level: " + lvl);
        }

        for (int i = 0; i < node.children.size(); i++) {
            Node child = node.children.get(i);
            System.out.println("Traversing child node: " + child.val);
            traverseTree(child, map, lvl);
        }
    }
}
