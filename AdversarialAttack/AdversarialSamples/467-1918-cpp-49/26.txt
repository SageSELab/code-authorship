class Solution {
public:
    int maximumScore(vector<int>& nums, int k) {
        int n = nums.size();
        if (n == 1) return nums[0];

        vector<int> left = fillArray(nums, n);
        vector<int> right = fillArray(nums, n, true);

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (left[i] < k && right[i] > k) {
                ans = max(ans, nums[i] * (right[i] - left[i] - 1));
            }
        }

        return ans;
    }

private:
    vector<int> fillArray(vector<int>& nums, int n, bool isRight = false) {
        stack<int> st;
        vector<int> result(n);

        for (int i = isRight ? n - 1 : 0; isRight ? (i >= 0) : (i < n); isRight ? i-- : i++) {
            while (!st.empty() && nums[i] <= nums[st.top()]) {
                st.pop();
            }
            result[i] = st.empty() ? (isRight ? n : -1) : st.top();
            st.push(i);
        }

        return result;
    }
};
