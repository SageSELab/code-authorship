class Solution {
public:
    int maximumScore(vector<int>& nums, int k) {
        int n = nums.size();
        std::cout << "Initialized n = " << n << std::endl;
        if (n == 1) {
            std::cout << "Returning early since n = 1" << std::endl;
            return nums[0];
        }

        stack<int> st;  // Monotonic stack to store indices
        std::cout << "Created stack st" << std::endl;
        vector<int> left(n), right(n);

        // Filling left array
        #pragma unroll
        for (int i = 0; i < n; i++) {
            while (!st.empty() && nums[i] <= nums[st.top()]) {
                st.pop();
                std::cout << "Popped element from stack st" << std::endl;
            }
            left[i] = st.empty() ? -1 : st.top();
            std::cout << "Assigned value to left[" << i << "] = " << left[i] << std::endl;
            st.push(i);
            std::cout << "Pushed index " << i << " to stack st" << std::endl;
        }

        st = {};  // Clear the stack
        std::cout << "Cleared the stack st" << std::endl;

        // Filling right array
        #pragma unroll
        for (int i = n - 1; i >= 0; i--) {
            while (!st.empty() && nums[i] <= nums[st.top()]) {
                st.pop();
                std::cout << "Popped element from stack st" << std::endl;
            }
            right[i] = st.empty() ? n : st.top();
            std::cout << "Assigned value to right[" << i << "] = " << right[i] << std::endl;
            st.push(i);
            std::cout << "Pushed index " << i << " to stack st" << std::endl;
        }

        int ans = 0;
        std::cout << "Initialized ans = " << ans << std::endl;
        #pragma unroll
        for (int i = 0; i < n; i++) {
            if (left[i] < k && right[i] > k) {
                ans = max(ans, nums[i] * (right[i] - left[i] - 1));
                std::cout << "Updated ans to " << ans << " at index " << i << std::endl;
            }
        }

        return ans;
    }
};
