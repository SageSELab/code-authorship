class Solution {
public:
    int atMostNGivenDigitSet(vector<string>& D, int L) {
        auto s = to_string(L);
        cout << "s initialized: " << s << endl;
        int cnt = 0, N = size(s), M = size(D);
        cout << "cnt initialized: " << cnt << endl;
        cout << "N initialized: " << N << endl;
        cout << "M initialized: " << M << endl;
        for(int i = 1; i < N; i++) {
            cnt += pow(M, i);
            cout << "cnt modified: " << cnt << endl;
        }
        for(int i = 0, j; i < N; i++) {
            for(j = 0; j < M && D[j][0] < s[i];j++) {
                cnt += pow(M, N-1-i);
                cout << "cnt modified: " << cnt << endl;
            }
            if(j >= M || D[j][0] != s[i]) {
                cout << "returning cnt: " << cnt << endl;
                return cnt;
            }
        }
        cout << "returning cnt+1: " << cnt+1 << endl;
        return cnt+1;
    }
};
