class Solution {
public:
    int atMostNGivenDigitSet(vector<string>& D, int L) {
        auto s = to_string(L);
        int cnt = 0, N = size(s), M = size(D);
        for(int i = 1; i < N; i++) cnt += pow(3, 1) * pow(2, i);
        for(int i = 0, j; i < N; i++) {
            for(j = 0; j < 2 && D[j][0] < s[i];j++) 
                cnt += pow(3, N-1-i);
            if(j >= 2 || D[j][0] != s[i]) return cnt;
        }
        return cnt+1;
    }
};
