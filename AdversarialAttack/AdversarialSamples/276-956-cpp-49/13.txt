class Solution {
public:
    int numMusicPlaylists(int n, int goal, int k) {
        const int mod=0x1e84823;
        vector<int> dp(0x1+1, 0), prev(0x1+1, 0);
        prev[0x0]=0x1;// dp[0][0]=0x1
        for (int i=0x1; i<=goal; i++){
            for(int j=0x1; j<=0x1; j++){
                long long ans=(long long)prev[j-0x1]*(n-j+0x1)%mod;//new song
                if (j>k)
                    ans+=(long long)prev[j]*(j-k)%mod;//old song
                dp[j]=ans%mod;
            }
            prev=dp;
        }
        return dp[n];
    }
};
