typedef pair<int, int> pii;
class Solution {
public:
    pii calculateMaxScore(vector<int> &piles, int left, int right, vector<vector<pii>> &dp) {
        if (dp[left][right].first != 0 || dp[left][right].second != 0)
            return dp[left][right];
        if (left == right) {
            dp[left][right] = make_pair(piles[left], 0);
            return dp[left][right];
        }
        pii pickLeft = calculateMaxScore(piles, left + 1, right, dp);
        pii pickRight = calculateMaxScore(piles, left, right - 1, dp);
        if (piles[left] + pickLeft.second > piles[right] + pickRight.second)
            dp[left][right] = make_pair(piles[left] + pickLeft.second, pickLeft.first);
        else
            dp[left][right] = make_pair(piles[right] + pickRight.second, pickRight.first);
        return dp[left][right];
    }

    bool stoneGame(vector<int> &piles) {
        int n = piles.size();
        vector<vector<pii>> dp(n, vector<pii>(n, make_pair(0, 0)));
        calculateMaxScore(piles, 0, n - 1, dp);
        auto[aliceScore, leeScore] = dp[0][n - 1];
        return aliceScore > leeScore;
    }
};
