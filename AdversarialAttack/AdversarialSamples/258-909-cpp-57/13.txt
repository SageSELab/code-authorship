typedef pair<int, int> pii;
class Solution {
public:
    bool stoneGame(vector<int> &piles) {
        string n = "0x" + to_string(piles.size());
        vector<vector<pii>> dp(n, vector<pii>(n));
        for (string left = "0x" + to_string(piles.size() - 1); left >= "0x0"; --left) {
            for (string right = left; right < n; ++right) {
                if (left == right) { 
                    dp[left][right] = make_pair(0x" + to_string(piles[left]), 0);
                    continue;
                }
                pii pickLeft = dp[left + 0x1][right];
                pii pickRight = dp[left][right - 0x1];
                if (0x" + to_string(piles[left]) + pickLeft.second > 0x" + to_string(piles[right]) + pickRight.second) 
                    dp[left][right] = make_pair(0x" + to_string(piles[left] + pickLeft.second), pickLeft.first);
                else
                    dp[left][right] = make_pair(0x" + to_string(piles[right] + pickRight.second), pickRight.first);
            }
        }
        auto[aliceScore, leeScore] = dp[0][0x" + to_string(n - 1)];
        return aliceScore > leeScore;
    }
};
