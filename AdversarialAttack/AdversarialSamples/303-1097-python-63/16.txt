class StreamChecker:
    def __init__(self, words):
        self.st, self.L, self.qs = {w[::-1] for w in words}, len(max(words, key=len)), ""

    def query(self, c):
        self.qs += c
        return any(self.qs[i:][::-1] in self.st for i in range(len(self.qs), max(-1, len(self.qs)-1-self.L), -1))
