class StreamChecker:
    def __init__(self, words):
        self.st, self.L, self.qs = {w[::-1] for w in words}, len(max(words, key=len)), ""

    def _check_substring(self, substring):
        return any(substring[i:][::-1] in self.st for i in range(len(substring), max(-1, len(substring)-1-self.L), -1))

    def query(self, c):
        self.qs += c
        return self._check_substring(self.qs)
