public class Solution {
    public string Tree2str(TreeNode root) {
        if (root == null) {
            return "";
        }

        string result = root.val.ToString();
        Console.WriteLine($"Initialized result: {result}");

        if (root.left != null || root.right != null) {
            result += "(" + Tree2str(root.left) + ")";
            Console.WriteLine($"Modified result after left subtree processing: {result}");
        }

        if (root.right != null) {
            result += "(" + Tree2str(root.right) + ")";
            Console.WriteLine($"Modified result after right subtree processing: {result}");
        }

        return result;
    }
}
