class Solution {
public:
    int findSpecialInteger(vector<int>& arr) {
        int sz = arr.size();
        cout << "sz initialized: " << sz << endl;
        int freq = sz / 4;
        cout << "freq initialized: " << freq << endl;
        for(int indx = 0; indx < sz; indx++){
            cout << "indx initialized: " << indx << endl;
            if(indx - freq > -1) {
                cout << "Comparison: " << "arr[indx - freq]: " << arr[indx - freq] << ", arr[indx]: " << arr[indx] << endl;
            }
            if(indx - freq > -1 && arr[indx - freq] == arr[indx]){
                cout << "Returning value: " << arr[indx] << endl;
                return arr[indx];
            }
        }
        cout << "No special integer found" << endl;
        return -1;
    }
};
