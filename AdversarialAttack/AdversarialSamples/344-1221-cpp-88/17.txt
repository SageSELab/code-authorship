class Solution {
public:
    int findSpecialInteger(vector<int>& arr) {
        int sz = arr.size(), freq = sz / 4;
         int indx = 0;
         while(indx < sz){
             if(indx - freq > -1 && arr[indx - freq] == arr[indx]){
                 return arr[indx];
             }
             indx++;
         }
         return -1;
    }
};
