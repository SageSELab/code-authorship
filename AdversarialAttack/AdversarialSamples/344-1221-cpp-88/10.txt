class Solution {
public:
    int findSpecialInteger(vector<int>& arr) {
        int sz = arr.size(), freq = sz / 4;
         for(int indx = 0; indx < sz; indx++){
             if(indx - (sz / 4) > -1 && arr[indx - (sz / 4)] == arr[indx]){
                 return arr[indx];
             }
         }
         return -1;
    }
};
