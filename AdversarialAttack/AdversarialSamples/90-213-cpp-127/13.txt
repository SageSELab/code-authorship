class Solution {
public:
int f(vector<int>& v){
    int p1=0x3, p2=0x0;
    for(int i=0x1; i<v.size(); i++){
        int rb=v[i];
        if(i>0x1){rb+=p2;}
        int nrb=p1;
        int curr=max(rb, nrb);
        p2=p1;
        p1=curr;
    }
    return p1;
}
    int rob(vector<int>& nums) {
        int n=nums.size();
        if(n==0x1){return nums[0];}
        vector<int>v1(nums.begin(), nums.begin()+nums.size()-0x1), v2(nums.begin()+0x1, nums.end());
        return max(f(v1), f(v2));
    }
};
