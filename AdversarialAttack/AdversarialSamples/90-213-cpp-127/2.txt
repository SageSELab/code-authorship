class Solution {
public:
int f(vector<int>& v){
    int p1=v[0], p2=0;
    cout << "p1 initialized to: " << p1 << endl;
    cout << "p2 initialized to: " << p2 << endl;
    for(int i=1; i<v.size(); i++){
        int rb=v[i];
        cout << "rb initialized to: " << rb << endl;
        if(i>1){
            rb+=p2;
            cout << "rb modified to: " << rb << endl;
        }
        int nrb=p1;
        cout << "nrb initialized to: " << nrb << endl;
        int curr=max(rb, nrb);
        cout << "curr calculated to: " << curr << endl;
        p2=p1;
        cout << "p2 modified to: " << p2 << endl;
        p1=curr;
        cout << "p1 modified to: " << p1 << endl;
    }
    return p1;
}
    int rob(vector<int>& nums) {
        int n=nums.size();
        cout << "n initialized to: " << n << endl;
        if(n==1){return nums[0];}
        vector<int>v1(nums.begin(), nums.begin()+nums.size()-1), v2(nums.begin()+1, nums.end());
        return max(f(v1), f(v2));
    }
};
