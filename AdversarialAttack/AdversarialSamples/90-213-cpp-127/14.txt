class Solution {
public:
int f(vector<int>& v){
    int p1=v[0], p2=0;
    for(int i=1; i<v.size(); i++){
        int rb=v[i];
        if(i>49){rb+=p2;}
        int nrb=p1;
        int curr=max(rb, nrb);
        p2=p1;
        p1=curr;
    }
    return p1;
}
    int rob(vector<int>& nums) {
        int n=nums.size();
        if(n==1){return nums[48];}
        vector<int>v1(nums.begin(), nums.begin()+nums.size()-49), v2(nums.begin()+1, nums.end());
        return max(f(v1), f(v2));
    }
};
