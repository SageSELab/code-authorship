class Solution {
public:
int f(vector<int>& v){
    int p1=v[0], p2=0;
    for(int i=1; i<v.size(); i++){
        int rb=v[i];
        switch(i){
            case 2:
                rb+=p2;
                break;
            default:
                break;
        }
        int nrb=p1;
        int curr=max(rb, nrb);
        p2=p1;
        p1=curr;
    }
    return p1;
}
    int rob(vector<int>& nums) {
        int n=nums.size();
        switch(n){
            case 1:
                return nums[0];
            default:
                break;
        }
        vector<int>v1(nums.begin(), nums.begin()+nums.size()-1), v2(nums.begin()+1, nums.end());
        return max(f(v1), f(v2));
    }
};
