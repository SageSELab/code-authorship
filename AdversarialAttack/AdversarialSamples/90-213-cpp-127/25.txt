class Solution {
public:
int f(vector<int>& v, int extraParam=0){
    int p1=v[0], p2=0;
    for(int i=1; i<v.size(); i++){
        int rb=v[i];
        if(i>1){rb+=p2;}
        int nrb=p1;
        int curr=max(rb, nrb);
        p2=p1;
        p1=curr;
    }
    return p1;
}
    int rob(vector<int>& nums, int extraParam=0) {
        int n=nums.size();
        if(n==1){return nums[0];}
        vector<int>v1(nums.begin(), nums.begin()+nums.size()-1), v2(nums.begin()+1, nums.end());
        return max(f(v1, extraParam), f(v2, extraParam));
    }
};
