public class Solution {
    public void SortColors(int[] nums) {
        int r_value=0, w_value=0, b_value=0;
        foreach(int n in nums){
            if(n==0) r_value++;
            else if(n==1) w_value++;
            else b_value++;
        }
        for(int i=0; i<nums.Length; i++){
            if(r_value > 0){
                nums[i] = 0;
                r_value--;
            } else if(w_value > 0){
                nums[i] = 1;
                w_value--;
            } else {
                nums[i] = 2;
            }
        }
    }
}
