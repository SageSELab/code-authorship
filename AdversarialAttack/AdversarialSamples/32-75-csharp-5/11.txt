public class Solution {
    public void SortColors(int[] nums) {
        int r = 0, w = 0, b = 0;
        foreach (int n in nums) {
            if (n == 0) {
                r += 1;
            } else if (n == 1) {
                w += 1;
            } else {
                b += 1;
            }
        }
        for (int i = 0; i < nums.Length; i++) {
            if (r > 0) {
                nums[i] = 0;
                r -= 1;
            } else if (w > 0) {
                nums[i] = 1;
                w -= 1;
            } else {
                nums[i] = 2;
            }
        }
    }
}
