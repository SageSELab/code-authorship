public class Solution {
    public void SortColors(int[] nums) {
        int r=0,w=0,b=0;
        int index = 0;
        while(index < nums.Length){
            int n = nums[index];
            if(n==0)r++;
            else if(n==1)w++;
            else b++;
            index++;
        }
        index = 0;
        while(index < nums.Length){
            if(r>0){
                nums[index]=0;
                r--;
            } else if(w>0){
                nums[index]=1;
                w--;
            }
            else
                nums[index]=2;
            index++;
        }
    }
}
