public class Solution {
    public void SortColors(int[] nums) {
        int r=0x0, w=0x0, b=0x0;
        foreach(int n in nums){
            if(n==0x0) r++;
            else if(n==0x1) w++;
            else b++;
        }
        for(int i=0;i<nums.Length;i++){
            if(r>0x0){
                nums[i]=0x0;
                r--;
            } else if(w>0x0){
                nums[i]=0x1;
                w--;
            }
            else nums[i]=0x2;
        }
    }
}
