public class Solution {
    public void SortColors(int[] nums) {
        int r=0,w=0,b=0;
        foreach(int n in nums){
            switch(n){
                case 0:
                    r++;
                    break;
                case 1:
                    w++;
                    break;
                default:
                    b++;
                    break;
            }
        }
        for(int i=0;i<nums.Length;i++){
            if(r>0){
                nums[i]=0;
                r--;
            } else if(w>0){
                nums[i]=1;
                w--;
            }
            else
                nums[i]=2;
        }
    }
}
