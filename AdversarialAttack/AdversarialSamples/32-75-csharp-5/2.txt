public class Solution {
    public void SortColors(int[] nums) {
        int r=0,w=0,b=0;
        Console.WriteLine("Initialized variables r=0, w=0, b=0");
        foreach(int n in nums){
            Console.WriteLine($"Processing number: {n}");
            if(n==0){
                r++;
                Console.WriteLine("Incremented value of r");
            }
            else if(n==1){
                w++;
                Console.WriteLine("Incremented value of w");
            } else {
                b++;
                Console.WriteLine("Incremented value of b");
            }
        }
        
        for(int i=0;i<nums.Length;i++){
            if(r>0){
                Console.WriteLine("Setting nums[i] to 0");
                nums[i]=0;
                r--;
                Console.WriteLine("Decremented value of r");
            } else if(w>0){
                Console.WriteLine("Setting nums[i] to 1");
                nums[i]=1;
                w--;
                Console.WriteLine("Decremented value of w");
            } else {
                Console.WriteLine("Setting nums[i] to 2");
                nums[i]=2;
            }
        }
    }
}
