class Solution {
public:
    int numTrees(int n) {
        if(n <= 1) {
            int init_ans = 1;
            return init_ans;
        }
        int ans = 0;
        for(int i = 1; i <= n; i++) {
            int left = numTrees(i-1);
            int right = numTrees(n-i);
            int new_ans = ans + left * right;
            ans = new_ans;
        }
        return ans;
    }
};
