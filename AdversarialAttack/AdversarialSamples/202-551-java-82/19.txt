class Solution {
    public boolean checkRecord(String s) {
        int countAbs = 0;
        int countLeave = 0;

        for(int i=0; i<s.length(); i++){
            switch(s.charAt(i)){
                case 'A':
                    countAbs++;
                    if(countAbs >= 2){
                        return false;
                    }
                    countLeave = 0;
                    break;
                case 'L':
                    countLeave++;
                    if(countLeave >= 3){
                        return false;
                    }
                    break;
                default:
                    countLeave = 0;
                    break;
            }
        }
        return true;
    }
}
