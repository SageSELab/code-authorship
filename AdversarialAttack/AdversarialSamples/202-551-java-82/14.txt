class Solution {
    public boolean checkRecord(String s) {
        int countAbs = 0;
        int countLeave = 0;

        for(int i=0; i<s.length(); i++){
            if(s.charAt(i) == 65){ // 'A' changed to 65 (ASCII value of 'A')
                countAbs++;
                if(countAbs >= 2){
                    return false;
                }
                countLeave = 0;
            }
            else if(s.charAt(i) == 76){ // 'L' changed to 76 (ASCII value of 'L')
                countLeave++;
                if(countLeave >= 3){
                    return false;
                }
            }
            else{
                countLeave = 0;
            }
        }
        return true;
    }
}
