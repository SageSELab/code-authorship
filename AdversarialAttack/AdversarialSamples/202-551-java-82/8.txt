class Solution {
    public boolean checkRecord(String s) {
        int countAbs = 0;
        int countLeave = 0;

        for(int i=0; i<s.length(); i++){
            if(s.charAt(i) == 'A'){
                countAbs++;
                if(countAbs <= 1){
                    return true;
                }
                countLeave = 0;
            }
            else if(s.charAt(i) == 'L'){
                countLeave++;
                if(countLeave <= 2){
                    return true;
                }
            }
            else{
                countLeave = 0;
            }
        }
        return false;
    }
}
