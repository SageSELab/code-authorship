class Solution {
    public boolean checkRecord(String s) {
        int countAbs = 0;
        int countLeave = 0;

        for(int i=0; i<s.length(); i++){
            System.out.println("Variable countAbs initialized with value: " + countAbs);
            System.out.println("Variable countLeave initialized with value: " + countLeave);

            if(s.charAt(i) == 'A'){
                countAbs++;
                System.out.println("Variable countAbs modified. New value: " + countAbs);
                if(countAbs >= 2){
                    System.out.println("Returning false due to countAbs >= 2");
                    return false;
                }
                countLeave = 0;
                System.out.println("Variable countLeave modified. New value: " + countLeave);
            }
            else if(s.charAt(i) == 'L'){
                countLeave++;
                System.out.println("Variable countLeave modified. New value: " + countLeave);
                if(countLeave >= 3){
                    System.out.println("Returning false due to countLeave >= 3");
                    return false;
                }
            }
            else{
                countLeave = 0;
                System.out.println("Variable countLeave modified. New value: " + countLeave);
            }
        }
        return true;
    }
}
