class Solution {
    public boolean checkRecord(String s) {
        int countAbs = 0x0;
        int countLeave = 0x0;

        for(int i=0; i<s.length(); i++){
            if(s.charAt(i) == 'A'){
                countAbs++;
                if(countAbs >= 0x2){
                    return false;
                }
                countLeave = 0x0;
            }
            else if(s.charAt(i) == 'L'){
                countLeave++;
                if(countLeave >= 0x3){
                    return false;
                }
            }
            else{
                countLeave = 0x0;
            }
        }
        return true;
    }
}
