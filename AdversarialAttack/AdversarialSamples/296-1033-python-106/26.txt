class Solution:
    def brokenCalc(self, X, Y):
        if X > Y: return X - Y
        if X == Y: return 0
        
        def helper(X, Y):
            if Y % 2 == 0:
                return helper(X, Y // 2) + 1
            else:
                return helper(X, Y + 1) + 1
        
        return helper(X, Y)
