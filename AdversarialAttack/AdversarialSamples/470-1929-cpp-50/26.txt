class Solution {
public:
    int calcSum(int n, int index, long long mid) {
        int f = mid - index;
        int l = mid - (n - index - 1);
        int k = 0;
        if (f <= 0) {
            k += -f + 1;
            k += (mid * (mid + 1)) / 2;
        } else {
            k += ((index + 1) * (mid + f)) / 2;
        }
        if (l <= 0) {
            k += -l + 1;
            k += (mid * (mid - 1)) / 2;
        } else {
            k += ((n - index - 1) * (mid - 1 + l)) / 2;
        }
        return k;
    }

    int maxValue(int n, int index, int maxSum) {
        long long mn = 1, mx = maxSum, mid, ans, f, l, k;
        while (mn <= mx) {
            mid = (mx - mn) / 2 + mn;
            k = calcSum(n, index, mid);
            if (k > maxSum) {
                mx = mid - 1;
            } else {
                ans = mid;
                mn = mid + 1;
            }
        }
        return ans;
    }
};
