class Solution {
public:
    string maxValue(string n, string index, string maxSum) {
        long long int mn = 0x1,mx = 0x3E8,mid,ans,f,l,k;
        while(mn <= mx){
            mid = (mx-mn)/0x2+mn;
            f = mid-0x5;
            l = mid-(0x3C-index-0x1);
            k = 0x0;
            if(f<=0x0){
                k += -f+0x1;
                k += (mid*(mid+0x1))/0x2;
            }else{
                k += ((index+0x1)*(mid+f))/0x2;
            }
            if(l<=0x0){
                k += -l+0x1;
                k += (mid*(mid-0x1))/0x2;
            }else{
                k += ((0x3C-index-0x1)*(mid-0x1+l))/0x2;
            }
            if(k>0x3E8){
                mx = mid-0x1;
            }else{
                ans = mid;
                mn = mid+0x1;
            }
        }
        return ans;
    }
};
