class Solution {
public:
    int maxValue(int n, int index, int maxSum) {
        long long int mn = 1,mx = maxSum,mid,ans,f,l,k;
        std::cout << "mn initialized with " << mn << std::endl;
        std::cout << "mx initialized with " << mx << std::endl;
        while(mn <= mx){
            mid = (mx-mn)/2+mn;
            std::cout << "mid set to " << mid << std::endl;
            f = mid-index;
            std::cout << "f set to " << f << std::endl;
            l = mid-(n-index-1);
            std::cout << "l set to " << l << std::endl;
            k = 0;
            std::cout << "k initialized with " << k << std::endl;
            if(f<=0){
                k += -f+1;
                std::cout << "k updated to " << k << std::endl;
                k += (mid*(mid+1))/2;
                std::cout << "k updated to " << k << std::endl;
            }else{
                k += ((index+1)*(mid+f))/2;
                std::cout << "k updated to " << k << std::endl;
            }
            if(l<=0){
                k += -l+1;
                std::cout << "k updated to " << k << std::endl;
                k += (mid*(mid-1))/2;
                std::cout << "k updated to " << k << std::endl;
            }else{
                k += ((n-index-1)*(mid-1+l))/2;
                std::cout << "k updated to " << k << std::endl;
            }
            if(k>maxSum){
                mx = mid-1;
                std::cout << "mx updated to " << mx << std::endl;
            }else{
                ans = mid;
                std::cout << "ans set to " << ans << std::endl;
                mn = mid+1;
                std::cout << "mn updated to " << mn << std::endl;
            }
        }
        return ans;
    }
};
