class Solution {
public:
    int maxValue(int n, int index, int maxSum) {
        long long int mn = 1, mx = maxSum, mid, ans, f, l, k;
        while(mn <= mx){
            mid = (mx - mn) / 2 + mn;
            f = mid - index;
            l = mid - (n - index - 1);
            k = 0;
            
            switch(f<=0){
                case true:
                    k += -f + 1;
                    k += (mid * (mid + 1)) / 2;
                    break;
                case false:
                    k += ((index + 1) * (mid + f)) / 2;
            }
            
            switch(l<=0){
                case true:
                    k += -l + 1;
                    k += (mid * (mid - 1)) / 2;
                    break;
                case false:
                    k += ((n - index - 1) * (mid - 1 + l)) / 2;
            }
            
            if(k > maxSum){
                mx = mid - 1;
            }else{
                ans = mid;
                mn = mid + 1;
            }
        }
        return ans;
    }
};
