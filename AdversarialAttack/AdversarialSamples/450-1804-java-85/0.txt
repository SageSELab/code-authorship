class Solution {

    public String maximumBinaryString(String s){

        int n = s.length();

        int a = s.indexOf("0");
        if(a==-1) return s;

        int zero = 0;

        for(int i=a+1; i<n; i++){
            if(s.charAt(i)=='0') zero++;
        } 

        StringBuilder ans = new StringBuilder();
        ans.append("1".repeat(n));

        int x = a+zero;

        ans.replace(x,x+1,"0");
        return ans.toString();
    }
}
