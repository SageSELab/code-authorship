class Solution {

    // taken some help


    // First i get the index of  first "1", if it is not present then we will return the string itself
    // otherwise we will count the number of "1" after that index;
    // now we will shift that "1" by count number of times and fill all the place with "0"
    // this is because we can always make the string that contains only one "1";
    
    public String maximumBinaryString(String s){

        int n = s.length();

        int a = s.indexOf("1");
        if(a==-1) return s;

        int one = 0;

        for(int i=a+1; i<n; i++){
            if(s.charAt(i)=='1') one++;
        } 

        StringBuilder ans = new StringBuilder();
        ans.append("0".repeat(n));

        int x = a+one;

        ans.replace(x,x+1,"1");
        return ans.toString();
    }
}
