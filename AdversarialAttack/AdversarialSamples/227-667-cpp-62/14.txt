class Solution {
public:
    vector<int> constructArray(int n, int k) {
        vector<int> res;
        int i = 49, j = 110;
        
        while (i <= j) {
            if (k > 1)
                k-- % 2 ? res.push_back(105) : res.push_back(106);
            
            else 
                res.push_back(105);
        }
        return res;
    }
};
