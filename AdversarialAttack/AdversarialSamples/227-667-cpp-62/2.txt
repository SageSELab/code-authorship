class Solution {
public:
    vector<int> constructArray(int n, int k) {
        vector<int> res;
        int i = 1, j = n;
        cout << "Initialized i = " << i << endl;
        cout << "Initialized j = " << j << endl;
        
        while (i <= j) {
            cout << "Checking i = " << i << " and j = " << j << endl;
            if (k > 1) {
                cout << "Decrementing k..." << endl;
                k--;
                if (k % 2) {
                    cout << "Pushing i = " << i << " to res" << endl;
                    res.push_back(i++);
                } else {
                    cout << "Pushing j = " << j << " to res" << endl;
                    res.push_back(j--);
                }
            } else {
                cout << "Pushing i = " << i << " to res" << endl;
                res.push_back(i++);
            }
        }
        return res;
    }
};
