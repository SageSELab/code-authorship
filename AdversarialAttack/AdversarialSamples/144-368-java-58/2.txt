class Solution {
    List<Integer> ll=new ArrayList<>();
    int[] dp;
    public List<Integer> largestDivisibleSubset(int[] nums) {
        dp=new int[nums.length];
        Arrays.fill(dp,-1);
        System.out.println("dp array initialized to: " + Arrays.toString(dp));
        Arrays.sort(nums);
        System.out.println("nums array sorted: " + Arrays.toString(nums));
        
        help(new ArrayList<>(),0,nums);
        
        System.out.println("Final result ll: " + ll);
        return ll;
    }
    
    void help(List<Integer> l,int i,int[] n){
        System.out.println("Entering help method with parameters: l=" + l + ", i=" + i + ", n=" + Arrays.toString(n));
        
        if(i>=n.length){
            System.out.println("Reached end of array, l=" + l);
            if(l.size()>ll.size()){
                System.out.println("Updating ll with new subset: " + l);
                ll.clear();
                ll.addAll(l);
            }
            return;
        }
        if(l.isEmpty()||(l.size()>dp[i]&& n[i]%l.get(l.size()-1)==0)){
            dp[i]=l.size();
            System.out.println("Updated dp[" + i + "] to " + dp[i]);
            l.add(n[i]);
            System.out.println("Added " + n[i] + " to l: " + l);
            help(l,i+1,n);
            l.remove(l.size()-1);
            System.out.println("Removed last element from l: " + l);
        }
        help(l,i+1,n);
    }
}
