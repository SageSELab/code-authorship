class Solution {
    List<Integer> ll=new ArrayList<>();
    int[] dp;
    public List<Integer> largestDivisibleSubset(int[] nums) {
        dp=new int[nums.length];
        Arrays.fill(dp,-1);
        Arrays.sort(nums);
        help(new ArrayList<>(),0,nums);
        return ll;
    }
    
    void help(List<Integer> l, int i, int[] n){
        if(i>=n.length){
            updateList(l);
            return;
        }
        if(isValidToAdd(l, i, n)){
            dp[i]=l.size();
            l.add(n[i]);
            help(l, i+1, n);
            l.remove(l.size()-1);
        }
        help(l, i+1, n);
    }
    
    void updateList(List<Integer> l){
        if(l.size() > ll.size()){
            ll.clear();
            ll.addAll(l);
        }
    }
    
    boolean isValidToAdd(List<Integer> l, int i, int[] n){
        return l.isEmpty() || (l.size() > dp[i] && n[i] % l.get(l.size()-1) == 0);
    }
}
