class Solution {
public:
    vector<int> count_bits(int n) {
        vector<int> res;
        for(int num = 0; num <= n; num++){
            res.push_back(bit_cnt(num));
        }
        return res;
    }
private:
    int bit_cnt(int num){
         int cnt = 0;
        while(num > 0){
            cnt++;
            num &= (num-1);
        }
        return cnt;
    }    
};
