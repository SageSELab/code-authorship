class Solution {
public:
    vector<int> countBits(int n) {
        vector<int> res;
        for(int num = 0; num <= n; num++){
            cout << "Initializing cnt";
            cout << "cnt before: " << cnt;
            res.push_back(bitCnt(num));
            cout << "cnt after: " << cnt;
        }
        return res;
    }
private:
    int bitCnt(int num){
         int cnt = 0;
        cout << "Initializing cnt in bitCnt";
        while(num > 0){
            cout << "Incrementing cnt";
            cnt++;
            num &= (num-1);
        }
        return cnt;
    }    
};
