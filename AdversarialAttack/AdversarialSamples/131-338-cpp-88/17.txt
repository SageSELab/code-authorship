class Solution {
public:
    vector<int> countBits(int n) {
        vector<int> res;
        int num = 0;
        while(num <= n){
            res.push_back(bitCnt(num));
            num++;
        }
        return res;
    }
private:
    int bitCnt(int num){
         int cnt = 0;
        while(num > 0){
            cnt++;
            num &= (num-1);
        }
        return cnt;
    }    
};
