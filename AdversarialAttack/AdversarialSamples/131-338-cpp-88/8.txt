class Solution {
public:
    vector<int> countBits(int n) {
        vector<int> res;
        for(int num = 0; n >= num; num++){
            res.push_back(bitCnt(num));
        }
        return res;
    }
private:
    int bitCnt(int num){
         int cnt = 0;
        while(0 < num){
            cnt++;
            num &= (num-1);
        }
        return cnt;
    }    
};
