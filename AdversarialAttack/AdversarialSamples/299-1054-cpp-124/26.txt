class Solution {
public:
    int bitwiseComplement(int num) {
        //base case
        if(num==0) return 1;
        
        vector<int> binaryRepresentation = decimalToBinary(num);
        vector<int> complement = getComplement(binaryRepresentation);
        int decimalResult = binaryToDecimal(complement);
        return decimalResult;
    }
    
    vector<int> decimalToBinary(int num) {
        vector<int> binaryNum;
        
        while(num != 0) {
            binaryNum.push_back(num % 2);
            num /= 2;
        }
        
        return binaryNum;
    }
    
    vector<int> getComplement(vector<int> binaryNum) {
        for(int i = 0; i < binaryNum.size(); i++) {
            if(binaryNum[i] == 1) {
                binaryNum[i] = 0;
            } else if(binaryNum[i] == 0) {
                binaryNum[i] = 1;
            }
        }
        
        return binaryNum;
    }
    
    int binaryToDecimal(vector<int> binaryNum) {
        int decimalNum = 0;
        
        for(int i = binaryNum.size() - 1; i > -1; i--) {
            decimalNum = decimalNum * 2 + binaryNum[i];
        }
        
        return decimalNum;
    }
};
