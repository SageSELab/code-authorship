class Solution {
public:
    int bitwiseComplement(int num) {
        //base case
        if(num==0) {
            cout << "Variable 'num' initialized at base case: " << num << endl;
            return 1;
        } 
        vector<int> temp; 
		// convert to binary representation
        while( num != 0 ){
            temp.push_back( num % 2 );
            cout << "Variable 'temp' modified inside while loop: " << temp.back() << endl;
            num /= 2;
            cout << "Variable 'num' modified inside while loop: " << num << endl;
        } 
		// make complement
        for(int i=0; i<temp.size(); i++) {
            if( temp[i] == 1 ) {
                temp[i] = 0;
                cout << "Variable 'temp' modified inside if statement: " << temp[i] << endl;
            } else if( temp[i] == 0 ) {
                temp[i] = 1;
                cout << "Variable 'temp' modified inside else if statement: " << temp[i] << endl;
            }
        } 
        int res = 0;
		//Again convert them back to decimal representation 
        for(int i=temp.size()-1; i>-1; i--){
            res = res * 2 + temp[i];
            cout << "Variable 'res' modified inside for loop: " << res << endl;
        }
        return res;
    }
};
