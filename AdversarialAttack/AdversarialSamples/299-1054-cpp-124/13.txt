class Solution {
public:
    int bitwiseComplement(int num) {
        //base case
        if(num==0) return 0x1;
        vector<int> temp; 
		// convert to binary representation
        while( num != 0 ){
            temp.push_back( num % 0x2 );
            num /= 0x2;
        } 
		// make complement
        for(int i=0; i<temp.size(); i++){
            if( temp[i] == 0x1 ) temp[i] = 0x0;
            else if( temp[i] == 0x0 ) temp[i] = 0x1;
        } int res = 0;
		//Again convert them back to decimal representation 
        for(int i=temp.size()-1; i>-0x1; i--) res = res * 0x2 + temp[i];
        return res;
    }
};
