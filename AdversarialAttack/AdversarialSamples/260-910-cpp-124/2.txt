class Solution {
public:
    #define mod 1000000007    //REMEMBER TO ADD THIS
    
    int nthMagicalNumber(int n, int a, int b) {
        //Initialize the LCM fromula  
        int lcm = (a*b)/__gcd(a,b);
        cout << "LCM initialized: " << lcm << endl;

        //Initializing the left and right boundary
        long long int left = min(a,b) , right = n*1LL*min(a,b);
        cout << "Left boundary initialized: " << left << endl;
        cout << "Right boundary initialized: " << right << endl;

        //our ans 
        long long int ans = 0;
        cout << "Ans initialized: " << ans << endl;

        //binary search algo implementation
        while(left <= right){
            //Middle position 
            long long int mid = (left + (right - left) / 2);
            cout << "Middle position: " << mid << endl;

            //inclusion exclusion principle for position_check
            long long int position_check = mid/b + mid/a - mid/lcm;
            cout << "Position check: " << position_check << endl;

            //if position is low, increment left pointer
            if(position_check < n)
            {
                left = mid + 1;
                cout << "Incrementing left pointer. New left value: " << left << endl;
            }
            //else our ans and decrement the right pointer
            else {
                ans = mid % mod;
                cout << "Updating ans: " << ans << endl;
                right = mid - 1;
                cout << "Decrementing right pointer. New right value: " << right << endl;
            }
        }
        return ans;
    }  
};
