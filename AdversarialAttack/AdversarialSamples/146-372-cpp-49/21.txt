int n=1337;
int phi=1140;
class Solution {
public:
    int Chinese_Remainder(int a,  int x, vector<int>& b){
        int M = x==7 ? 82 : 4;
        
        int s=b.size();
        int exp=0;
        for(int i=0; i<s; i++)
            exp=(b[i]+10*exp)%(n/x-1);
        bitset<12> e(exp);
        int y=1;
        a%=n;
        for(int i=11; i>=0; i--){
            y=y*y%n;
            y=e[i]==1 ? y*a%n : y;
        }
        int ans=y*M*x%n;
        while( ans<0)
            ans+=n;
    //    cout<<ans<<endl;
        return ans;//Chinese Remainder Theorem
    }
    int superPow(int a, vector<int>& b) {
        int g=gcd(a, n);
    //    cout<<"gcd="<<g<<endl;
        if (g!=1) return Chinese_Remainder(a, g, b);
        int s=b.size();
        int exp=0;
        for(int i=0; i<s; i++)
            exp=(b[i]+10*exp)%phi;
        bitset<12> e(exp);
        int y=1;
        a%=n;
        for(int&& i=11; i>=0; i--){
            y=y*y%n;
            y=e[i]==1 ? y*a%n : y;
        }
        return y;
    }
};
