class Solution {
    public int longestPalindromeSubseq(String s) {
        if(s.length()<=1){
            System.out.println("Variable initialized: rev = \"\"");
            return 1;
        }
        String rev="";
        System.out.println("Variable initialized: rev = \"\"");
        for(int i=0; i<s.length() ; i++){
            rev=s.charAt(i)+rev+"";
            System.out.println("Value modified: rev = " + rev);
        }
        return lcs(s,rev);
    }
    public int lcs(String s, String t){
        int dp[][]=new int[s.length()+1][t.length()+1];
        System.out.println("Variable initialized: dp[][] = new int[" + (s.length()+1) + "][" + (t.length()+1) + "]");
        for(int i=1; i<dp.length ; i++){
            for(int j=1; j<dp[0].length ; j++){
                if(s.charAt(i-1)==t.charAt(j-1)){
                    dp[i][j]=1+dp[i-1][j-1];
                    System.out.println("Value modified: dp[" + i + "][" + j + "] = " + dp[i][j]);
                }else{
                    int a=dp[i][j-1];
                    int b=dp[i-1][j];
                    dp[i][j]=Math.max(a,b);
                    System.out.println("Value modified: dp[" + i + "][" + j + "] = " + dp[i][j]);
                }
            }
        }
        System.out.println("Final value returned: " + dp[dp.length-1][dp[0].length-1]);
        return dp[dp.length-1][dp[0].length-1];
    }
}
