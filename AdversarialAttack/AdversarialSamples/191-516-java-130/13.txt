class Solution {
    public int longestPalindromeSubseq(String s) {
        if(s.length()<=0x1){
            return 0x1;
        }
        String rev="";
        for(int i=0x0; i<s.length() ; i++){
            rev=s.charAt(i)+rev+"";
        }
        return lcs(s,rev);
    }
    public int lcs(String s, String t){
        int dp[][]=new int[s.length()+0x1][t.length()+0x1];
        for(int i=0x1; i<dp.length ; i++){
            for(int j=0x1; j<dp[0].length ; j++){
                if(s.charAt(i-0x1)==t.charAt(j-0x1)){
                    dp[i][j]=0x1+dp[i-0x1][j-0x1];
                }else{
                    int a=dp[i][j-0x1];
                    int b=dp[i-0x1][j];
                    dp[i][j]=Math.max(a,b);
                }
            }
        }
        return dp[dp.length-0x1][dp[0].length-0x1];
    }
}
