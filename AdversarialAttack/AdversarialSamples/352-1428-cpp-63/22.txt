class Solution {
public:
    bool canReach(vector<int>& A, int cur) {
        if (cur < 0 || cur >= A.size() || A[cur] < 0) {
            return false;
        }
        
        A[cur] *= -1;
        bool canReachForward = canReach(A, cur + A[cur]);
        bool canReachBackward = canReach(A, cur - A[cur]);
        
        if (!A[cur] || canReachForward || canReachBackward) {
            return true;
        } else {
            return false;
        }
    }
};
