public class Solution {
    public List<List<Integer>> permuteUnique(int[] nums) {
        List<List<Integer>> res = new ArrayList<List<Integer>>();
        System.out.println("res initialized: " + res);
        boolean[] checked = new boolean[nums.length];
        System.out.println("checked initialized: " + Arrays.toString(checked));
        List<Integer> list = new ArrayList<Integer>();
        System.out.println("list initialized: " + list);
        Arrays.sort(nums);
        System.out.println("nums sorted: " + Arrays.toString(nums));
        solve(nums, checked, list, res);
        return res;
    }

    public void solve(int[] nums, boolean[] checked, List<Integer> ds, List<List<Integer>> res){
        System.out.println("Entering solve method");
        if(ds.size()==nums.length){
            res.add(new ArrayList<Integer>(ds));
            System.out.println("Added new list to res: " + res);
            return;
        }
        for(int i=0;i<nums.length;i++){
            System.out.println("Iterating i: " + i);
            if(checked[i]) continue;
            if(i>0 && nums[i-1]==nums[i] && !checked[i-1] ) continue;
            checked[i]=true;
            System.out.println("checked[" + i + "] set to true: " + Arrays.toString(checked));
            ds.add(nums[i]);
            System.out.println("Added " + nums[i] + " to ds: " + ds);
            solve(nums,checked,ds,res);
            checked[i]=false;
            System.out.println("checked[" + i + "] set to false: " + Arrays.toString(checked));
            ds.remove(ds.size()-1);
            System.out.println("Removed last element from ds: " + ds);
        }
    }
}
