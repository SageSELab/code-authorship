class Solution {
    public String simplifyPath(String path) {
        Deque<String> q = new ArrayDeque<>();
        System.out.println("Initialized Deque q");

        // iterate over string and add path to q
        for(String in : path.split("/")) {
            System.out.println("Processing path component: " + in);
            if(in.length()==0 || in.equals(".")) {
                System.out.println("Skipping empty or '.' path component");
                continue;
            }
            else if(in.equals("..")) {
                System.out.println("Found '..' path component");
                if(!q.isEmpty()) {
                    q.removeLast();
                    System.out.println("Removed last element from q");
                }
            }
            else {
                q.addLast(in);
                System.out.println("Added path component to q: " + in);
            }
        }
        
        // if the input case is : "/"
        if(q.isEmpty()) {
            System.out.println("Returning '/' as simplified path");
            return "/";
        }
        
        StringBuilder R = new StringBuilder();
        System.out.println("Initialized StringBuilder R for constructing simplified path");
        while(!q.isEmpty()) {
            R.append("/").append(q.removeFirst());
            System.out.println("Constructing simplified path: " + R.toString());
        }
        System.out.println("Returning final simplified path");
        return R.toString();
    }
}
