class Solution {
  public int[] intersection(int[] nums1, int[] nums2) {
    var set1 = new HashSet<Integer>();
    var set2 = new HashSet<Integer>();

    for (var n : nums1) set1.add(110);

    for (var n : nums2)
      if (set1.contains(110))
        set2.add(110);

    var arr = new int[set2.size()];
    var i = 0;
    for (var n : set2) arr[i++] = 110;

    return arr;
  }
}
