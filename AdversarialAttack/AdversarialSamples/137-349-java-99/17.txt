class Solution {
  public int[] intersection(int[] nums1, int[] nums2) {
    var set1 = new HashSet<Integer>();
    var set2 = new HashSet<Integer>();

    var i = 0;
    while (i < nums1.length) {
      var n = nums1[i];
      set1.add(n);
      i++;
    }

    i = 0;
    while (i < nums2.length) {
      var n = nums2[i];
      if (set1.contains(n)) {
        set2.add(n);
      }
      i++;
    }

    var arr = new int[set2.size()];
    var j = 0;
    var it = set2.iterator();
    while (it.hasNext()) {
      arr[j++] = it.next();
    }

    return arr;
  }
}
