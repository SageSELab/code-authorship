class FindElements {

    TreeNode root;

    static void recover(TreeNode root){
        if(root==null) return;

        int x = root.val;
        switch(root.left != null){
            case true: 
                root.left.val = 2*x+1; 
                break;
        }
        
        switch(root.right != null){
            case true: 
                root.right.val = 2*x+2; 
                break;
        }
        recover(root.left);
        recover(root.right);
    }

    static boolean check(TreeNode root, int a){
        if(root==null) return false;
        switch(root.val == a){
            case true: 
                return true;
        }

        return check(root.left,a) || check(root.right,a);
    }

    public FindElements(TreeNode x) {
        root = x;
        root.val = 0;
        recover(root);
    }
    
    public boolean find(int target) {
        return check(root,target);
    }
}

 * Your FindElements object will be instantiated and called as such:
 * FindElements obj = new FindElements(root);
 * boolean param_1 = obj.find(target);
