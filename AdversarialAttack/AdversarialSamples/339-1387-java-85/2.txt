class FindElements {

    TreeNode root;

    static void recover(TreeNode root){
        if(root==null) return;

        int x = root.val;
        System.out.println("x initialized: " + x);
        if(root.left!=null) {
            root.left.val = 2*x+1;
            System.out.println("root.left.val modified: " + root.left.val);
        }
        if(root.right!=null) {
            root.right.val = 2*x+2;
            System.out.println("root.right.val modified: " + root.right.val);
        }
        recover(root.left);
        recover(root.right);
    }

    static boolean check(TreeNode root, int a){
        if(root==null) return false;
        System.out.println("root.val initialized: " + root.val);
        if(root.val==a) return true;

        return check(root.left,a) || check(root.right,a);
    }

    public FindElements(TreeNode x) {
        root = x;
        root.val = 0;
        System.out.println("root.val set to 0: " + root.val);
        recover(root);
    }

    public boolean find(int target) {
        return check(root,target);
    }
}
