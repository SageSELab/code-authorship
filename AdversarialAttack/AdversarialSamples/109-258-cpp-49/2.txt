class Solution {
public:
    int addDigits(int num) {
        if (num==0) {
            std::cout << "Initializing ans to 0" << std::endl;
            return 0;
        }
        std::cout << "Initializing ans to num%9" << std::endl;
        int ans=num%9;
        std::cout << "Checking if ans is equal to 0 or not" << std::endl;
        return (ans==0)?9:ans;
    }
};
