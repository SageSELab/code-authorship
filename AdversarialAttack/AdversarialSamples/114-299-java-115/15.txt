class Solution {
    public String getHint(char[] secret, char[] guess) {
        StringBuilder first = new StringBuilder(String.valueOf(secret));
        StringBuilder second = new StringBuilder(String.valueOf(guess));
        HashMap<Character, Integer> dict = new HashMap<>();
        int numA = 0;
        int numB = 0;
        for (int i = secret.length - 1; i > -1; i--)
        {
            if (first.charAt(i) == second.charAt(i)) {
                numA++;
                first.deleteCharAt(i);
                second.deleteCharAt(i);
            }
            else {
                if (!dict.containsKey(secret[i]))
                    dict.put(secret[i], 0);
                dict.put(secret[i], dict.get(secret[i]) + 1);
            }
        }
        for (int i = 0; i < second.length(); i++)
        {
            if(dict.containsKey(second[i]) && dict.get(second[i]) > 0)
            {
                numB++;
                dict.put(second[i], dict.get(second[i]) - 1);
            }
        }
        return numA + "A" + numB + "B";
    }
}
