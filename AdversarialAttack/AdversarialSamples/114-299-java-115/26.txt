class Solution {
    public String getHint(String secret, String guess) {
        StringBuilder first = new StringBuilder(secret);
        StringBuilder second = new StringBuilder(guess);
        HashMap<Character, Integer> dict = new HashMap<>();
        int numA = calculateNumAandRemoveMatchingChars(first, second);
        int numB = calculateNumB(first, second, dict);
        return numA + "A" + numB + "B";
    }

    private int calculateNumAandRemoveMatchingChars(StringBuilder first, StringBuilder second) {
        int numA = 0;
        for (int i = first.length() - 1; i > -1; i--) {
            if (first.charAt(i) == second.charAt(i)) {
                numA++;
                first.deleteCharAt(i);
                second.deleteCharAt(i);
            }
        }
        return numA;
    }

    private int calculateNumB(StringBuilder first, StringBuilder second, HashMap<Character, Integer> dict) {
        int numB = 0;
        for (int i = 0; i < second.length(); i++) {
            if (dict.containsKey(second.charAt(i)) && dict.get(second.charAt(i)) > 0) {
                numB++;
                dict.put(second.charAt(i), dict.get(second.charAt(i)) - 1);
            }
        }
        return numB;
    }
}
