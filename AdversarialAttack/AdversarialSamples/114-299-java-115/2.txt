class Solution {
    public String getHint(String secret, String guess) {
        System.out.println("Initializing first StringBuilder: " + secret);
        StringBuilder first = new StringBuilder(secret);
        System.out.println("Initializing second StringBuilder: " + guess);
        StringBuilder second = new StringBuilder(guess);
        System.out.println("Initializing HashMap dict");
        HashMap<Character, Integer> dict = new HashMap<>();
        int numA = 0;
        int numB = 0;
        for (int i = secret.length() - 1; i > -1; i--) {
            System.out.println("Checking characters at index: " + i);
            if (first.charAt(i) == second.charAt(i)) {
                System.out.println("Incrementing numA");
                numA++;
                System.out.println("Deleting char from first: " + first.charAt(i));
                first.deleteCharAt(i);
                System.out.println("Deleting char from second: " + second.charAt(i));
                second.deleteCharAt(i);
            } else {
                System.out.println("Checking and updating dict for character: " + secret.charAt(i));
                if (!dict.containsKey(secret.charAt(i)))
                    dict.put(secret.charAt(i), 0);
                dict.put(secret.charAt(i), dict.get(secret.charAt(i)) + 1);
            }
        }
        for (int i = 0; i < second.length(); i++) {
            System.out.println("Checking for character in dict: " + second.charAt(i));
            if (dict.containsKey(second.charAt(i)) && dict.get(second.charAt(i)) > 0) {
                System.out.println("Incrementing numB");
                numB++;
                System.out.println("Decreasing count for character in dict: " + second.charAt(i));
                dict.put(second.charAt(i), dict.get(second.charAt(i)) - 1);
            }
        }
        return numA + "A" + numB + "B";
    }
}
