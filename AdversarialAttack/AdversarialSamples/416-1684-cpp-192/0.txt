class Solution {
public:
    vector<int> parent,Rank,count,groupCounts;
    int Find(int x)
    {
        return parent[x]==x?x:parent[x]=Find(parent[x]);
    }
    void Union(int x,int y)
    {
        int xset=Find(x),yset=Find(y);
        if(xset!=yset)
        {
            groupCounts[count[xset]]--,groupCounts[count[yset]]--;
            count[xset]=count[yset]=count[xset]+count[yset];
            groupCounts[count[xset]]++;
            Rank[xset]<Rank[yset]?parent[xset]=yset:parent[yset]=xset;
            if(Rank[xset]==Rank[yset])
                Rank[xset]++;
        }
    }
    int findLatestStep(vector<int>& arr, int m)
    {
        parent.resize(arr.size()+1),Rank.resize(arr.size()+1,0);
        for(int i=1;i<=arr.size();i++)
            parent[i]=i;
        vector<bool> visited(arr.size()+1,false);
        count.resize(arr.size()+1,1);
        groupCounts.resize(arr.size()+1,0);
        int last=-1,step=1;
        for(int &i:arr)
        {
            groupCounts[1]++;
            if(i+1<=arr.size()&&visited[i+1])
                Union(i,i+1);
            if(i-1>0&&visited[i-1])
                Union(i,i-1);
            visited[i]=true;
            if(groupCounts[m])
                last=step;
            step++;
        }
        return last;
    }
};
