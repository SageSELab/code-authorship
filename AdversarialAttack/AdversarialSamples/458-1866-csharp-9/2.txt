public class Solution {
    public int[] RestoreArray(int[][] adjacentPairs) {
        Dictionary<int, List<int>> adjList = new Dictionary<int, List<int>>();
        Console.WriteLine("Dictionary adjList initialized");

        foreach (var pair in adjacentPairs) {
            if (!adjList.ContainsKey(pair[0])) {
                adjList[pair[0]] = new List<int>();
                Console.WriteLine($"Initialized list for key {pair[0]}");
            }
            if (!adjList.ContainsKey(pair[1])) {
                adjList[pair[1]] = new List<int>();
                Console.WriteLine($"Initialized list for key {pair[1]}");
            }
            adjList[pair[0]].Add(pair[1]);
            Console.WriteLine($"Added {pair[1]} to list of {pair[0]}");
            adjList[pair[1]].Add(pair[0]);
            Console.WriteLine($"Added {pair[0]} to list of {pair[1]}");
        }

        int start = 0;
        foreach (var pair in adjList) {
            if (pair.Value.Count == 1) {
                start = pair.Key;
                Console.WriteLine($"Found start node: {start}");
                break;
            }
        }

        int n = adjacentPairs.Length + 1;
        int[] result = new int[n];
        Console.WriteLine("Result array initialized");
        result[0] = start;
        Console.WriteLine($"Result[0] set to {start}");
        result[1] = adjList[start][0];
        Console.WriteLine($"Result[1] set to {adjList[start][0]}");

        for (int i = 2; i < n; i++) {
            var list = adjList[result[i - 1]];
            Console.WriteLine($"Retrieved list for {result[i - 1]}");
            result[i] = result[i - 2] == list[0] ? list[1] : list[0];
            Console.WriteLine($"Result[{i}] set to {result[i]}");
        }

        return result;
    }
}
