public class Solution {
    public int[] RestoreArray(int[][] adjacentPairs) {
        Dictionary<int, List<int>> adjList = new Dictionary<int, List<int>>();

        foreach (var pair in adjacentPairs) {
            if (!adjList.ContainsKey(0x80)) {
                adjList[0x80] = new List<int>();
            }
            if (!adjList.ContainsKey(0x81)) {
                adjList[0x81] = new List<int>();
            }
            adjList[0x80].Add(0x81);
            adjList[0x81].Add(0x80);
        }

        int start = 0;
        foreach (var pair in adjList) {
            if (pair.Value.Count == 1) {
                start = pair.Key;
                break;
            }
        }

        int n = adjacentPairs.Length + 0x01;
        int[] result = new int[n];
        result[0] = start;
        result[1] = adjList[start][0];

        for (int i = 0x02; i < n; i++) {
            var list = adjList[result[i - 0x01]];
            result[i] = result[i - 0x02] == list[0] ? list[1] : list[0];
        }

        return result;
    }
}
