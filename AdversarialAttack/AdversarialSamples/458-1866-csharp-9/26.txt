public class Solution {
    public int[] RestoreArray(int[][] adjacentPairs) {
        Dictionary<int, List<int>> adjList = new Dictionary<int, List<int>>();

        foreach (var pair in adjacentPairs) {
            AddToAdjList(adjList, pair[0], pair[1]);
            AddToAdjList(adjList, pair[1], pair[0]);
        }

        int start = FindStart(adjList);

        int[] result = BuildResult(adjList, adjacentPairs.Length + 1, start);

        return result;
    }

    private void AddToAdjList(Dictionary<int, List<int>> adjList, int key, int value) {
        if (!adjList.ContainsKey(key)) {
            adjList[key] = new List<int>();
        }
        adjList[key].Add(value);
    }

    private int FindStart(Dictionary<int, List<int>> adjList) {
        int start = 0;
        foreach (var pair in adjList) {
            if (pair.Value.Count == 1) {
                start = pair.Key;
                break;
            }
        }
        return start;
    }

    private int[] BuildResult(Dictionary<int, List<int>> adjList, int n, int start) {
        int[] result = new int[n];
        result[0] = start;
        result[1] = adjList[start][0];

        for (int i = 2; i < n; i++) {
            var list = adjList[result[i - 1]];
            result[i] = result[i - 2] == list[0] ? list[1] : list[0];
        }

        return result;
    }
}
