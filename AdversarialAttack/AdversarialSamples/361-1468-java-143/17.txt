class Solution {
    public boolean checkIfExist(int[] arr) {
        Map<Integer, Integer> map = new HashMap<>();
        int n = arr.length;
        int i = 0;
        while (i < n){
            map.put(arr[i], i);
            i++;
        }
        i = 0;
        while (i < n) {
            if (map.containsKey(2 * arr[i]) &&
                map.get(2 * arr[i]) != i) {
                return true;
            }
            i++;
        }

        return false;
    }
}

// TC: O(n) + O(n * n) => O(n ^ 2)
// SC: O(n)
// HashMap.containsKey() costs O(n) at worst case
