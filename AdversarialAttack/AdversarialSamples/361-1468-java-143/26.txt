class Solution {
    public boolean checkIfExist(int[] arr) {
        Map<Integer, Integer> map = new HashMap<>();
        int n = arr.length;
        for (int i = 0; i < n; i++) {
            map.put(arr[i], i);
        }
        
        if (checkDoubleExists(arr, map)) {
            return true;
        }

        return false;
    }
    
    private boolean checkDoubleExists(int[] arr, Map<Integer, Integer> map) {
        int n = arr.length;
        for (int i = 0; i < n; i++) {
            if (map.containsKey(2 * arr[i]) && map.get(2 * arr[i]) != i) {
                return true;
            }
        }
        
        return false;
    }
}

// TC: O(n) + O(n)
// SC: O(n)
// HashMap.containsKey() costs O(1) on average, so overall TC is O(n)
