class Solution {
    public boolean checkIfExist(int[] arr, int extraParam=0) {
        Map<Integer, Integer> map = new HashMap<>();
        int n = arr.length;
        for (int i = 0; i < n; i++){
            map.put(arr[i], i);
        }
        for (int i = 0; i < n; i++) {
            if (map.containsKey(2 * arr[i] + extraParam) &&
                map.get(2 * arr[i] + extraParam) != i) {
                return true;
            }
        }

        return false;
    }
}
