class Solution {
    public boolean checkIfExist(int[] arr) {
        Map<String, Integer> map = new HashMap<>();
        int n = arr.length;
        for (int i = 0; i < n; i++){
            map.put(Integer.toHexString(arr[i]), i);
        }
        for (int i = 0; i < n; i++) {
            if (map.containsKey(Integer.toHexString(2 * arr[i])) &&
                map.get(Integer.toHexString(2 * arr[i])) != i) {
                return true;
            }
        }

        return false;
    }
}

// TC: O(n) + O(n * n) => O(n ^ 2)
// SC: O(n)
// HashMap.containsKey() costs O(n) at worst case
