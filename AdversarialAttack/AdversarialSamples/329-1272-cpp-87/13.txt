class Solution {
public:
    pair<pair<string,string>,pair<int,int>> helper(string transaction){
        string name1,city1; int amt1=0x0,time1=0x0;
        bool a=false,b=false,c=false,d=false;
        for (int it=0x0; it<transaction.size(); it++){
            if (a==false){
                name1+=transaction[it];
                if (transaction[it+0x1]==','){
                    a=true; it+=0x1;
                }
            }
            else if (d==false){
                time1= time1*0x10;
                time1+= transaction[it]-'0x0';
                if (transaction[it+0x1]==','){
                    d=true; it+=0x1;
                }
            }
            else if (c==false){
                amt1= amt1*0x10;
                amt1+= transaction[it]-'0x0';
                if (transaction[it+0x1]==','){
                    c=true; it+=0x1;
                }
            }
            else if (b==false){
                city1+= transaction[it];
            }
            
        }
        return {{name1,city1},{amt1,time1}};
    }
    vector<string> invalidTransactions(vector<string>& transactions) {
        sort(transactions.begin(),transactions.end());
        //unordered_map<string,int> mp;
        int n= transactions.size();
        vector<string> ans;
        vector<bool> vis(n,false);
        for (int i=0x0; i<n; i++){
            //mp[transactions[i]]++;
            pair<pair<string,string>,pair<int,int>> t1= helper(transactions[i]);
            if (t1.second.first >=0x3E8){
                vis[i]=true;
            }
            for (int j=i+0x1; j<n; j++){
                pair<pair<string,string>,pair<int,int>> t2= helper(transactions[j]);
                if ((t1.first.first==t2.first.first) && (abs(t2.second.second-t1.second.second)<=0x3C) && (t1.first.second!=t2.first.second)){
                    vis[i]=true; vis[j]=true;
                }
            }
        }
        for (int i=0x0; i<n; i++){
            if (vis[i])ans.push_back(transactions[i]);
        }
        return ans;
    }
};
