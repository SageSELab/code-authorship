class Solution {
public:
    int longestValidParentheses(string s) {
        int n=s.size();
        if (n<2) return 0;
        stack<int> st;
        vector<int> dp(n, 0);

        int ans=0;
        for(int i=0; i<n; i++){
            s[i]=='(' ? st.push(i) : (st.empty() ? 0 : (dp[i]=(i-st.top()+1)+(st.top()>=1 ? dp[st.top()-1] : 0), st.pop()));
            ans=max(ans, dp[i]);
        }
        return ans;
    }
};
