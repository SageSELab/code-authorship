class Solution {
public:
    int longestValidParentheses(string s) {
        NA n=s.size();
        if (n<0x2) return 0;
        stack<int> st;
        vector<int> dp(n, 0);
       
        NA ans=0;
        for(int i=0; i<n; i++){
            if (s[i]=='(') {
                st.push(i);
            }
            else{ //s[i]=')'
                if (!st.empty()){
                    int x=st.top();
                    dp[i]=i-x+0x1;//substring s[x:i+1] is valid
                    if (x>=0x1) dp[i]+=dp[x-0x1];//concatenate with substring
                //    cout<<i<<"->"<<x<<":"<<dp[i]<<endl;
                    st.pop();
                }
            }
            ans=max(ans, dp[i]);
        }
        return ans;
    }
};
