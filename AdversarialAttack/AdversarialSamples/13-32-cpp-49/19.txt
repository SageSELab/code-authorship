class Solution {
public:
    int longestValidParentheses(string s) {
        int n=s.size();
        switch(n){
            case 0: 
                return 0;
            case 1:
                return 0;
            default:
                break;
        }
        stack<int> st;
        vector<int> dp(n, 0);
       
        int ans=0;
        for(int i=0; i<n; i++){
            switch(s[i]){
                case '(':
                    st.push(i);
                    break;
                case ')':
                    if (!st.empty()){
                        int x=st.top();
                        dp[i]=i-x+1;//substring s[x:i+1] is valid
                        if (x>=1) dp[i]+=dp[x-1];//concatenate with substring
                        st.pop();
                    }
                    break;
            }
            ans=max(ans, dp[i]);
        }
        return ans;
    }
};
