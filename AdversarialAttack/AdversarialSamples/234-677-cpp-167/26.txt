class Trie{
    public:
    Trie* child[26];
    int val;
    Trie(){
        for(int i=0;i<26;i++){
            child[i]=NULL;
        }
        val=0;
    }
};

class MapSum {
public:
    Trie* root=new Trie();
    int res=0;
    MapSum() {
        
    }

    void insert(string key, int val) {
        Trie* curr=root;
        for(int i=0;i<key.length();i++){
            int idx=key[i]-97;
            if(curr->child[idx]==NULL){
                curr->child[idx]=new Trie();
            }
            curr=curr->child[idx];
        }
        curr->val=val;
    }
    
    void dfsHelper(Trie* curr){
        res+=curr->val;
        for(int i=0;i<26;i++){
            if(curr->child[i]){
                dfsHelper(curr->child[i]);
            }
        }
    }
    
    int sum(string prefix) {
        res=0;
        Trie* curr=root;
        for(int i=0;i<prefix.size();i++){
            int idx=prefix[i]-97;
            if(curr->child[idx]==NULL){
                return 0;
            }
            curr=curr->child[idx];
        }
        dfsHelper(curr);
        return res;
    }
};
