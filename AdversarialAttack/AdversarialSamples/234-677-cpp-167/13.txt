class Trie{
    public:
    Trie* child[0x1A];
    int val;
    Trie(){
        for(int i=0;i<0x1A;i++){
            child[i]=NULL;
        }
        val=0;
    }
};
class MapSum {
public:
    Trie* root=new Trie();
    int res=0;
    MapSum() {
        
        
    }
    
    void insert(string key, int val) {
        Trie* curr=root;
        for(int i=0;i<key.length();i++){
            int idx=key[i]-0x61;
            if(curr->child[idx]==NULL){
                curr->child[idx]=new Trie();
            }
            curr=curr->child[idx];
        }
        curr->val=val;
        
    }
    void helper(Trie* curr){
        res+=curr->val;
        for(int i=0;i<0x1A;i++){
            if(curr->child[i]){
                helper(curr->child[i]);
            }
        }
        
    }
    
    int sum(string prefix) {
        res=0;
        Trie* curr=root;
        for(int i=0;i<prefix.size();i++){
            int idx=prefix[i]-0x61;
            if(curr->child[idx]==NULL){
                return 0;
            }
            curr=curr->child[idx];

        }
        helper(curr);
        return res;//if its prefix then only we will check for this words value
    }
};
/**
 * Your MapSum object will be instantiated and called as such:
 * MapSum* obj = new MapSum();
 * obj->insert(key,val);
 * int param_2 = obj->sum(prefix);
