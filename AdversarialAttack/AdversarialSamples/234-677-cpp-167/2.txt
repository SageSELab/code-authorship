class Trie{
    public:
    Trie* child[26];
    int val;
    Trie(){
        cout << "Initializing Trie node" << endl;
        for(int i=0;i<26;i++){
            child[i]=NULL;
        }
        val=0;
        cout << "Initialized child nodes to NULL and val to 0" << endl;
    }
};
class MapSum {
public:
    Trie* root=new Trie();
    int res=0;
    MapSum() {
        cout << "MapSum constructor called" << endl;
    }
    
    void insert(string key, int val) {
        cout << "Inserting key: " << key << " with value: " << val << endl;
        Trie* curr=root;
        for(int i=0;i<key.length();i++){
            int idx=key[i]-97;
            if(curr->child[idx]==NULL){
                cout << "Creating new Trie node for character: " << key[i] << endl;
                curr->child[idx]=new Trie();
            }
            curr=curr->child[idx];
        }
        curr->val=val;
        cout << "Value set for key: " << key << " as: " << val << endl;
    }
    void helper(Trie* curr){
        cout << "Calculating sum for Trie node" << endl;
        res+=curr->val;
        for(int i=0;i<26;i++){
            if(curr->child[i]){
                helper(curr->child[i]);
            }
        }
    }
    
    int sum(string prefix) {
        res=0;
        cout << "Calculating sum for prefix: " << prefix << endl;
        Trie* curr=root;
        for(int i=0;i<prefix.size();i++){
            int idx=prefix[i]-97;
            if(curr->child[idx]==NULL){
                cout << "Prefix not found" << endl;
                return 0;
            }
            curr=curr->child[idx];
        }
        helper(curr);
        cout << "Sum calculated for prefix: " << prefix << " is: " << res << endl;
        return res;
    }
};




/**
 * Your MapSum object will be instantiated and called as such:
 * MapSum* obj = new MapSum();
 * obj->insert(key,val);
 * int param_2 = obj->sum(prefix);
