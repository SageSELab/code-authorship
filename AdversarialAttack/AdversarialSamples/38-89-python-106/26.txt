class Solution:
    def grayCode(self, n):
        if n == 0:
            return [0]

        def generate_gray_code(k):
            if k == 0:
                return [0]
            t = generate_gray_code(k-1)
            return t + [i+(1<<(k-1)) for i in t][::-1]

        return generate_gray_code(n)
