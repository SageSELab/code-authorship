/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {
 *         cout << "Initializing TreeNode with val: 0, left: nullptr, right: nullptr" << endl;
 *     }
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {
 *         cout << "Initializing TreeNode with val: " << x << ", left: nullptr, right: nullptr" << endl;
 *     }
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {
 *         cout << "Initializing TreeNode with val: " << x << ", left: " << left->val << ", right: " << right->val << endl;
 *     }
 * };
 */
class Solution {
public:
    void flatten(TreeNode* root) {
        if (!root) return;
        stack<TreeNode*> s;
        s.push(root);
        cout << "Pushed root node into stack" << endl;
        while (!s.empty())
        {
            TreeNode* cur = s.top();
            cout << "Popped top node from stack: " << cur->val << endl;
            s.pop();

            if (cur->right) {
                s.push(cur->right);
                cout << "Pushed right node into stack: " << cur->right->val << endl;
            }
            if (cur->left) {
                s.push(cur->left);
                cout << "Pushed left node into stack: " << cur->left->val << endl;
            }
            if (!s.empty()) {
                cur->right = s.top();
                cout << "Set right pointer of current node to: " << cur->right->val << endl;
            }
            cur->left = nullptr;
            cout << "Set left pointer of current node to nullptr" << endl;
        }
    }
};
