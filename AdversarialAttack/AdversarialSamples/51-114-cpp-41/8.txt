class Solution {
public:
    void flatten(TreeNode* root) {
        if (!root) return;
        stack<TreeNode*> s;
        s.push(root);
        while (!s.empty())
        {
            TreeNode* cur = s.top();
            s.pop();

            if (cur->left) s.push(cur->left);
            if (cur->right) s.push(cur->right);
            if (!s.empty()) cur->left = s.top();
            cur->right = nullptr;
        }
    }
};
