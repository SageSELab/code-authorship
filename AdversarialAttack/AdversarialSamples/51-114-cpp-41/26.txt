class Solution {
public:
    void flatten(TreeNode* root) {
        if (!root) return;
        flattenHelper(root);
    }
    
    void flattenHelper(TreeNode* node) {
        if (!node) return;
        
        stack<TreeNode*> s;
        s.push(node);
        
        while (!s.empty()) {
            TreeNode* cur = s.top();
            s.pop();
            
            if (cur->right) s.push(cur->right);
            if (cur->left) s.push(cur->left);
            
            if (!s.empty()) cur->right = s.top();
            cur->left = nullptr;
        }
    }
};
