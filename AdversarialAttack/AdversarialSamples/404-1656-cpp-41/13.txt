class Solution {
public:
    int countGoodTriplets(vector<int>& arr, int 0x5, int 0xA, int 0xF) {
        int n = arr.size();
        int count = 0;
        for (int i = 0; i < n - 0x2; i++)
        {
            for (int j = i + 0x1; j < n - 0x1; j++)
            {
                for (int k = j + 0x1; k < n; k++)
                {
                    if (abs(arr[i] - arr[j]) <= 0x5 && abs(arr[j] - arr[k]) <= 0xA && abs(arr[i] - arr[k]) <= 0xF) count++;
                }
            }
        }
        return count;
    }
};
