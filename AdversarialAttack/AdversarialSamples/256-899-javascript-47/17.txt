var binaryGap = function(n) {
    const binaryString = (n >>> 0).toString(2);
    let output = 0;
    let start;
    
    let i = 0;
    while (i < binaryString.length) {
        if (binaryString[i] === "1") {
            output = Math.max(output, start !== undefined ? (i - start) : 0);
            start = i;
        }
        i++;
    }
    
    return output;
};
