class Solution {
    public int maxProfit(int[] prices) {
        int [][]dp= new int[prices.length+2][2];
        System.out.println("Initialized dp array with size: " + (prices.length + 2));
        for(int i=prices.length-1;i>=0;i--)
        {
            System.out.println("Current index i: " + i);
            for(int buy=0;buy<2;buy++)
            {
                System.out.println("Buy flag: " + buy);
                if(buy==1)
                    dp[i][buy]=Math.max(-prices[i]+dp[i+1][0],dp[i+1][1]);
                else
                    dp[i][buy]=Math.max(prices[i]+dp[i+2][1],dp[i+1][0]);
                System.out.println("Updated dp[" + i + "][" + buy + "]: " + dp[i][buy]);
            }
        }
        System.out.println("Final result: " + dp[0][1]);
        return dp[0][1];
    }
}
