class Solution {
    public int maxProfit(int[] prices) {
        int [][]dp= new int[(2 * prices.length) + 4][2];
        for(int i=(2 * prices.length) + 2;i>=0;i--)
        {
            for(int buy=0;buy<2;buy++)
            {
                if(buy==1)
                    dp[i][buy]=Math.max(-prices[i/2]+dp[i+1][0],dp[i+1][1]);
                else
                    dp[i][buy]=Math.max(prices[i/2]+dp[i+2][1],dp[i+1][0]);          
            }
        }
        return dp[0][1];
    }
}
