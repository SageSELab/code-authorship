class Solution {
    public int maxProfit(int[] prices) {
        int [][]dp= new int[prices.length+2][2];
        int i = prices.length - 1;
        while (i >= 0)
        {
            int buy = 0;
            while (buy < 2)
            {
                if(buy==1)
                    dp[i][buy]=Math.max(-prices[i]+dp[i+1][0],dp[i+1][1]);
                else
                    dp[i][buy]=Math.max(prices[i]+dp[i+2][1],dp[i+1][0]);          
                buy++;
            }
            i--;
        }
        return dp[0][1];
    }
}
