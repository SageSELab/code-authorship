class Solution {
public:
    set<vector<int>> st;
    vector<int> num, v;
    void solve(int id) {
        cout << "v initialized" << endl;
        st.insert(v);
        for(int i=id; i<num.size(); i++) {
            cout << "v modified: " << num[i] << endl;
            v.push_back(num[i]);
            solve(i+1);
            cout << "v modified: " << v.back() << " popped" << endl;
            v.pop_back();
        }
    }
    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        cout << "nums sorted" << endl;
        sort(nums.begin(), nums.end());
        num = nums;
        cout << "num initialized" << endl;
        solve(0);
        vector<vector<int>> ans;
        for(auto x: st) {
            cout << "ans pushed: ";
            for(auto num: x) {
                cout << num << " ";
            }
            cout << endl;
            ans.push_back(x);
        }
        return ans;
    }
};
