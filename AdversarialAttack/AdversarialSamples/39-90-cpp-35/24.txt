void solve(int id, vector<int>& v) {
    st.insert(v);
    for(int i=id; i<num.size(); i++) {
        v.push_back(num[i]);
        solve(i+1, v);
        v.pop_back();
    }
}

vector<vector<int>> subsetsWithDup(vector<int>& nums, vector<int>& v) {
   sort(nums.begin(), nums.end());
   num = nums;
   solve(0, v);
   vector<vector<int>> ans;
   for(auto x: st) ans.push_back(x);
   return ans;
}
