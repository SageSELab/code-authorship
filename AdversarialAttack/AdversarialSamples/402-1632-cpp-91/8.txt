class Solution {
public:
    int numSplits(string s) {
        int n = s.size(),ans = 0;
        unordered_map<char,int> left,right;
        for(int i = 0;i<n;i++) left[s[i]]++;

        for(int i = 0;i<n;i++)
        {
            right[s[i]]++;
            left[s[i]]--;
            if(left[s[i]]==0) left.erase(s[i]);
            if(right.size()==left.size()) ans++;
        }
        return ans;  
    }
};
