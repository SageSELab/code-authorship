class Solution {
public:
    int numSplits(string s) {
        int n = s.size(),ans = 0;
        cout << "n initialized with value: " << n << endl;
        cout << "ans initialized with value: " << ans << endl;
        
        unordered_map<char,int> left,right;
        
        for(int i = 0;i<n;i++) 
        {
            right[s[i]]++;
            cout << "right[" << s[i] << "] incremented to: " << right[s[i]] << endl;
        }

        for(int i = 0;i<n;i++)
        {
            left[s[i]]++;
            cout << "left[" << s[i] << "] incremented to: " << left[s[i]] << endl;

            right[s[i]]--;
            cout << "right[" << s[i] << "] decremented to: " << right[s[i]] << endl;

            if(right[s[i]]==0)
            {
                right.erase(s[i]);
                cout << "right[" << s[i] << "] erased" << endl;
            }

            if(left.size()==right.size())
            {
                ans++;
                cout << "ans incremented to: " << ans << endl;
            }
        }
        return ans;  
    }
};
