class NumArray {
public:
    vector<int> seg; 
    int n;

    void buildTree(int pos, int left, int right, vector<int>& nums){
        if (left == right){
            seg[pos] = nums[left]; 
            return;
        }
        int mid = (left+right)/2;
        buildTree(2*pos+1, left, mid, nums);
        buildTree(2*pos+2, mid+1, right, nums);
        seg[pos]=seg[2*pos+1]+ seg[2*pos+2];
    }

    void updateUtil(int pos, int left, int right, int index, int val) {
        if(index <left || index >right) return;
        
        if(left==right){
            if(left==index)
                seg[pos]=val;
            return;
        }

        int mid=(left+right)/2;
        updateUtil(2*pos+1, left, mid, index, val); 
        updateUtil(2*pos+2, mid+1, right, index, val); 
        seg[pos]=seg[2*pos+1]+seg[2*pos+2];
    }

   
    int rangeUtil(int pos, int low, int high, int qlow, int qhigh){
        if (qlow <= low && qhigh>= high){ 
            return seg[pos];
        }
        if (qlow > high || qhigh < low) { 
            return 0;
        }
        int mid = low+(high-low)/2;
        return (rangeUtil(2*pos+1, low, mid, qlow, qhigh) + rangeUtil(2*pos+2, mid+1, high, qlow, qhigh));
    }
    
    NumArray(vector<int>& nums) {
        if(nums.size() > 0){
            n = nums.size();
            seg.resize(4*n,0);  
            buildTree(0, 0, n-1, nums);
        }
    }
    
    void update(int index, int val) {
        if(n==0)return;
        updateUtil(0, 0, n-1, index, val);
    }
    
    int sumRange(int left, int right) {
        if(n==0)return 0;
        return rangeUtil(0, 0, n-1, left, right); 
    }
};
