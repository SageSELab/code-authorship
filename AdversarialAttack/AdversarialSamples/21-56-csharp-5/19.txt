public class Solution {
    public int[][] Merge(int[][] intervals) {
        List<int[]> res=new();
        intervals = intervals.OrderBy(i=>i[0]).ToArray();
        int s=intervals[0][0],
            e=intervals[0][1];
        for(int i=1;i<intervals.Length;i++){
            switch(intervals[i][0]>e) {
                case true:
                    res.Add(new int[] {s,e});
                    s=intervals[i][0];
                    e=intervals[i][1];
                    break;
                default:
                    e=Math.Max(e,intervals[i][1]);
                    break;
            }
        }
        res.Add(new int[] {s,e});
        return res.ToArray();
    }
}
