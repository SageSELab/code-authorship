public class Solution {
    public int[][] Merge(int[][] intervals) {
        List<int[]> res=new();
        Console.WriteLine("intervals initialized");
        intervals = intervals.OrderBy(i=>i[0]).ToArray();
        Console.WriteLine("intervals sorted");
        int s=intervals[0][0],
            e=intervals[0][1];
        Console.WriteLine("s and e initialized");
        for(int i=1;i<intervals.Length;i++){
            if(intervals[i][0]>e){
                res.Add(new int[] {s,e});
                Console.WriteLine("new interval added to res");
                s=intervals[i][0];
                e=intervals[i][1];
                Console.WriteLine("s and e updated");
            }
            else
                e=Math.Max(e,intervals[i][1]);
                Console.WriteLine("e updated");
        }
        res.Add(new int[] {s,e});
        Console.WriteLine("last interval added to res");
        return res.ToArray();
    }
}
