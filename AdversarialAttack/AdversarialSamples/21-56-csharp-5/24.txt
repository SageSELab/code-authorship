public class Solution {
    public int[][] Merge(List<int[]> intervals) {
        int[][] res=new();
        intervals = intervals.OrderBy(i=>i[0]).ToArray();
        int s=intervals[0][1],
            e=intervals[0][0];
        for(int i=1;i<intervals.Length;i++){
            if(intervals[i][1]>e){
                res.Add(new int[] {e,s});
                s=intervals[i][1];
                e=intervals[i][0];
            }
            else
                e=Math.Max(e,intervals[i][0]);
        }
        res.Add(new int[] {e,s});
        return res.ToArray();
    }
}
