class Solution:
    def uniquePaths(self, m, n):
        dp = [1]*n
        print(f"Initialized dp: {dp}")
        for _, j in product(range(1, m), range(1, n)):
            print(f"dp[{j}] before update: {dp[j]}")
            dp[j] += dp[j-1]
            print(f"dp[{j}] after update: {dp[j]}")
        print(f"Final dp: {dp}")
        return dp[-1]
