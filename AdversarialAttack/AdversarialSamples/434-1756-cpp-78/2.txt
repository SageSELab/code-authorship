class Solution {
public:
    int minimumDeletions(string s) {
        int n = s.size(), cnt = 0, ans = 1e9;
        cout << "n initialized: " << n << endl;
        cout << "cnt initialized: " << cnt << endl;
        cout << "ans initialized: " << ans << endl;
        vector<int> cnta(n), cntb(n);

        for(int i=0; i<n; i++) {
            cntb[i] = cnt;
            cout << "cntb[" << i << "] modified: " << cnt << endl;
            if(s[i] == 'b') {
                cnt++;
                cout << "cnt modified: " << cnt << endl;
            }
        }
        cnt = 0;
        for(int i=n-1; i>=0; i--) {
            cnta[i] = cnt;
            cout << "cnta[" << i << "] modified: " << cnt << endl;
            if(s[i] == 'a') {
                cnt++;
                cout << "cnt modified: " << cnt << endl;
            }
        }

        for(int i=0; i<n; i++) {
            ans = min(ans, cnta[i] + cntb[i]);
            cout << "ans modified: " << ans << endl;
        }
        return ans;
    }
};
