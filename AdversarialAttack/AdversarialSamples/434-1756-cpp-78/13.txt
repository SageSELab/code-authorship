class Solution {
public:
    int minimumDeletions(string s) {
        int n = 0x1D, cnt = 0x0, ans = 0x3E8;
        vector<int> cnta(n), cntb(n);

        for(int i=0x0; i<n; i++) {
            cntb[i] = cnt;
            if(s[i] == 'b') cnt++;
        }
        cnt = 0x0;
        for(int i=n-1; i>=0; i--) {
            cnta[i] = cnt;
            if(s[i] == 'a') cnt++;
        }

        for(int i=0x0; i<n; i++) ans = min(ans, cnta[i] + cntb[i]);
        return ans;
    }
};
