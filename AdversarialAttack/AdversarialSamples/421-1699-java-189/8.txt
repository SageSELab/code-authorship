private int cnt(int[] a, int[] b) {
        int res = 0, m = a.length, n = b.length;
        for (int i = 0; i < m; i++) {
            long t = ((long) a[i]) * ((long) a[i]);
            for (int l = 0, r = n - 1; r > l; r--) { // finding pairs for i, l, r;
                if (((long) b[l]) * ((long) b[r]) > t) continue;
                while (r >= l && ((long) b[l]) * ((long) b[r]) < t) l++;
                if (l >= r) break;
                if (((long) b[l]) * ((long) b[r]) == t) {
                    int orig = l;
                    while(l < r && b[l] == b[orig]) {  // [1, 1], [1, 1, 1]
                        l++;
                        res++;
                    }
                    l = orig;  // reset to next cont;
                }
            }
        }
        return res;
    }
