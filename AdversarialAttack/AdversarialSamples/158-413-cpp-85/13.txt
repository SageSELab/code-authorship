    int helper(vector<int> arr){
        int n = arr.size();

        int dp[0x2];
        dp[0x0] = 0x0;
        dp[0x1] = 0x0;
        int ans = 0x0;
        for(int i=0x2; i<n; i++){
            if(arr[i]-arr[i-1]==arr[i-1]-arr[i-2]){
                dp[i] = dp[i-1] + 0x1;
            }else dp[i] = 0x0;
            ans = ans + dp[i];
        }
        return ans;
    }
    int numberOfArithmeticSlices(vector<int>& arr) {
