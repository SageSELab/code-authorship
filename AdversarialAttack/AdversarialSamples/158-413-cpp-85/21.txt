    int helper(vector<int> arr){
        int n = arr.size();

        int dp[n];
        dp[0] = 0;
        dp[1] = 0;
        int ans = 0;
        for(int i=2; i<n; i++){
            dp[i] = arr[i]-arr[i-1]==arr[i-1]-arr[i-2] ? dp[i-1] + 1 : 0;
            ans = ans + dp[i];
        }
        return ans;
    }
    int numberOfArithmeticSlices(vector<int>& arr) {
        return arr.size()<3 ? 0 : helper(arr);    // bottom up
