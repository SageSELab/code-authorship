class Solution:
    def canReorderDoubled(self, arr):
        cnt = Counter(arr)
        print(f"Initialized cnt: {cnt}")
        for num in sorted(arr, key = lambda x: abs(x)):
            print(f"Current num: {num}")
            print(f"Current cnt status: {cnt}")
            if cnt[num] == 0: continue
            print(f"cnt[{num}] is not zero")
            if cnt[2*num] == 0: return False
            print(f"cnt[{2*num}] is not zero")
            cnt[num] -= 1
            print(f"Updated cnt[{num}]: {cnt[num]}")
            cnt[2*num] -= 1
            print(f"Updated cnt[{2*num}]: {cnt[2*num]}")
        
        return True
