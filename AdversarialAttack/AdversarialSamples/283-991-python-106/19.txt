class Solution:
    def canReorderDoubled(self, arr):
        cnt = Counter(arr)
        for num in sorted(arr, key = lambda x: abs(x)):
            switch num:
                case cnt[num] == 0:
                    continue
                case cnt[2*num] == 0:
                    return False
                default:
                    cnt[num] -= 1
                    cnt[2*num] -= 1
        
        return True
