class Solution:
    def canReorderDoubled(self, arr):
        cnt = Counter(arr)
        num = 0
        while num < len(arr):
            if cnt[arr[num]] == 0: continue
            if cnt[2*arr[num]] == 0: return False
            cnt[arr[num]] -= 1
            cnt[2*arr[num]] -= 1
            num += 1
        
        return True
