class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        int n=nums.size();
        cout << "n initialized with value: " << n << endl;
        int n_3=n/3;
        cout << "n_3 initialized with value: " << n_3 << endl;
        
        unordered_set<int> ans;
        unordered_map<int, int> mp;
        
        for(int x : nums){
            cout << "x value is: " << x << endl;
            
            if (mp.count(x)==0) {
                mp[x]=1;
                cout << "mp[" << x << "] initialized with value: " << 1 << endl;
            } else {
                mp[x]++;
                cout << "mp[" << x << "] updated to value: " << mp[x] << endl;
            }
            
            if (mp[x]>n_3) {
                ans.insert(x);
                cout << "Inserting " << x << " into ans" << endl;
            }
        }
        return vector<int>(ans.begin(), ans.end());
    }
};
