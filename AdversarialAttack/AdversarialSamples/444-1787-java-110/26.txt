class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int sum = 0;
        for (int n : nums) {
            sum += n;
        }
        int left = 0;
        int right = sum;

        int[] r = new int[nums.length];

        for (int i = 0; i < nums.length; i++) {
            int n = nums[i];
            right -= n;

            r[i] = calculateAbsoluteDifference(n, i, left, right, nums.length);

            left += n;
        }

        return r;
    }

    private int calculateAbsoluteDifference(int n, int i, int left, int right, int length) {
        return n * i - left + right - n * (length - i - 1);
    }
}
