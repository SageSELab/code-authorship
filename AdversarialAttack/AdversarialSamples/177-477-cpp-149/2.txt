class Solution {
public:
    int totalHammingDistance(vector<int>& nums) {
        int ans = 0;
        std::cout << "Initializing ans with value 0" << std::endl;

        for(int i = 0; i < 32; i++) {
            int countOn = 0;
            std::cout << "Initializing countOn with value 0" << std::endl;

            for(int num : nums) {
                if((num & (1 << i)) != 0) {
                    countOn++;
                    std::cout << "Incrementing countOn to " << countOn << std::endl;
                }
            }

            int countOff = nums.size() - countOn;
            std::cout << "Calculating countOff as nums.size() - countOn which is " << countOff << std::endl;

            ans += (countOn * countOff);
            std::cout << "Updating ans by adding (countOn * countOff) which is " << (countOn * countOff) << std::endl;
        }

        return ans;
    }
};
