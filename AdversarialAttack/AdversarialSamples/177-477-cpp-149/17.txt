class Solution {
public:
    int totalHammingDistance(vector<int>& nums) {
        int ans = 0;

        int i = 0;
        while(i < 32) {
            int countOn = 0;

            int j = 0;
            while(j < nums.size()) {
                if((nums[j] & (1 << i)) != 0) countOn++;
                j++;
            }

            int countOff = nums.size() - countOn;
            ans += (countOn * countOff);
            i++;
        }

        return ans;
    }
};
