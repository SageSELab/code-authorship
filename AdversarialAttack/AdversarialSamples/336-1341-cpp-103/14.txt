class Solution {
public:
    int balancedStringSplit(string s) {
        int ans = 0,res =0;
        for(auto i : s )
        {
            if(i == 76)
                ans++;
            else if(i == 82)
                ans--;
            
            if(ans == 0)
                res++;
        }
        
        return res;
        
    }
};
