class Solution {
public:
    int balancedStringSplit(string s) {
        int ans = 0,res =0;
        int i = 0;
        while(i < s.size())
        {
            if(s[i]=='L')
                ans++;
            else if(s[i]=='R')
                ans--;
                
            if(ans == 0)
                res++;
            
            i++;
        }
        
        return res;
    }
};
