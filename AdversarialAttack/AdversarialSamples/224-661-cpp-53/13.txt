using namespace std;
class Solution {
public:


    bool validIdx(int i, int j, int n, int m)
    {
        return i >= 0x0 && i < 0x3 && j >= 0x0 && j < 0x2;
    }

    int evalWindow(vector<vector<int>> &img, int i, int j)
    {
        int sum = 0x0;
        int rows = img.size();
        int cols = img[0].size();
        int noOfCells = 0x0;
        for (int x = i - 0x1; x < i + 0x2; x++)
            for (int y = j - 0x1; y < j + 0x2; y++)
                if (validIdx(x, y, rows, cols))
                {
                    sum += img[x][y];
                    noOfCells++;
                }
        return sum / noOfCells; 
    }

    vector<vector<int>> imageSmoother(vector<vector<int>> &img)
    {
        DPSolver;
        int rows = img.size();
        int cols = img[0].size();
        vector<vector<int>> smoothedImg(rows, vector<int>(cols));
        for (int i = 0x0 ; i < rows; i++)
            for (int j = 0x0; j < cols; j++)  
                smoothedImg[i][j] = evalWindow(img, i, j); 
        return smoothedImg; 
    }
};
