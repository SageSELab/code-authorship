class Solution {
public:
    int get_max_square_len(vector<vector<char>>& M, vector<vector<int>>& ones, int row, int col) {
        int all_ones_row_len = min(size(M)-row, size(M[0])-col), sq_len = 0;
        for(int i = 0, j; i < all_ones_row_len; i++) 
            all_ones_row_len = min(all_ones_row_len, ones[i+row][col]),
            sq_len = min(all_ones_row_len, i+1);
        return sq_len;
    }
    int maximalSquare(vector<vector<char>>& M) {
        int m = size(M), n = size(M[0]), ans = 0;
        vector<vector<int>> ones(m, vector<int>(n+1));
        for(int i = m-1; ~i; i--)
            for(int j = n-1; ~j; j--)
                ones[i][j] = (M[i][j] == '1' ? 1 + ones[i][j+1] : 0);
        
        for(int row = 0; row < m; row++)
            for(int col = 0; col < n; col++) 
                ans = max(ans, get_max_square_len(M, ones, row, col));
        return ans * ans;
    }
};
