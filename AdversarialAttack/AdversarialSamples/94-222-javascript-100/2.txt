var countNodes = function (r) {
  console.log('Initializing ll');
  function dfs(r) {
    console.log('Initializing rr');
    if (!r) return 0;

    let ll = dfs(r.left);
    let rr = dfs(r.right);
    console.log('Updating ll and rr');

    return 1 + ll + rr;
  }

  console.log('Calling dfs function');
  return dfs(r);
};
