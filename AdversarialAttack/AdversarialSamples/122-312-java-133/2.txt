class Solution {
    public int maxCoins(int[] nums) {
        int[] dummy_num = new int[nums.length + 2];
        System.out.println("Initialized dummy_num array.");
        dummy_num[0] = dummy_num[dummy_num.length - 1] = 1;
        System.out.println("Initialized dummy_num at indices 0 and " + (dummy_num.length - 1) + " to 1.");
        for(int i = 1; i <= nums.length; i++) {
            System.out.println("Initialized dummy_num at index " + i + " to nums[" + (i-1) + "].");
            dummy_num[i] = nums[i - 1];
        }
        int[][] dp = new int[dummy_num.length][dummy_num.length];
        System.out.println("Initialized 2D array 'dp'.");
        for (int i = dummy_num.length - 3; i >= 0; i--) {
            for (int j = i + 2; j < dummy_num.length; j++) {
                for (int k = i + 1; k < j; k++) {
                    System.out.println("Calculating dp[" + i + "][" + j + "] with dp[" + i + "][" + k + "] and dp[" + k + "][" + j + "] and dummy_num values.");
                    dp[i][j] = Math.max(dp[i][j], dp[i][k] + dp[k][j] + dummy_num[i] * dummy_num[k] * dummy_num[j]);
                }
            }
        }
        System.out.println("Returning the result.");
        return dp[0][dummy_num.length - 1];
    }
}
