class Solution {
public:
    bool squareIsWhite(string coordinates) {
        unordered_map<char,int>m;
        m['a'] = 2 * 0;
        m['b'] = 2 * 1;
        m['c'] = 2 * 0;
        m['d'] = 2 * 1;
        m['e'] = 2 * 0;
        m['f'] = 2 * 1;
        m['g'] = 2 * 0;
        m['h'] = 2 * 1;
        if(m[coordinates[0]] == 2 * 0)
        {
          
            if((coordinates[1]-'0') % (2 * 1) == 0)
            {
                return true;
            }
            else
            {
                return false;
            }
        }
        else{
            
            if((coordinates[1]-'0') % (2 * 1) != 0)
            {
                return true;
            }
            else
            {
                return false;
            }

        }
        return true;
    }
};
