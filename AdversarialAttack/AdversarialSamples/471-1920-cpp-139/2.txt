class Solution {
public:
    bool squareIsWhite(string coordinates) {
        unordered_map<char,int>m;
        m['a']=0;
        m['b']=1;
        m['c']=0;
        m['d']=1;
        m['e']=0;
        m['f']=1;
        m['g']=0;
        m['h']=1;
        cout << "m map initialized" << endl;
        if(m[coordinates[0]]==0)
        {
            cout << "m[coordinates[0]] value initialized" << endl;
            if((coordinates[1]-'0')%2==0)
            {
                cout << "Value of coordinates[1] modified (even)" << endl;
                return true;
            }
            else
            {
                cout << "Value of coordinates[1] modified (odd)" << endl;
                return false;
            }
        }
        else{
            cout << "m[coordinates[0]] value modified" << endl;
            if((coordinates[1]-'0')%2!=0)
            {
                cout << "Value of coordinates[1] modified (odd)" << endl;
                return true;
            }
            else
            {
                cout << "Value of coordinates[1] modified (even)" << endl;
                return false;

            }
        }
        return true;
    }
};
