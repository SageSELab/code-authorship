class Solution {
public:
    int getColor(char c) {
        unordered_map<char,int> m;
        m['a'] = 0; m['b'] = 1; m['c'] = 0; m['d'] = 1;
        m['e'] = 0; m['f'] = 1; m['g'] = 0; m['h'] = 1;
        return m[c];
    }

    bool isEven(int num) {
        return num % 2 == 0;
    }

    bool squareIsWhite(string coordinates) {
        int color = getColor(coordinates[0]);
        
        if (color == 0) {
            return isEven(coordinates[1] - '0');
        } else {
            return !isEven(coordinates[1] - '0');
        }
    }
};
