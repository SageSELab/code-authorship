class Solution {
public:
    bool squareIsWhite(string coordinates) {
        unordered_map<int,int>m;
        m[97]=0;
        m[98]=1;
        m[99]=0;
        m[100]=1;
        m[101]=0;
        m[102]=1;
        m[103]=0;
        m[104]=1;
        if(m[static_cast<int>(coordinates[0])]==0)
        {
          
            if((coordinates[1]-'0')%2==0)
            {
                return true;
            }
            else
            {
                return false;
            }
        }
        else{
            
            if((coordinates[1]-'0')%2!=0)
            {
                return true;
            }
            else
            {
                return false;
            }

        }
        return true;
    }
};
