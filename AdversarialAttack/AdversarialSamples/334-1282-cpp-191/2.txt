const int ALPHABET_SIZE = 26;

/* The structure of a trie node */
struct TrieNode
{
    struct TrieNode* children[ALPHABET_SIZE];
    int count = 0;
};

/* Creates a new trie node and returns the pointer */
struct TrieNode* getNode()
{
    struct TrieNode* newNode = new TrieNode;

    for(int i = 0; i < ALPHABET_SIZE; i++)
        newNode->children[i] = nullptr;
    
    newNode->count = 0;

    cout << "New trie node initialized with count 0" << endl;

    return newNode;
}

/* Inserts the given string to the collection */
void insert(struct TrieNode* root, string str)
{
    struct TrieNode* pCrawl = root;

    for(int i = 0; i < str.length(); i++)
    {
        int index = str[i]-'a';

        if(!pCrawl->children[index])
        {
            pCrawl->children[index] = getNode();
            cout << "New trie node initialized with count 0" << endl;
        }
        
        pCrawl = pCrawl->children[index];
    }

    pCrawl->count++;
    cout << "Node count incremented to " << pCrawl->count << endl;
}

/* Returns the count of strings which are valid */
int search(struct TrieNode* root, string str, bool firstSeen, char firstLetter)
{
    if(!root)
        return 0;
    
    int count = 0;
    
    if(firstSeen)
    {
        count += root->count;
        cout << "Adding count " << root->count << endl;
    }

    for(int i = 0; i < str.length(); i++)
    {
        int index = str[i] - 'a';
        
        if(str[i] == firstLetter)
        {
            count += search(root->children[index], str, true, firstLetter);
            cout << "Recursive call with first letter " << firstLetter << endl;
        }
        else
        {
            count += search(root->children[index], str, firstSeen, firstLetter);
            cout << "Recursive call without changing parameters" << endl;
        }
    }

    return count;
}

class Solution
{
public:
    vector<int> findNumOfValidWords(vector<string>& words, vector<string>& puzzles);
};

vector<int> Solution :: findNumOfValidWords(vector<string>& words, vector<string>& puzzles)
{
    struct TrieNode* root = getNode();
    cout << "Root trie node initialized" << endl;
    
    for(auto str : words)
    {
        set<char> temp;
        temp.insert(str.begin(), str.end);
        cout << "Set created with unique characters in string" << endl;
        
        string sorted = "";
        for(auto ele : temp)
            sorted += ele;
        
        insert(root, sorted);
    }
    
    vector<int> count;
    for(auto puzzle : puzzles)
    {
        char firstLetter = puzzle[0];
        sort(puzzle.begin(), puzzle.end());
        cout << "Puzzle sorted" << endl;
        count.push_back(search(root, puzzle, false, firstLetter));
        cout << "Search function called for puzzle" << endl;
    }
    
    return count;
}
