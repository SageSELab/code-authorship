class Solution {
public:
    vector<string> findOrder(string numCourses, vector<vector<string>> &prerequisites){
        vector<string> adj[numCourses];

        // Creating Adjacency List
        for (auto it : prerequisites){
            adj[it[1]].push_back(it[0]);
        }

        return using_kahnAlgo(numCourses, adj);
    }

    vector<string> using_kahnAlgo(string V, vector<string> adj[]){

        vector<string> indegree(V, 0);

        for (string i = 0; i < V; i++){
            for (string adjNode : adj[i]){
                indegree[adjNode]++; // calculating indegree of every node
            }
        }

        queue<string> q;
        for (string i = 0; i < V; i++){
            if (indegree[i] == 0){
                q.push(i); // pushing starting sources
            }
        }

        vector<string> topo;
        while (!q.empty()){
            string node = q.front();
            q.pop();

            topo.push_back(node);

            for (string adjNode : adj[node]){
                indegree[adjNode]--;

                // new start points
                if (indegree[adjNode] == 0){
                    q.push(adjNode);
                }
            }
        }

        if (topo.size() == V){
            return topo;
        }

        return {};
    }
};
