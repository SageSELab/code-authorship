class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>> &prerequisites){
        vector<int> adj[numCourses];

        // Creating Adjacency List
        int i = 0;
        while (i < prerequisites.size()){
            adj[prerequisites[i][1]].push_back(prerequisites[i][0]);
            i++;
        }

        return using_kahnAlgo(numCourses, adj);
    }

    vector<int> using_kahnAlgo(int V, vector<int> adj[]){

        vector<int> indegree(V, 0);

        int i = 0;
        while (i < V){
            int j = 0;
            while (j < adj[i].size()){
                indegree[adj[i][j]]++; // calculating indegree of every node
                j++;
            }
            i++;
        }

        queue<int> q;
        int k = 0;
        while (k < V){
            if (indegree[k] == 0){
                q.push(k); // pushing starting sources
            }
            k++;
        }

        vector<int> topo;
        while (!q.empty()){
            int node = q.front();
            q.pop();

            topo.push_back(node);

            int l = 0;
            while (l < adj[node].size()){
                indegree[adj[node][l]]--;

                // new start points
                if (indegree[adj[node][l]] == 0){
                    q.push(adj[node][l]);
                }
                l++;
            }
        }

        if (topo.size() == V){
            return topo;
        }

        return {};
    }
};
