class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>> &prerequisites){
        vector<int> adj[numCourses];

        cout << "Adjacency List Initialization" << endl;
        // Creating Adjacency List
        for (auto it : prerequisites){
            adj[it[1]].push_back(it[0]);
            cout << "Added prerequisite: " << it[1] << " " << it[0] << endl;
        }
        
        cout << "Calling using_kahnAlgo function" << endl;
        return using_kahnAlgo(numCourses, adj);
    }

    vector<int> using_kahnAlgo(int V, vector<int> adj[]){
        vector<int> indegree(V, 0);

        cout << "Indegree Initialization" << endl;
        for (int i = 0; i < V; i++){
            for (int adjNode : adj[i]){
                indegree[adjNode]++; // calculating indegree of every node
                cout << "Incremented indegree: " << adjNode << endl;
            }
        }

        queue<int> q;
        for (int i = 0; i < V; i++){
            if (indegree[i] == 0){
                q.push(i); // pushing starting sources
                cout << "Pushed source node: " << i << endl;
            }
        }

        vector<int> topo;
        while (!q.empty()){
            int node = q.front();
            q.pop();

            topo.push_back(node);
            cout << "Added node to topological order: " << node << endl;

            for (int adjNode : adj[node]){
                indegree[adjNode]--;
                cout << "Decremented indegree: " << adjNode << endl;

                // new start points
                if (indegree[adjNode] == 0){
                    q.push(adjNode);
                    cout << "Pushed new source node: " << adjNode << endl;
                }
            }
        }

        if (topo.size() == V){
            return topo;
        }

        return {};
    }
};
