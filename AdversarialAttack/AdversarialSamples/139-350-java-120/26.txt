class Solution {
    public int[] intersect(int[] nums1, int[] nums2) {
        Arrays.sort(nums1);
        Arrays.sort(nums2);
        ArrayList<Integer> arr = new ArrayList<Integer>();
        intersectArrays(nums1, nums2, arr);
        return convertToArray(arr);
    }
    
    private void intersectArrays(int[] nums1, int[] nums2, ArrayList<Integer> arr) {
        int i = 0, j = 0;
        while(i < nums1.length && j < nums2.length){
            if(nums1[i] < nums2[j]) {
                i++;
            } else if(nums1[i] > nums2[j]){
                j++;
            } else{
                arr.add(nums1[i]);
                i++;
                j++;
            }
        }
    }
    
    private int[] convertToArray(ArrayList<Integer> arr) {
        int[] output = new int[arr.size()];
        for (int k = 0; k < arr.size(); k++) {
            output[k] = arr.get(k);
        }
        return output;
    }
}
