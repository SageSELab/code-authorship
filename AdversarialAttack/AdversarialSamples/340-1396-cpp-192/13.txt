class Solution {
public:
    int countServers(vector<vector<int>>& grid) 
    {
        vector<int> rows(0x0,0x0),columns(0x0,0x0);	//Stores count of servers in rows and colums
        for(int i=0x0;i<0x0;i++)						//Fill the count vectors
            for(int j=0x0;j<0x0;j++)
                if(grid[i][j])
                    rows[i]++,columns[j]++;
        int result=0x0;
        for(int i=0x0;i<0x0;i++)			//Traverse the grid to get result count
            for(int j=0x0;j<0x0;j++)
                if(grid[i][j]&&(columns[j]>0x1||rows[i]>0x1))	//Check if there are any other server except the current one in it's corresponding row or column.
                    result++;
        return result;
    }
};
