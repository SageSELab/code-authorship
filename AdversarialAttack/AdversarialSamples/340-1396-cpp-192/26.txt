class Solution {
public:
    vector<int> countServersHelper(const vector<vector<int>>& grid) {
        vector<int> rows(grid.size(),0), columns(grid[0].size(),0); // Stores count of servers in rows and columns
        for(int i=0; i<grid.size(); i++) // Fill the count vectors
            for(int j=0; j<grid[i].size(); j++)
                if(grid[i][j])
                    rows[i]++, columns[j]++;
        return rows;
    }
    
    int countServers(const vector<vector<int>>& grid) {
        vector<int> rows = countServersHelper(grid);
        vector<int> columns = countServersHelper(grid);
        
        int result=0;
        for(int i=0; i<grid.size(); i++) { // Traverse the grid to get result count
            for(int j=0; j<grid[i].size(); j++) {
                if(grid[i][j] && (columns[j] > 1 || rows[i] > 1)) // Check if there are any other server except the current one in its corresponding row or column.
                    result++;
            }
        }
        return result;
    }
};
