public class Solution {
    public int LengthOfLongestSubstring(string s) {

        var l=0;
        var count=0;
        var subStrings = new HashSet<int>();     

        for(int i=0; i < s.Length; i++){

            while(subStrings.Contains((int)s[i])){

                subStrings.Remove((int)s[l]);
                l++;
            }

            subStrings.Add((int)s[i]);
            var subStringCount = subStrings.Count;            
            count = Math.Max(count,subStringCount);
        }

        return count;

    }
}
