public class Solution {
    public int LengthOfLongestSubstring(string s) {
        
        var l=0;
        var count=0;
        var subStrings = new HashSet<char>();     
 
        for(int i=0; i < s.Length; i++){
            
            while(subStrings.Contains(s[i])){
                subStrings.Remove(s[l]);
                l++;
            }

            subStrings.Add(s[i]);
            var subStringCount = GetSubstringCount(subStrings);
            count = Math.Max(count,subStringCount);
        }
        
        return count;

    }

    private int GetSubstringCount(HashSet<char> subStrings) {
        return subStrings.Count;
    }
}
