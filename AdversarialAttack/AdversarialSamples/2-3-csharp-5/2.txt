public class Solution {
    public int LengthOfLongestSubstring(string s) {
        
        var l=0;
        Console.WriteLine("l initialized to: " + l);
        var count=0;
        Console.WriteLine("count initialized to: " + count);
        var subStrings = new HashSet<char>();     
        Console.WriteLine("subStrings initialized with no elements");
 
        for(int i=0; i < s.Length; i++){
            Console.WriteLine("i set to: " + i);
            
            while(subStrings.Contains(s[i])){
                Console.WriteLine("Removing element " + s[l] + " from subStrings");
                
                subStrings.Remove(s[l]);
                l++;
                Console.WriteLine("l incremented to: " + l);
            }

            subStrings.Add(s[i]);
            Console.WriteLine("Adding element " + s[i] + " to subStrings");
            
            var subStringCount = subStrings.Count;      
            Console.WriteLine("subStrings count: " + subStringCount);
                     
            count = Math.Max(count,subStringCount);
            Console.WriteLine("count updated to: " + count);
        }
        
        return count;

    }
}
