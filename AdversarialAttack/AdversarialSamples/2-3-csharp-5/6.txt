public class Solution {
    public int LengthOfLongestSubstring(string s) {
        
        var l = 0;
        var count = 0;
        var subStrings = new HashSet<char>();     
 
        for (int i = 0; i < s.Length; i++){
            
            while (subStrings.Contains(s[i])){

                subStrings.Remove(s[l]);
                l++;
            }

            subStrings.Add(s[i]);
            var subStringCount = subStrings.Count;            
            count = Math.Max(count, subStringCount);
        }
        
        return count;

    }
}
