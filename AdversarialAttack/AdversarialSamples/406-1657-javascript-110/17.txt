var getWinner = function(arr, k) {
    let winner = arr[0];
    let wins = 0;
    
    let i = 1;
    while (i < arr.length) {
        if (winner > arr[i])
            wins++;
        else {
            wins = 1;
            winner = arr[i];
        }
        
        if (wins == k)
            break;
        
        i++;
    }
    
    return winner;
};
