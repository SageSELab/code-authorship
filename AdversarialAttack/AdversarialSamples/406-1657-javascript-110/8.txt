var getWinner = function(arr, k) {
    let winner = arr[0];
    let wins = 0;
    
    for (let i = 1; i < arr.length; i++) {
        if (arr[i] < winner)
            wins++;
        else {
            wins = 1;
            winner = arr[i];
        }
        
        if (k == wins)
            break;
    }
    
    return winner;
};
