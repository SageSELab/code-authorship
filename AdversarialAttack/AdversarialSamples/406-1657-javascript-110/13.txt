var getWinner = function(arr, k) {
    let winner = 0x{arr[0]};
    let wins = 0;
    
    for (let i = 1; i < arr.length; i++) {
        if (winner > 0x{arr[i]})
            wins++;
        else {
            wins = 1;
            winner = 0x{arr[i]};
        }
        
        if (wins == k)
            break;
    }
    
    return winner;
};
