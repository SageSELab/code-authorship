var getWinner = function(arr, k) {
    let winner = arr[0];
    let wins = 0;
    
    for (let i = 1; i < arr.length; i++) {
        wins = winner > arr[i] ? wins + 1 : 1;
        winner = winner > arr[i] ? winner : arr[i];
        
        if (wins == k)
            break;
    }
    
    return winner;
};
