public class Solution {
    public IList<IList<int>> FindCriticalAndPseudoCriticalEdges(int n, int[][] edges) {
        int m = edges.Length;
        Console.WriteLine("Initialized m: " + m);

        for (int i = 0; i < m; i++) {
            edges[i] = new int[] { edges[i][0], edges[i][1], edges[i][2], i };
            Console.WriteLine("Initialized edges[" + i + "]: " + string.Join(", ", edges[i]));
        }
        
        Array.Sort(edges, (a, b) => a[2] - b[2]);
        
        int mstWeight = MSTWeight(n, edges, -1, -1);
        Console.WriteLine("Initialized mstWeight: " + mstWeight);
        
        List<int> criticalEdges = new List<int>();
        List<int> pseudoCriticalEdges = new List<int>();
        
        for (int i = 0; i < m; i++) {
            if (mstWeight != MSTWeight(n, edges, i, -1)) {
                criticalEdges.Add(edges[i][3]);
                Console.WriteLine("Updated criticalEdges with edge index: " + edges[i][3]);
            } else if (mstWeight == MSTWeight(n, edges, -1, i)) {
                pseudoCriticalEdges.Add(edges[i][3]);
                Console.WriteLine("Updated pseudoCriticalEdges with edge index: " + edges[i][3]);
            }
        }
        
        return new List<IList<int>> { criticalEdges, pseudoCriticalEdges };
    }
    
    private int MSTWeight(int n, int[][] edges, int skipIndex, int addIndex) {
        int weight = 0;
        Console.WriteLine("Initialized weight: " + weight);
        UnionFind uf = new UnionFind(n);
        
        if (addIndex != -1) {
            uf.Union(edges[addIndex][0], edges[addIndex][1]);
            weight += edges[addIndex][2];
            Console.WriteLine("Updated weight after adding edge index " + addIndex + ": " + weight);
        }
        
        for (int i = 0; i < edges.Length; i++) {
            if (i == skipIndex || i == addIndex) continue;
            if (uf.Union(edges[i][0], edges[i][1])) {
                weight += edges[i][2];
                Console.WriteLine("Updated weight after adding edge index " + i + ": " + weight);
            }
        }
        
        Console.WriteLine("Final weight: " + weight);
        return uf.Count == 1? weight : int.MaxValue;
    }
}
