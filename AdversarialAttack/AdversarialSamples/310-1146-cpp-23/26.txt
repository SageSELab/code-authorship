class Solution {
public:
    string gcdOfStrings(string str1, string str2) {
        
        if (!hasCommonDivisor(str1, str2)) {
            return "";
        }
        
        int gcd_length = findGCDLength(str1, str2);
        
        return str1.substr(0, gcd_length);
    }
    
private:
    bool hasCommonDivisor(string str1, string str2) {
        return str1 + str2 == str2 + str1;
    }
    
    int findGCDLength(string str1, string str2) {
        return __gcd(str1.size(), str2.size());
    }
};
