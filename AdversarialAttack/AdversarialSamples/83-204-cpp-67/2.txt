class Solution {
public:
    int countPrimes(int n) 
    {
        int count=0;
        cout << "initialized count with value: " << count << endl;
        bool* prime=new bool[n];
        for(int i=2;i<n;i++)
        {
            prime[i]=true;
            cout << "initialized prime[" << i << "] with value: " << prime[i] << endl;
        }
        for(int i=2;i*i<n;i++)
        {
            cout << "checking if not prime[" << i << "] to continue" << endl;
            if(!prime[i])
                continue;
            for(int j=i*i;j<n;j+=i)
            {
                prime[j]=false;
                cout << "modified prime[" << j << "] to false" << endl;
            }
        }
        for(int i=2;i<n;i++)
        {
            if(prime[i])
            {
                count++;
                cout << "incremented count to: " << count << endl;
            }
        }
        return count;
    }
};
