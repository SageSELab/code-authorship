class Solution {
public:
    int countPrimes(int n) 
    {
        int count=0;
        bool* primeArr=new bool[n];
        for(int i=2;i<n;i++)
        {
            primeArr[i]=true;
        }
        for(int i=2;i*i<n;i++)
        {
            if(!primeArr[i])
            continue;
            for(int j=i*i;j<n;j+=i)
            {
                primeArr[j]=false;
            }
        }
        for(int i=2;i<n;i++)
        {
            if(primeArr[i])
            count++;
        }
        return count;
    }
};
