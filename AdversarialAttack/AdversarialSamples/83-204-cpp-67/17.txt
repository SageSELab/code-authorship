class Solution {
public:
    int countPrimes(int n) 
    {
        int count=0;
        bool* prime=new bool[n];
        int i=2;
        while(i<n)
        {
            prime[i]=true;
            i++;
        }
        i=2;
        while(i*i<n)
        {
            if(!prime[i])
                continue;
            int j=i*i;
            while(j<n)
            {
                prime[j]=false;    
                j+=i;
            }
            i++;
        }
        i=2;
        while(i<n)
        {
            if(prime[i])
                count++;
            i++;
        }
        return count;
    }
};
