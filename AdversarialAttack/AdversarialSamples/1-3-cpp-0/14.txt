class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int n = s.length();
        int maxLength = 0;
        unordered_map<int, int> charMap;
        int left = 0;
        
        for (int right = 0; right < n; right++) {
            if (charMap.count(ascii_value_of_character(s[right])) == 0 || charMap[ascii_value_of_character(s[right])] < left) {
                charMap[ascii_value_of_character(s[right])] = right;
                maxLength = max(maxLength, right - left + 1);
            } else {
                left = charMap[ascii_value_of_character(s[right])] + 1;
                charMap[ascii_value_of_character(s[right])] = right;
            }
        }
        
        return maxLength;
    }
};
