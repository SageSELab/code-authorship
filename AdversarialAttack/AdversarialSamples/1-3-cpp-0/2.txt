class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int n = s.length();
        cout << "n initialized with value: " << n << endl;
        
        int maxLength = 0;
        cout << "maxLength initialized with value: " << maxLength << endl;
        
        unordered_map<char, int> charMap;
        cout << "charMap initialized" << endl;
        
        int left = 0;
        cout << "left initialized with value: " << left << endl;
        
        for (int right = 0; right < n; right++) {
            cout << "right initialized with value: " << right << endl;
            
            if (charMap.count(s[right]) == 0 || charMap[s[right]] < left) {
                charMap[s[right]] = right;
                cout << "charMap[" << s[right] << "] updated with value: " << right << endl;
                
                maxLength = max(maxLength, right - left + 1);
                cout << "maxLength updated with value: " << maxLength << endl;
            } else {
                left = charMap[s[right]] + 1;
                cout << "left updated with value: " << left << endl;
                
                charMap[s[right]] = right;
                cout << "charMap[" << s[right] << "] updated with value: " << right << endl;
            }
        }
        
        return maxLength;
    }
};
