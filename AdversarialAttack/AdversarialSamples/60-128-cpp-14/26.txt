class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        int n = nums.size();
        
        // Base case: If the input vector is empty, there are no consecutive elements.
        if (n == 0) {
            return 0;
        }

        // Sort the input vector in ascending order.
        sort(nums.begin(), nums.end());

        // Function to calculate the maximum consecutive sequence length.
        auto calculateConsecutiveLength = [](vector<int>& nums) {
            int cnt = 1;
            int maxi = 0;
            
            for (int i = 1; i < nums.size(); i++) {
                if (nums[i] != nums[i - 1]) {
                    if (nums[i] == nums[i - 1] + 1) {
                        cnt++;
                    } else {
                        maxi = max(maxi, cnt);
                        cnt = 1;
                    }
                }
            }
            
            return max(maxi, cnt);
        };

        // Calculate and return the maximum consecutive sequence length.
        return calculateConsecutiveLength(nums);
    }
};
