// OJ: https://leetcode.com/problems/construct-the-lexicographically-largest-valid-sequence/
// Author: github.com/lzl124631x
// Time: O(N!)
// Space: O(N)
class Solution {
    vector<int> used;
    bool dfs(vector<int> &ans, int i) {
        if (i == ans.size()) {
            cout << "Filled all the numbers, found the answer\n";
            return true; // filled all the numbers, found the answer
        }
        if (ans[i]) {
            cout << "This index is already filled, continue to fill the next index.\n";
            return dfs(ans, i + 1); // this index is already filled, continue to fill the next index.
        }
        for (int j = used.size() - 1; j > 0; --j) {
            cout << "Trying number: " << j << endl; // try each number in decending order from n to 1.
            if (used[j]) {
                cout << j << " is already used, skip\n"; // j is already used, skip
                continue;
            }
            if (j != 1 && (i + j >= ans.size() || ans[i + j])) {
                cout << "Cannot fill ans[" << i + j << "] with " << j << ". Skip.\n"; // we can't fill `ans[i + j]` either because `i + j` is out of bound or `ans[i + j]` is already filled. Skip.
                continue;
            }
            used[j] = 1; // mark number `j` as used.
            cout << "Marked " << j << " as used\n";
            ans[i] = j; // fill `ans[j]` and `ans[i + j]` (if needed) with `j`.
            cout << "Filled ans[" << i << "] with " << j << endl;
            if (j != 1) {
                ans[i + j] = j;
                cout << "Filled ans[" << i + j << "] with " << j << endl;
            }
            if (dfs(ans, i + 1)) return true;
            ans[i] = 0; // this filling is invalid, backtrack and try the next number.
            cout << "Invalid filling, backtracking\n";
            if (j != 1) {
                ans[i + j] = 0;
                cout << "Reset ans[" << i + j << "]\n";
            }
            used[j] = 0;
            cout << "Reset used[" << j << "]\n";
        }
        return false;
    }
public:
    vector<int> constructDistancedSequence(int n) {
        vector<int> ans(2 * n - 1);
        used.assign(n + 1, 0); // numbers 1 ~ n are unused initially
        cout << "Filled used with 0s\n";
        dfs(ans, 0); // try filling numbers from index 0.
        return ans;
    }
};
