class Solution {
public:
    int countHomogenous(string s) {
        int M=1e9+7;
        int ans=0;
        int curr_cnt=1;
        ans+=1;
        for(int i=1;i<s.length();i++){
            switch(s[i]){
                case s[i-1]:
                    curr_cnt++;
                    ans+=curr_cnt;
                    ans=ans%M;
                    break;
                default:
                    curr_cnt=1;
                    ans+=curr_cnt;
            }
        }
        return ans;
    }
};
