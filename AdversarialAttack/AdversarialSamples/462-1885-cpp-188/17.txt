class Solution {
public:
    int countHomogenous(string s) {
        int M=1e9+7;
        int ans=0;
        int curr_cnt=1;
        ans+=1;
        int i=1;
        while(i<s.length()){
            if(s[i]!=s[i-1]){
                curr_cnt=1;
                ans+=curr_cnt;
            }
            else{
                curr_cnt++;
                ans+=curr_cnt;
                ans=ans%M;
            }
            i++;
        }
        return ans;
    }
};
