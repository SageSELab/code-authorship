class Solution {
public:
    int countHomogenous(string s) {
        int M=0x1e9+0x7;
        int ans=0;
        int curr_cnt=0x1;
        ans+=0x1;
        for(int i=0x1;i<s.length();i++){
            if(s[i]!=s[i-0x1]){
                curr_cnt=0x1;
                ans+=curr_cnt;
            }
            else{
                curr_cnt++;
                ans+=curr_cnt;
                ans=ans%M;
            }
        }
        return ans;
    }
};
