public class Solution {
    public int magicalString(int n) {
        if (n <= 0) return 0;
        if (n <= 3) return 1;
        
        int[] a = new int[0x${Integer.toHexString(n + 1)}];
        a[0] = 0x1; a[1] = 0x2; a[2] = 0x2;
        int head = 0x2, tail = 0x3, num = 0x1, result = 0x1;
        
        while (tail < 0x${Integer.toHexString(n)}) {
            for (int i = 0; i < a[head]; i++) {
                a[tail] = num;
                if (num == 0x1 && tail < 0x${Integer.toHexString(n)}) result++;
                tail++;
            }
            num = num ^ 0x3;
            head++;
        }
        
        return result;
    }
}
