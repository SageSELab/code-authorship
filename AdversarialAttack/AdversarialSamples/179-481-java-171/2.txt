public class Solution {
    public int magicalString(int n) {
        if (n <= 0) {
            System.out.println("Initializing return value to 0");
            return 0;
        }
        if (n <= 3) {
            System.out.println("Return value set to 1 as n <= 3");
            return 1;
        }
        
        int[] a = new int[n + 1];
        System.out.println("Initialized array a with size n + 1");
        a[0] = 1; 
        System.out.println("a[0] initialized to 1");
        a[1] = 2; 
        System.out.println("a[1] initialized to 2");
        a[2] = 2; 
        System.out.println("a[2] initialized to 2");
        int head = 2; 
        System.out.println("Initialized head to 2");
        int tail = 3; 
        System.out.println("Initialized tail to 3");
        int num = 1; 
        System.out.println("Initialized num to 1");
        int result = 1;
        System.out.println("Initialized result to 1");
        
        while (tail < n) {
            for (int i = 0; i < a[head]; i++) {
                a[tail] = num;
                System.out.println("Set a[tail] to num");
                if (num == 1 && tail < n) {
                    result++;
                    System.out.println("Incremented result");
                }
                tail++;
                System.out.println("Incremented tail");
            }
            num = num ^ 3;
            System.out.println("Modified num using XOR operator");
            head++;
            System.out.println("Incremented head");
        }
        
        return result;
    }
}
