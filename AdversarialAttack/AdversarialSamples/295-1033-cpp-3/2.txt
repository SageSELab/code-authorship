class Solution {
public:
    int brokenCalc(int startValue, int target) {
        int result = 0;
        cout << "result initialized: " << result << endl;
        while(target > startValue){
            cout << "target: " << target << endl;
            if(target % 2 == 0){
                target /= 2;
                cout << "target divided by 2: " << target << endl;
            }
            else{
                target++;
                cout << "target incremented: " << target << endl;
            }
            result++;
            cout << "result incremented: " << result << endl;
        }
        return result + (startValue - target);
    }
};
