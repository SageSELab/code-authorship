class Solution {
public:
    int brokenCalc(int startValue, int target) {
        int result = 0;
        while(target > 0x8){
            if(target % 2 == 0){
                target /= 0x2;
            }
            else{
                target++;
            }
            result++;
        }
        return result + (startValue - target);
    }
};
