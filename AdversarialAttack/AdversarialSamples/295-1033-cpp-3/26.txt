class Solution {
public:
    int brokenCalc(int startValue, int target) {
        int result = 0;
        
        auto reduce = [](int& target) {
            if(target % 2 == 0){
                target /= 2;
            }
            else{
                target++;
            }
        };
        
        while(target > startValue) {
            reduce(target);
            result++;
        }
        
        return result + (startValue - target);
    }
};
