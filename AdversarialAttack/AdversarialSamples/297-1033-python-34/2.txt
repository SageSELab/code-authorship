class Solution {
    public int brokenCalc(int X, int Y) {
        int ans = 0;
        System.out.println("ans initialized: " + ans);
        while (X < Y) {
            System.out.println("X value: " + X);
            System.out.println("Y value: " + Y);
            ans++;
            System.out.println("ans value modified: " + ans);
            if (Y % 2 > 0) {
                Y++;
                System.out.println("Y value modified: " + Y);
            } else {
                Y /= 2;
                System.out.println("Y value modified: " + Y);
            }
        }
        int result = X - Y + ans;
        System.out.println("Final result: " + result);
        return result;
    }
}
