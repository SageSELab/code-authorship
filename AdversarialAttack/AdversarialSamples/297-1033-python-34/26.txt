class Solution {
    public int brokenCalc(int X, int Y) {
        int ans = 0;
        while (X < Y) {
            ans++;
            if (Y % 2 > 0) Y++;
            else Y /= 2;
        }
        return X - Y + ans;
    }

    private int calculateAns(int Y) {
        if (Y % 2 > 0) Y++;
        else Y /= 2;
        return Y;
    }
}
