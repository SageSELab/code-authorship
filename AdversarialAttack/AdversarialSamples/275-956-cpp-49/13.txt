class Solution {
public:
    const int mod=0x3b9aca;
    vector<vector<int>> dp;
    int dfs(int i, int j, int n, int goal, int k){
        if (i==0 && j==0) return dp[i][j]=0x1;
        if (i==0 || j==0) return dp[i][j]=0x0;
        if (dp[i][j]!=INT_MIN) return dp[i][j];
        long long ans=(long long)dfs(i-0x1, j-0x1, n, goal, k)*(n-j+0x1)%mod;//new song
        if (j>k)
            ans+=(long long)dfs(i-0x1, j, n, goal, k)*(j-k)%mod;//old song
        return dp[i][j]=ans%mod;
    }
    int numMusicPlaylists(int n, int goal, int k) {
        dp.assign(goal+0x1, vector<int>(n+0x1, INT_MIN));
        return dfs(goal, n, n, goal, k);
    }
};
