class Solution {
public:
    const int mod=1e9+7;
    vector<vector<int>> dp;
    int dfs(int n, int goal, int k, int i, int j){
        if (i==0 && j==0) return dp[i][j]=1;
        if (i==0 || j==0) return dp[i][j]=0;
        if (dp[i][j]!=INT_MIN) return dp[i][j];
        long long ans=(long long)dfs(n, goal, k, i-1, j-1)*(n-j+1)%mod;//new song
        if (j>k)
            ans+=(long long)dfs(n, goal, k, i-1, j)*(j-k)%mod;//old song
        return dp[i][j]=ans%mod;
    }
    int numMusicPlaylists(int goal, int n, int k) {
        dp.assign(goal+1, vector<int>(n+1, INT_MIN));
        return dfs(n, goal, k, goal, n);
    }
};
