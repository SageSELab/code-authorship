class Solution {
    public int[] getStrongest(int[] arr, int k) {
        Arrays.sort(arr);
        int m = arr[(arr.length - 1) / 2];
        PriorityQueue<Integer> pq = new PriorityQueue<>(
                (a, b) -> Math.abs(a - m) == Math.abs(b - m) ? b - a : Math.abs(b - m) - Math.abs(a - m)
        );

        int i = 0;
        while (i < arr.length) {
            pq.offer(arr[i]);
            i++;
        }

        int[] ans = new int[k];
        int j = 0;
        while (j < k) {
            ans[j] = pq.poll();
            j++;
        }

        return ans;
    }
}

// TC: O(n * logn) + O(n) => O(n * logn)
// SC: O(n + k)
