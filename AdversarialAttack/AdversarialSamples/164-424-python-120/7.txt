# Space Complexity : O(1)
class Solution(object):

        return self.character_replacement(s,k)
    def character_replacement(self, s, k):
        maxlen, largest_count = 0, 0
        arr = collections.Counter()
        for idx in xrange(len(s)):
            arr[s[idx]] += 1
            largest_count = max(largest_count, arr[s[idx]])
            if maxlen - largest_count >= k:
                arr[s[idx - maxlen]] -= 1
            else:
                maxlen += 1
        return maxlen
