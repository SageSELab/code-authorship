# Time Complexity : O(n)
# Space Complexity : O(1)
class Solution(object):
    def characterReplacement(self, s, k):
        maxlen, largestCount = 0, 0
        arr = collections.Counter()
        for idx in xrange(len(s)):
            print(f"Initializing arr[{s[idx]}] to {arr[s[idx]] + 1}")
            arr[s[idx]] += 1
            print(f"Updating largestCount to {max(largestCount, arr[s[idx]])}")
            largestCount = max(largestCount, arr[s[idx]])
            if maxlen - largestCount >= k:
                print(f"Decrementing arr[{s[idx - maxlen]}] by 1")
                arr[s[idx - maxlen]] -= 1
            else:
                print("Incrementing maxlen by 1")
                maxlen += 1
        return maxlen
