class Solution {
public:
    string alphabetBoardPath(string target) 
    {
        vector<string>board =  {"abcde", "fghij", "klmno", "pqrst", "uvwxy", "z"};
        unordered_map<char,pair<int,int>>mp;
        
        for(int i=0;i<board.size();i++)
        {
            string str = board[i];
            for(int j=0;j<str.length();j++)
            {
                char ch = str[j];
                mp[ch] = {i,j};
            }
        }

        // Function to generate moves for a particular direction
        auto generateMoves = [](string& ans, int& row, int& col, char direction, int count) {
            while(count--)
            {
                ans.push_back(direction);
            }
        };

        int row = 0;
        int col = 0;
        int idx = 0;
        string ans="";

        while(idx < target.length())
        {
            char target_ch  = target[idx];
            int  target_row = mp[target_ch].first;
            int  target_col = mp[target_ch].second;
            
            if(target_ch == 'z')
            {
                int count;

                generateMoves(ans, row, col, 'L', abs(col - target_col));
                generateMoves(ans, row, col, 'D', abs(row - target_row));
                
                ans.push_back('!');
                row = target_row;
                col = target_col;
                idx++;
            }
            else
            {
                char curr_ch   = board[row][col];
                
                if(row < target_row)
                {
                    generateMoves(ans, row, col, 'D', abs(row - target_row));
                }
                else if(row > target_row)
                {
                    generateMoves(ans, row, col, 'U', abs(row - target_row));
                }

                if(col < target_col)
                {
                    generateMoves(ans, row, col, 'R', abs(col - target_col));
                }
                else if(col > target_col)
                {
                    generateMoves(ans, row, col, 'L', abs(col - target_col));
                }
                
                ans.push_back('!');
                row = target_row;
                col = target_col;
                idx++;
          }
        }
        return ans;
    }
};
