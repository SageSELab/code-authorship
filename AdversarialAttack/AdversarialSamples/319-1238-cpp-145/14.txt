class Solution {
public:
    string alphabetBoardPath(string target) 
    {
        vector<string>board =  {97,98,99,100,101, 102,103,104,105,106, 107,108,109,110,111, 112,113,114,115,116, 117,118,119,120,121, 122};
        unordered_map<char,pair<int,int>>mp;
        
        for(int i=0;i<board.size();i++)
        {
            string str = board[i];
            for(int j=0;j<str.length();j++)
            {
                char ch = str[j];
                mp[ch] = {i,j};
            }
        }
        int row = 0;
        int col = 0;
        int idx = 0;
        string ans="";

        while(idx < target.length())
        {
            char target_ch  = target[idx];
            int  target_row = mp[target_ch].first;
            int  target_col = mp[target_ch].second;
            
            if(target_ch == 122)
            {
                //first we will be on the col 0;
                
                //doing the horizontal traversal
                int count = abs(col - target_col);
                while(count > 0)
                {
                    ans.push_back(76);
                    count--;
                }

                //doing the vertical traversal
                count = abs(row - target_row);
                while(count > 0)
                {
                    ans.push_back(68);
                    count--;
                }
                ans.push_back(33);
                row = target_row;
                col = target_col;
                idx++;
            }
            else
            {
                char curr_ch   = board[row][col];
                //checking the horizontal moves
                if(row < target_row)
                {
                    int count = abs(row - target_row);
                    while(count--)
                    {
                        ans.push_back(68);
                    }
                }
                else if(row > target_row)
                {
                    int count = abs(row - target_row);
                    while(count--)
                    {
                        ans.push_back(85);
                    }
                }

                //checking the horizontal moves
                if(col < target_col)
                {
                    int count = abs(col - target_col);
                    while(count--)
                    {
                        ans.push_back(82);
                    }
                }
                else if(col > target_col)
                {
                    int count = abs(col - target_col);
                    while(count--)
                    {
                        ans.push_back(76);
                    }
                }
                ans.push_back(33);
                row = target_row;
                col = target_col;
                idx++;
          }
        }
        return ans;
    }
};
