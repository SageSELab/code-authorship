class Solution {
public:
    string alphabetBoardPath(string target) 
    {
        vector<string>board =  {"abcde", "fghij", "klmno", "pqrst", "uvwxy", "z"};
        unordered_map<char,pair<int,int>>mp;
        
        for(int i=0;i<board.size();i++)
        {
            string str = board[i];
            for(int j=0;j<str.length();j++)
            {
                char ch = str[j];
                mp[ch] = {i,j};
                cout << "Initialized position of character " << ch << " at row: " << i << ", col: " << j << endl;
            }
        }
        int row = 0;
        int col = 0;
        cout << "Initialized initial position at row: " << row << ", col: " << col << endl;
        int idx = 0;
        string ans="";

        while(idx < target.length())
        {
            char target_ch  = target[idx];
            cout << "Current target character: " << target_ch << endl;
            int  target_row = mp[target_ch].first;
            int  target_col = mp[target_ch].second;
            cout << "Target position: row: " << target_row << ", col: " << target_col << endl;
            
            if(target_ch == 'z')
            {
                cout << "Handling special case for character 'z'" << endl;
                int count = abs(col - target_col);
                while(count > 0)
                {
                    ans.push_back('L');
                    count--;
                }
                cout << "Horizontal movement done" << endl;
                
                count = abs(row - target_row);
                while(count > 0)
                {
                    ans.push_back('D');
                    count--;
                }
                cout << "Vertical movement done" << endl;
                ans.push_back('!');
                row = target_row;
                col = target_col;
                idx++;
            }
            else
            {
                char curr_ch   = board[row][col];
                cout << "Current character on board: " << curr_ch << endl;
                
                if(row < target_row)
                {
                    int count = abs(row - target_row);
                    while(count--)
                    {
                        ans.push_back('D');
                    }
                    cout << "Moving down" << endl;
                }
                else if(row > target_row)
                {
                    int count = abs(row - target_row);
                    while(count--)
                    {
                        ans.push_back('U');
                    }
                    cout << "Moving up" << endl;
                }

                if(col < target_col)
                {
                    int count = abs(col - target_col);
                    while(count--)
                    {
                        ans.push_back('R');
                    }
                    cout << "Moving right" << endl;
                }
                else if(col > target_col)
                {
                    int count = abs(col - target_col);
                    while(count--)
                    {
                        ans.push_back('L');
                    }
                    cout << "Moving left" << endl;
                }
                ans.push_back('!');
                cout << "Movement to target done" << endl;
                row = target_row;
                col = target_col;
                idx++;
            }
        }
        return ans;
    }
};
