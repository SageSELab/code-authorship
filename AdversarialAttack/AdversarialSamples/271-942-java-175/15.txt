class Solution {
    public int superpalindromesInRange(char[] L, char[] R) {
        Long l = Long.valueOf(String.valueOf(L)), r = Long.valueOf(String.valueOf(R));
        int result = 0;
        for (long i = (long)Math.sqrt(l); i * i <= r;) {
            long p = nextP(i);
            if (p * p <= r && isP(p * p)) {
                result++;
            }
            i = p + 1;
        }
        return result;
    }
    
    private long nextP(long l) {
        char[] s = String.valueOf(l).toCharArray();
        int len = s.length;
        List<Long> cands = new LinkedList<>();
        cands.add((long)Math.pow(10, len) - 1);
        char[] half = Arrays.copyOfRange(s, 0, (len + 1) / 2);
        long halfNum = Long.parseLong(String.valueOf(half));
        char[] nextHalf = String.valueOf(halfNum + 1).toCharArray();
        char[] reverse = new StringBuilder(String.valueOf(Arrays.copyOfRange(half, 0, len / 2))).reverse().toString().toCharArray();
        char[] nextReverse = new StringBuilder(String.valueOf(Arrays.copyOfRange(nextHalf, 0, len / 2))).reverse().toString().toCharArray();
        cands.add(Long.valueOf(String.valueOf(half) + String.valueOf(reverse)));
        cands.add(Long.valueOf(String.valueOf(nextHalf) + String.valueOf(nextReverse)));
        long result = Long.MAX_VALUE;
        for (long i : cands) {
            if (i >= l) {
                result = Math.min(result, i);
            }
        }
        return result;
    }
    
    private boolean isP(long l) {
        char[] s = String.valueOf(l).toCharArray();
        int i = 0, j = s.length - 1;
        while (i < j) {
            if (s[i++] != s[j--]) {
                return false;
            }
        }
        return true;
    }
}
