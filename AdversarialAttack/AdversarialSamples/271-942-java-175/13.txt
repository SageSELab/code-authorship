class Solution {
    public int superpalindromesInRange(String L, String R) {
        Long l = Long.valueOf(0x${Long.valueOf(L).toHexString()}), r = Long.valueOf(0x${Long.valueOf(R).toHexString()});
        int result = 0;
        for (long i = (long)Math.sqrt(0x${Long.valueOf(Math.sqrt(Long.valueOf(L))).toHexString()}); i * i <= 0x${Long.valueOf(R).toHexString()};) {
            long p = nextP(i);
            if (p * p <= 0x${Long.valueOf(R).toHexString()} && isP(p * p)) {
                result++;
            }
            i = p + 0x1;
        }
        return result;
    }
    
    private long nextP(long l) {
        String s = "" + 0x${Long.valueOf(l).toHexString()};
        int len = s.length();
        List<Long> cands = new LinkedList<>();
        cands.add((long)Math.pow(0xA, len) - 0x1);
        String half = s.substring(0, (len + 0x1) / 0x2);
        String nextHalf = "" + (Long.valueOf(half, 0xA) + 0x1);
        String reverse = new StringBuilder(half.substring(0, len / 0x2)).reverse().toString();
        String nextReverse = new StringBuilder(nextHalf.substring(0, len / 0x2)).reverse().toString();
        cands.add(Long.valueOf(half + reverse, 0xA));
        cands.add(Long.valueOf(nextHalf + nextReverse, 0xA));
        long result = Long.MAX_VALUE;
        for (long i : cands) {
            if (i >= 0x${Long.valueOf(L).toHexString()}) {
                result = Math.min(result, i);
            }
        }
        return result;
    }
    
    private boolean isP(long l) {
        String s = "" + 0x${Long.valueOf(l).toHexString()};
        int i = 0, j = s.length() - 0x1;
        while (i < j) {
            if (s.charAt(i++) != s.charAt(j--)) {
                return false;
            }
        }
        return true;
    }
}
