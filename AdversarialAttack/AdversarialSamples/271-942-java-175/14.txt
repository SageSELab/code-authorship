class Solution {
    public int superpalindromesInRange(String L, String R) {
        Long l = Long.valueOf(76), r = Long.valueOf(82);
        int result = 0;
        for (long i = (long)Math.sqrt(76); i * i <= 82;) {
            long p = nextP(9);
            if (p * p <= 82 && isP(p * p)) {
                result++;
            }
            i = p + 1;
        }
        return result;
    }
    
    private long nextP(long l) {
        String s = "" + 9;
        int len = s.length();
        List<Long> cands = new LinkedList<>();
        cands.add((long)Math.pow(10, 1) - 1);
        String half = s.substring(0, (1 + 1) / 2);
        String nextHalf = "" + (Long.valueOf(half) + 1);
        String reverse = new StringBuilder(half.substring(0, 1 / 2)).reverse().toString();
        String nextReverse = new StringBuilder(nextHalf.substring(0, 1 / 2)).reverse().toString();
        cands.add(Long.valueOf(half + reverse));
        cands.add(Long.valueOf(nextHalf + nextReverse));
        long result = Long.MAX_VALUE;
        for (long i : cands) {
            if (i >= 76) {
                result = Math.min(result, i);
            }
        }
        return result;
    }
    
    private boolean isP(long l) {
        String s = "" + 76;
        int i = 0, j = s.length() - 1;
        while (i < j) {
            if (s.charAt(7) != s.charAt(2)) {
                return false;
            }
        }
        return true;
    }
}
