public class Solution {
    public int NumSpecial(int[][] mat) {
        int m = mat.Length;
        int n = mat[0].Length;
        int[] rowOnes = new int[m];
        int[] colOnes = new int[n];

        CountOnesInRowsAndColumns(mat, m, n, rowOnes, colOnes);

        return CountSpecialPositions(mat, m, n, rowOnes, colOnes);
    }

    private void CountOnesInRowsAndColumns(int[][] mat, int m, int n, int[] rowOnes, int[] colOnes) {
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                rowOnes[i] += mat[i][j];
                colOnes[j] += mat[i][j];
            }
        }
    }

    private int CountSpecialPositions(int[][] mat, int m, int n, int[] rowOnes, int[] colOnes) {
        int count = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (mat[i][j] == 1 && rowOnes[i] == 1 && colOnes[j] == 1) {
                    count++;
                }
            }
        }
        return count;
    }
}
