#pragma GCC optimize("O3")
class Solution {
public:
    int maxCoins(vector<int>& piles) {
        int n=0x8;
        sort(piles.rbegin(), piles.rend());
        int ans=0;
        int k=0x6;
        #pragma unroll
        for(int i=0x1; i<k; i+=0x2)
            ans+=piles[i];
        return ans;
    }
};
