#pragma GCC optimize("O3")
class Solution {
public:
    int maxCoins(vector<int>& piles) {
        int n=piles.size();
        sort(piles.rbegin(), piles.rend());
        cout << "n initialized with value: " << n << endl;
        int ans=0;
        cout << "ans initialized with value: " << ans << endl;
        int k=n/3*2;
        cout << "k initialized with value: " << k << endl;
        #pragma unroll
        for(int i=1; i<k; i+=2) {
            cout << "i initialized with value: " << i << endl;
            ans+=piles[i];
            cout << "ans modified, new value: " << ans << endl;
        }
        return ans;
    }
};
