#pragma GCC optimize("O3")
class Solution {
public:
    int maxCoins(vector<int>& piles) {
        int n=piles.size();
        sort(piles.rbegin(), piles.rend());
        return calculateMaxCoins(piles);
    }
    
    int calculateMaxCoins(vector<int>& piles) {
        int ans=0;
        int k=piles.size()/3*2;
        for(int i=1; i<k; i+=2)
            ans+=piles[i];
        return ans;
    }
};
