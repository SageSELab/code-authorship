class Solution {
public:
    vector<vector<vector<int>>>dp;
    int findMaxForm(vector<string>& arr, int m, int n) 
   {
    dp.resize(arr.size()+1,vector<vector<int>>(m+1,vector<int>(n+1,-1))); 
    cout << "Initialized dp vector with dimensions: " << arr.size()+1 << " " << m+1 << " " << n+1 << endl;
    vector<pair<int,int>>v;  //pair<one,zero>
    
	for(int i=0;i<arr.size();i++)
	{
		string str=arr[i];
		int one=0;
		int zero=0;
		cout << "Initialized one and zero variables: " << one << " " << zero << endl;
		
		for(int j=0;j<str.length();j++)
		{
			if(str[j]=='0')
			{
				zero++;
				cout << "Incremented zero to: " << zero << endl;
			}
			else if(str[j]=='1')
			{
				one++;
				cout << "Incremented one to: " << one << endl;
			}
		}
	    v.push_back(make_pair(zero,one));
	    cout << "Pushed pair into vector: " << zero << " " << one << endl;
	}
	int idx=0;
    cout << "Initialized idx variable: " << idx << endl;
	return fun(arr,v,idx,m,n);        
}
int fun(vector<string>&arr,vector<pair<int,int>>&v,int idx,int m,int n)
{
	//base 
	if(idx>=arr.size())
	{
		return 0;
	}
    cout << "Checking if dp value is already calculated at index: " << idx << " " << m << " " << n << endl;
	if(dp[idx][m][n]!=-1)
    {
        // cout<<"hello"<<endl;
        cout << "Found precalculated value in dp at index: " << idx << " " << m << " " << n << endl;
        return dp[idx][m][n];
    }
	int choise1=0;
    int choise2=0;
    cout << "Initialized choise1 and choise2 variables: " << choise1 << " " << choise2 << endl;
    
   //if i can take that element //considering arr[idx]
    if(m - v[idx].first >=0 and n-v[idx].second >=0)
    {
    	choise1=1+fun(arr,v,idx+1,m-v[idx].first,n-v[idx].second);
        cout << "Calculated choise1 with value: " << choise1 << endl;
	}
   
    //or i cannot take that element
    choise2=0+fun(arr,v,idx+1,m,n);
    cout << "Calculated choise2 with value: " << choise2 << endl;
   
    return dp[idx][m][n] = max(choise1,choise2);
}
};
