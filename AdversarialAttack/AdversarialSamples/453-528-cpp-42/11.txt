class Solution {
public:
    ListNode* swapNodes(ListNode* head, int k) {
        ListNode*st;
        int c=0;
        st=head;

        ListNode* v1,*v2;

        while(st != nullptr) {
            c += 1;
            if(c == k) {
                v1 = st;
            }
            st = st->next;
        }
        cout << c << endl;

        c = c - k;
        st = head;

        while(st && c) {
            st = st->next;
            c -= 1;
        }

        v2 = st;
        
        int t = v1->val;
        v1->val = v2->val;
        v2->val = t;

        return head;
        
    }
};
