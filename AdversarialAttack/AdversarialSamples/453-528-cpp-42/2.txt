/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {
 *         cout << "Node initialized with val: " << val << endl;
 *     }
 *     ListNode(int x) : val(x), next(nullptr) {
 *         cout << "Node initialized with val: " << val << endl;
 *     }
 *     ListNode(int x, ListNode *next) : val(x), next(next) {
 *         cout << "Node initialized with val: " << val << endl;
 *     }
 * };
 */
class Solution {
public:
    ListNode* swapNodes(ListNode* head, int k) {
        ListNode*st;
        cout << "Variable st initialized" << endl;
        int c=0;
        cout << "Variable c initialized" << endl;
        st=head;

        ListNode* v1,*v2;

        while(st!=NULL){
            c++;
            cout << "Incremented c to: " << c << endl;
            if(c==k){
                v1=st;
                cout << "Assigning value to v1" << endl;
            }
            st=st->next;
        }
        cout<<c<<endl;

        c=c-k;
        cout << "Updated c to: " << c << endl;
        st=head;

        while(st and c){
            st=st->next;
            c--;

        }

        v2=st;
        cout << "Assigning value to v2" << endl;
        
        int t=v1->val;
        cout << "Variable t initialized to: " << t << endl;
        v1->val=v2->val;
        cout << "Swapped values for v1" << endl;
        v2->val=t;
        cout << "Swapped values for v2" << endl;

        return head;
        
    }
};
