class Solution {
public:
    ListNode* swapNodes(ListNode* head, int k) {
        ListNode* st;
        int c = 0;
        st = head;

        ListNode* v1, *v2;

        while (st != NULL) {
            c++;
            if (c == k) {
                v1 = st;
            }
            st = st->next;
        }

        c = c - k;
        st = head;

        while (st and c) {
            st = st->next;
            c--;
        }

        v2 = st;

        int t = v1->val;
        v1->val = v2->val;
        v2->val = t;

        cout << c << endl;

        return head;
        
    }
};
