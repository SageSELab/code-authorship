/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* swapNodes(ListNode* head, int k) {
        ListNode* v1 = getNodeAtIndex(head, k);
        ListNode* v2 = getNodeAtIndex(head, getSize(head) - k + 1);

        int t = v1->val;
        v1->val = v2->val;
        v2->val = t;

        return head;
    }

private:
    ListNode* getNodeAtIndex(ListNode* head, int index) {
        ListNode* st = head;
        int c = 0;
        
        while (st) {
            c++;
            if (c == index) {
                return st;
            }
            st = st->next;
        }
        
        return nullptr;
    }

    int getSize(ListNode* head) {
        int size = 0;
        ListNode* st = head;
        while (st) {
            size++;
            st = st->next;
        }
        return size;
    }
};
