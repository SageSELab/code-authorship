class Solution {
public:
    int solve(int k, vector<int>& nums)
    {
        int i=0,res=0;
        unordered_map<int,int> m;
        for(int j=0;j<nums.size();j++)
        {
            if(!m[nums[j]]++)
                k--;
            while(k<0)
            {
                if(!--m[nums[i]])
                    k++;
                i++;
            }
            res+=j-i+1;
        }
        return res;
        
    }
    int subarraysWithKDistinct(int k, vector<int>& nums)
    {
        return solve(k,nums)-solve(k-1,nums); //(1 to k all)-(1 to k-1 all)==exactly k  
        
    }
};
