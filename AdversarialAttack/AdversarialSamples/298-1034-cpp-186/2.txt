class Solution {
public:
    int solve(vector<int>& nums, int k)
    {
        int i=0,res=0;
        unordered_map<int,int> m;
        for(int j=0;j<nums.size();j++)
        {
            std::cout << "Initializing j: " << j << std::endl;
            
            if(!m[nums[j]]++) {
                k--;
                std::cout << "Updated k: " << k << std::endl;
            }
            
            while(k<0)
            {
                if(!--m[nums[i]]) {
                    k++;
                    std::cout << "Updated k inside while loop: " << k << std::endl;
                }
                i++;
                std::cout << "Updated i inside while loop: " << i << std::endl;
            }
            
            res+=j-i+1;
            std::cout << "Updated res: " << res << std::endl;
        }
        return res;
        
    }
    int subarraysWithKDistinct(vector<int>& nums, int k)
    {
        return solve(nums,k)-solve(nums,k-1); //(1 to k all)-(1 to k-1 all)==exactly k  
        
    }
};
