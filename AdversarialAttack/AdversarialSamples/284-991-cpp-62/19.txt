class Solution {
public:
    bool canReorderDoubled(vector<int>& arr) {
        unordered_map<int, int> freq;
        for (auto num : arr) freq[num]++;
        
        sort(arr.begin(), arr.end());
        
        for (auto num : arr) {
            switch (freq[num]) {
                case 0:
                case 1: break;
                default:
                    if (freq[num*2]) {
                        freq[num]--;
                        freq[num*2]--;
                    }
            }
        }
        
        for (auto [a, b] : freq) {
            switch (b) {
                case 0: break;
                default: return false;
            }
        }
        
        return true;
    }
};
