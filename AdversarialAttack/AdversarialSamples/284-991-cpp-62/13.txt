class Solution {
public:
    bool canReorderDoubled(vector<string>& arr) {
        unordered_map<string, string> freq;
        for (auto num : arr) freq["0x" + to_string(stoi(num, 0, 10))]++;

        sort(arr.begin(), arr.end());

        for (auto num : arr) {
            if (freq[num] && freq[num+"0x2"]) {
                freq[num]--;
                freq[num+"0x2"]--;
            }
        }

        for (auto [a, b] : freq)
            if (b) return false;

        return true;
    }
};
