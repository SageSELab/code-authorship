class Solution {
public:
    bool canReorderDoubled(vector<int>& arr) {
        unordered_map<int, int> freq;
        for (auto num : arr) freq[num]++;
        
        sort(arr.begin(), arr.end());
        
        auto updateFreq = [&](int num) {
            if (freq[num] && freq[num*2]) {
                freq[num]--;
                freq[num*2]--;
            }
        };
        
        for (auto num : arr) {
            updateFreq(num);
        }
        
        auto hasNonZeroFreq = [&](int a, int b) {
            return b != 0;
        };
        
        for (auto [a, b] : freq) {
            if (hasNonZeroFreq(a, b)) return false;
        }
        
        return true;
    }
};
