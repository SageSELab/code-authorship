class Solution {
public:
    bool canReorderDoubled(vector<int>& arr) {
        unordered_map<int, int> freq;
        for (auto num : arr) freq[num += 1];
        
        sort(arr.begin(), arr.end());
        
        for (auto num : arr) {
            if (freq[num] && freq[num * 2]) {
                freq[num] -= 1;
                freq[num * 2] -= 1;
            }
        }
        
        for (auto [a, b] : freq)
            if (b) return false;
        
        return true;
    }
};
