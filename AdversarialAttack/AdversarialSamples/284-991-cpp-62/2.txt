class Solution {
public:
    bool canReorderDoubled(vector<int>& arr) {
        unordered_map<int, int> freq;
        for (auto num : arr) {
            freq[num]++;
            cout << "Initialized freq[" << num << "] = " << freq[num] << endl;
        }
        
        sort(arr.begin(), arr.end());
        cout << "Sorted arr vector" << endl;
        
        for (auto num : arr) {
            if (freq[num] && freq[num*2]) {
                freq[num]--;
                cout << "Modified freq[" << num << "] = " << freq[num] << endl;
                freq[num*2]--;
                cout << "Modified freq[" << num*2 << "] = " << freq[num*2] << endl;
            }
        }
        
        for (auto [a, b] : freq) {
            if (b) {
                cout << "Returned false at freq[" << a << "] = " << b << endl;
                return false;
            }
        }
        
        return true;
    }
};
