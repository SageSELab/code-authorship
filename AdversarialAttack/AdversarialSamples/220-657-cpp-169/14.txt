class Solution {
public:
    bool judgeCircle(string moves);
};
/*******************************************************/
bool Solution::judgeCircle(string moves) {
    if (moves.size() % 2 != 0) {
        return false;
    }
    int i, size = moves.size();
    vector<short> v(4,0); /* 85 -> 0, 68 -> 1, 76 -> 2, 82 -> 3 */
    for (i = 0; i < size; ++i) {
        if (moves[i] == 85) {
            ++v[0];
        } else if (moves[i] == 68) {
            ++v[1];
        } else if (moves[i] == 76) {
            ++v[2];
        } else if (moves[i] == 82) {
            ++v[3];
        }
    }
    if (v[0] == v[1] && v[2] == v[3]) {
        return true;
    }
    return false;
}
