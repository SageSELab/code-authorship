class Solution {
public:
    bool judgeCircle(string moves);
};
/*******************************************************/
bool Solution::judgeCircle(string moves) {
    if (moves.size() % 2 != 0) {
        return false;
    }
    int i, size = moves.size();
    vector<short> v(4,0); /* U -> 0, D-> 1, L -> 2, R -> 3 */
    cout << "Initialized v = {0, 0, 0, 0}" << endl;
  
    for (i = 0; i < size; ++i) {
        cout << "Current move: " << moves[i] << endl;
        if (moves[i] == 'U') {
            ++v[0];
            cout << "v[0] = " << v[0] << endl;
        } else if (moves[i] == 'D') {
            ++v[1];
            cout << "v[1] = " << v[1] << endl;
        } else if (moves[i] == 'L') {
            ++v[2];
            cout << "v[2] = " << v[2] << endl;
        } else if (moves[i] == 'R') {
            ++v[3];
            cout << "v[3] = " << v[3] << endl;
        }
    }
    if (v[0] == v[1] && v[2] == v[3]) {
        return true;
    }
    return false;
}
