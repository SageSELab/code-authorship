class Solution {
public:
    bool judgeCircle(string moves);
private:
    bool checkMoveCount(const vector<int>& movesCount);
};

/*******************************************************/
bool Solution::judgeCircle(string moves) {
    if (moves.size() % 2 != 0) {
        return false;
    }
    vector<int> v(4, 0); /* U -> 0, D-> 1, L -> 2, R -> 3 */
    for (int i = 0; i < moves.size(); ++i) {
        if (moves[i] == 'U') {
            ++v[0];
        } else if (moves[i] == 'D') {
            ++v[1];
        } else if (moves[i] == 'L') {
            ++v[2];
        } else if (moves[i] == 'R') {
            ++v[3];
        }
    }
    return checkMoveCount(v);
}

bool Solution::checkMoveCount(const vector<int>& movesCount) {
    return movesCount[0] == movesCount[1] && movesCount[2] == movesCount[3];
}
