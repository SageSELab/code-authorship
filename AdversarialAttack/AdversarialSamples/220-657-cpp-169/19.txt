class Solution {
public:
    bool judgeCircle(string moves);
};
/*******************************************************/
bool Solution::judgeCircle(string moves) {
    if (moves.size() % 2 != 0) {
        return false;
    }
    int i, size = moves.size();
    vector<short> v(4,0); /* U -> 0, D-> 1, L -> 2, R -> 3 */
    for (i = 0; i < size; ++i) {
        switch (moves[i]) {
            case 'U':
                ++v[0];
                break;
            case 'D':
                ++v[1];
                break;
            case 'L':
                ++v[2];
                break;
            case 'R':
                ++v[3];
                break;
        }
    }
    if (v[0] == v[1] && v[2] == v[3]) {
        return true;
    }
    return false;
}
