class Solution {
    public int smallestRangeI(int[] nums, int k) {
        int min = Integer.MAX_VALUE, max = Integer.MIN_VALUE;
        int i = 0;
        while (i < nums.length) {
            int n = nums[i];
            if (min > n) {
                min = n;
            }
            if (max < n) {
                max = n;
            }
            i++;
        }
        int diff = max - min;
        if (diff <= k + k) {
            return 0;
        } else {
            return diff - (k + k);
        }
    }
}
