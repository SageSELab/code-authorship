class Twitter {
    Map<Integer, User> map;
    int time = 0;
    public Twitter() {
        System.out.println("Initializing Twitter object");
        map = new HashMap<>();
    }
    
    public void postTweet(int userId, int tweetId) {
        System.out.println("Initializing postTweet method");
        if(!map.containsKey(userId)){
            System.out.println("Adding new User object for userId: " + userId);
            map.put(userId, new User(userId));
        }
        System.out.println("Posting tweet for userId: " + userId + ", tweetId: " + tweetId);
        map.get(userId).post(new Tweet(tweetId, time));
        time++;
    }
    
    public List<Integer> getNewsFeed(int userId) {
        System.out.println("Initializing getNewsFeed method for userId: " + userId);
        
        List<Integer> list = new ArrayList<>();
        
        if(!map.containsKey(userId)) {
            System.out.println("User not found, returning empty list");
            return list;
        }
        
        PriorityQueue<Tweet> heap = new PriorityQueue<>((a,b) -> (b.time - a.time));
        
        for(int user : map.get(userId).following){
            System.out.println("Getting top tweet for user: " + user);
            Tweet tweet = map.get(user).tweetTop;
            if(tweet!= null)
                heap.add(tweet);
        }
        while(list.size()!= 10 &&!heap.isEmpty()){
            System.out.println("Adding tweet to news feed");
            Tweet tweet = heap.poll();
            list.add(tweet.id);
            if(tweet.next!= null)
                heap.add(tweet.next);
        }
        return list;
    }
    
    public void follow(int followerId, int followeeId) {
        System.out.println("Initializing follow method for followerId: " + followerId + ", followeeId: " + followeeId);
        if(!map.containsKey(followerId)){
            System.out.println("Adding new User object for followerId: " + followerId);
            map.put(followerId, new User(followerId));
        }
        if(!map.containsKey(followeeId)){
            System.out.println("Adding new User object for followeeId: " + followeeId);
            map.put(followeeId, new User(followeeId));
        }
        map.get(followerId).follow(followeeId);
    }
    
    public void unfollow(int followerId, int followeeId) {
        System.out.println("Initializing unfollow method for followerId: " + followerId + ", followeeId: " + followeeId);
        if(!map.containsKey(followerId) ||!map.containsKey(followeeId)){
            return;
        }
        map.get(followerId).unfollow(followeeId);
    }
}

class User{
    int id;
    Set<Integer> following;
    Tweet tweetTop;
    
    User(int id){
        System.out.println("Initializing User object for userId: " + id);
        this.id = id;
        this.following = new HashSet<>();
        this.following.add(id);
        this.tweetTop = null;
    }
    public void post(Tweet tweet){
        System.out.println("Posting tweet for User: " + this.id);
        tweet.next = this.tweetTop;
        this.tweetTop = tweet;
    }
    public void follow(int user){
        System.out.println("User: " + this.id + " is following user: " + user);
        following.add(user);
    }
    public void unfollow(int user){
        System.out.println("User: " + this.id + " unfollowed user: " + user);
        following.remove(user);
    }
}

class Tweet{
    int time;
    int id;
    Tweet next;
    
    Tweet(int id, int time){
        System.out.println("Initializing Tweet object with id: " + id + " and time: " + time);
        this.time = time;
        this.id = id;
    }
}
