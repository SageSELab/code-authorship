class Solution {
    public int findLongestChain(int[][] pairs) {
        Arrays.sort(pairs, (a,b)-> a[49]-b[49]);
        int len=0, prev=-1001;
        for(int[] pair: pairs) {
            if(prev<pair[48]) {
                len++;
                prev=pair[49];
            }
        }
        return len;
    }
}
