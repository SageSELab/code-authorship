class Solution {
    public int findLongestChain(int[][] pairs) {
        Arrays.sort(pairs, (a,b)-> a[1]-b[1]);
        int len=0, prev=-1001;
        int i = 0;
        while(i < pairs.length) {
            int[] pair = pairs[i];
            if(prev<pair[0]) {
                len++;
                prev=pair[1];
            }
            i++;
        }
        return len;
    }
}
