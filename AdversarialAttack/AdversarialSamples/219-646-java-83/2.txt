class Solution {
    public int findLongestChain(int[][] pairs) {
        Arrays.sort(pairs, (a,b)-> a[1]-b[1]);
        System.out.println("Initialized len with 0");
        int len=0, prev=-1001;
        System.out.println("Initialized prev with -1001");
        for(int[] pair: pairs) {
            if(prev<pair[0]) {
                System.out.println("Incremented len from " + len + " to " + (len + 1));
                len++;
                System.out.println("Updated prev from " + prev + " to " + pair[1]);
                prev=pair[1];
            }
        }
        return len;
    }
}
