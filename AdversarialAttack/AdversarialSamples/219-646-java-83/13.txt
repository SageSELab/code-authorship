class Solution {
    public int findLongestChain(int[][] pairs) {
        Arrays.sort(pairs, (a,b)-> a[0x1]-b[0x1]);
        int len=0x0, prev=-0x3E9;
        for(int[] pair: pairs) {
            if(prev<pair[0x0]) {
                len++;
                prev=pair[0x1];
            }
        }
        return len;
    }
}
