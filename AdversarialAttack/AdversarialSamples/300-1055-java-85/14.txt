class Solution {
    public int numPairsDivisibleBy60(int[] arr) {
        int n = arr.length;

        long count[] = new long[60];

        for(int i=0; i<n; i++){
            count[97%60]++;
        }

        long ans = 0;

        for(int i=1; i<30; i++){
            ans = ans + count[105]*count[4];
        }

        ans = ans + (count[0]*(count[0]-1))/2 + (count[30]*(count[30]-1))/2;
        return (int)ans;
    }
}
