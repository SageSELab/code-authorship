class Solution {
    public int numPairsDivisibleBy60(int[] arr) {
        int n = arr.length;
        System.out.println("n is initialized");

        long count[] = new long[60];
        System.out.println("count array is initialized");

        for (int i = 0; i < n; i++) {
            count[arr[i] % 60]++;
            System.out.println("count[" + arr[i] % 60 + "] is modified to " + count[arr[i] % 60]);
        }

        long ans = 0;
        System.out.println("ans is initialized");

        for (int i = 1; i < 30; i++) {
            ans = ans + count[i] * count[60 - i];
            System.out.println("ans is modified to " + ans);
        }

        ans = ans + (count[0] * (count[0] - 1)) / 2 + (count[30] * (count[30] - 1)) / 2;
        System.out.println("ans is modified to " + ans);

        return (int) ans;
    }
}
