class Solution {
    public int numPairsDivisibleBy60(int[] arr) {
        int n = arr.length;

        long count[] = new long[0x3c];

        for(int i=0; i<n; i++){
            count[arr[i]%0x3c]++;
        }

        long ans = 0;

        for(int i=0x1; i<0x1e; i++){
            ans = ans + count[i]*count[0x3c-i];
        }

        ans = ans + (count[0]*(count[0]-1))/2 + (count[0x1e]*(count[0x1e]-1))/2;
        return (int)ans;
    }
}
