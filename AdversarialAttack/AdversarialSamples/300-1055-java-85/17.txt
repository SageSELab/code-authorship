class Solution {
    public int numPairsDivisibleBy60(int[] arr) {
        int n = arr.length;

        long count[] = new long[60];

        int i = 0;
        while(i < n){
            count[arr[i]%60]++;
            i++;
        }

        long ans = 0;

        int i = 1;
        while(i < 30){
            ans = ans + count[i]*count[60-i];
            i++;
        }

        ans = ans + (count[0]*(count[0]-1))/2 + (count[30]*(count[30]-1))/2;
        return (int)ans;
    }
}
