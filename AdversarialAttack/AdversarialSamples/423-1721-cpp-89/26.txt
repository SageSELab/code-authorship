class Solution {
public:
    int minOperationsMaxProfit(vector<int>& customers, int boardingCost, int runningCost) {
        int calculateProfit(vector<int>& customers, int boardingCost, int runningCost, int& number, int& waiting, int& boarded) {
            waiting += customers[number];
            boarded += min(4, waiting);
            int currProfit = boarded * boardingCost - number * runningCost;
            return currProfit;
        }

        int profit = INT_MIN;
        int optimal = -1;
        int waiting = 0;
        int number = 1;
        int boarded = 0;

        for (int i = 0; i < customers.size(); ++i) {
            int currProfit = calculateProfit(customers, boardingCost, runningCost, number, waiting, boarded);
            if (currProfit > profit) {
                profit = currProfit;
                optimal = number;
            }
            waiting -= min(4, waiting);
            number += 1;
        }

        while (waiting > 0) {
            int currProfit = calculateProfit(customers, boardingCost, runningCost, number, waiting, boarded);
            if (currProfit > profit) {
                profit = currProfit;
                optimal = number;
            }
            waiting -= min(4, waiting);
            number += 1;
        }

        return profit > 0 ? optimal : -1;
    }
};
