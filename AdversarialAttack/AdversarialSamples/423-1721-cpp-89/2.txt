class Solution {
public:
    int minOperationsMaxProfit(vector<int>& customers, int boardingCost, int runningCost) {
        // Intution
        // The idea in here is really very simple. We will maintain the variable to keep track of baorded customers and also we will keep the number of rotations made till now. At any instance we can stop the gondola so we will at every position check if the profit is maximum or not. If the profit at the index is max then we will change the current index to the number which we obtained now. We need to keep track of waiting number of customers and every time we will board a minimum of waiting and the board and the current profit earned will be the number of persons enjoyed the rotation and the number of times wheel was rotated for running the gondon.

        
        // This is used to store max profit earned till now

        int profit = INT_MIN;
        std::cout << "Initialized profit: " << profit << std::endl;

        // This is used to store the optimal number of moves

        int optimal = -1;
        std::cout << "Initialized optimal: " << optimal << std::endl;

        // Variable to store number of waiting customers

        int waiting = 0;
        std::cout << "Initialized waiting: " << waiting << std::endl;

        // Variable to stores number of rotations made
        int number = 1;
        std::cout << "Initialized number: " << number << std::endl;

        // Vairable to store boarded number of customers
        int boarded = 0;
        std::cout << "Initialized boarded: " << boarded << std::endl;

        // Looping and checking if at the current turn we can obtain max answer
        for(auto &cust : customers){

            waiting += cust;
            std::cout << "Updated waiting: " << waiting << std::endl;

            boarded += min(4, waiting);
            std::cout << "Updated boarded: " << boarded << std::endl;

            int currProfit = boarded * boardingCost - number * runningCost;
            std::cout << "Calculated currProfit: " << currProfit << std::endl;

            if(currProfit > profit){
                profit = currProfit;
                std::cout << "Updated profit: " << profit << std::endl;
                optimal = number;
                std::cout << "Updated optimal: " << optimal << std::endl;
            }

            waiting -= min(4, waiting);
            std::cout << "Updated waiting (after boarding): " << waiting << std::endl;

            number += 1;
            std::cout << "Updated number: " << number << std::endl;
        }

        // Also, checking for the waiting customers
        while(waiting > 0){
            boarded += min(4, waiting);
            std::cout << "Updated boarded (while waiting): " << boarded << std::endl;

            int currProfit = boarded * boardingCost - number * runningCost;
            std::cout << "Calculated currProfit (while waiting): " << currProfit << std::endl;

            if(currProfit > profit){
                profit = currProfit;
                std::cout << "Updated profit (while waiting): " << profit << std::endl;
                optimal = number;
                std::cout << "Updated optimal (while waiting): " << optimal << std::endl;
            }

            waiting -= min(4, waiting);
            std::cout << "Updated waiting (while waiting): " << waiting << std::endl;

            number += 1;
            std::cout << "Updated number (while waiting): " << number << std::endl;
        }

        return profit > 0? optimal : -1;

    }
};
