class RandomizedCollection {
public:
    
    vector<int>v;
    unordered_map<int,vector<int>>m;
    
    RandomizedCollection() {
        
    }
    
    bool insert(int val) {
        m[val].push_back(v.size());
        v.push_back(val);
        
        return m[val].size()==1;
    }
    
    bool remove(int val) {
        
        if(m[val].size()==0)return false;
        
        int idx=m[val].back();
        
        v[idx]=v.back();
        int ele=v[idx];
        
        int i=0;
        while(i<m[ele].size()){
            if(m[ele][i]==v.size()-1){
                m[ele][i] = idx;
                break;
            } 
            i++;
        }
        
        v.pop_back();
        
        m[val].pop_back();
        return true;
        
        
    }
    
    int getRandom() {
       
        int n=rand()%v.size();
        return v[n];
    }
};
