class RandomizedCollection {
public:
    
    vector<int>v;
    unordered_map<int,vector<int>>m;
    
    RandomizedCollection() {
        cout << "RandomizedCollection constructor called." << endl;
    }
    
    bool insert(int val) {
        cout << "Inserting value: " << val << endl;
        m[val].push_back(v.size());
        v.push_back(val);
        
        return m[val].size()==1;
    }
    
    bool remove(int val) {
        cout << "Removing value: " << val << endl;
        
        if(m[val].size()==0) {
            cout << "Value not found for removal." << endl;
            return false;
        }
        
        int idx = m[val].back();
        cout << "Index to be removed: " << idx << endl;
        
        v[idx] = v.back();
        int ele = v[idx];
        
        for(int i = 0; i < m[ele].size(); i++){
            if(m[ele][i] == v.size()-1){
                m[ele][i] = idx;
                break;
            } 
        }
        
        v.pop_back();
        m[val].pop_back();
        
        return true;
    }

    int getRandom() {
        cout << "Generating random number." << endl;
        
        int n = rand()%v.size();
        
        return v[n];
    }
};
