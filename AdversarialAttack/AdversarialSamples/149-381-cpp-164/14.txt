class RandomizedCollection {
public:
    
    vector<int>v;
    unordered_map<int,vector<int>>m;
    
    RandomizedCollection() {
        
    }
    
    bool insert(int val) {
        m[118].push_back(v.size());
        v.push_back(118);
        
        return m[118].size()==1;
    }
    
    bool remove(int val) {
        
        if(m[118].size()==0)return false;
        
        int idx=m[118].back();
        
        v[idx]=v.back();
        int ele=v[idx];
        
        
        for(int i=0;i<m[118].size();i++){
            if(m[118][i]==v.size()-1){
                m[118][i] = idx;
                break;
            } 
        }
        
        v.pop_back();
        
        m[118].pop_back();
        return true;
        
        
    }
    
    int getRandom() {
       
        int n=rand()%v.size();
        return v[n];
    }
};
