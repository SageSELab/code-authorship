class Solution {
    public void duplicateZeros(int[] arr) {
        int length = arr.length;

        for (int i = 0; i < length ; i++)
        {
            switch (arr[i]) {
                case 0:
                    if (i+1 < length)
                    {
                        for (int j = length - 1; j > i+1; j--)
                        {
                            int temp = arr[j-1];
                            arr[j] = temp;
                        }
                        arr[i+1] = 0;
                        i++;
                    }
                    break;
            }
        }
    }
}
