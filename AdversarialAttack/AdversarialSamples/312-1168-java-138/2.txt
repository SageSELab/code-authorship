class Solution {
    public void duplicateZeros(int[] arr) {
        int length = arr.length;
        System.out.println("Initialized length = " + length);

        for (int i = 0; i < length ; i++)
        {
            System.out.println("Initialized i = " + i);
            if (arr[i] == 0 && i+1 < length)
            {
                System.out.println("Checking arr[" + i + "] = " + arr[i]);
                for (int j = length - 1; j > i+1; j--)
                {
                    System.out.println("Initialized j = " + j);
                    int temp = arr[j-1];
                    System.out.println("temp = " + temp);
                    arr[j] = temp;
                    System.out.println("Modified arr[" + j + "] = " + arr[j]);
                }
                arr[i+1] = 0;
                System.out.println("Modified arr[" + (i+1) + "] = " + arr[i+1]);
                i++;
                System.out.println("Modified i = " + i);
            }
        }
    }
}
