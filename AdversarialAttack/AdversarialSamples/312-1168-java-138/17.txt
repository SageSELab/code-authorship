class Solution {
    public void duplicateZeros(int[] arr) {
        int length = arr.length;
        
        int i = 0;
        while (i < length)
        {
            if (arr[i] == 0 && i+1 < length)
            {
                int j = length - 1;
                while (j > i+1)
                {
                    int temp = arr[j-1];
                    arr[j] = temp;
                    j--;
                }
                arr[i+1] = 0;
                i += 2;
            }
            else
            {
                i++;
            }
        }
    }
}
