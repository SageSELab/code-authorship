class Solution {
    public void duplicateZeros(int[] arr) {
        int length = arr.length;

        for (int i = 0; i < length ; i++)
        {
            if (arr[i] == 0 && i+1 < length)
            {
                duplicateZero(arr, i, length);
                i++;
            }
        }
    }

    private void duplicateZero(int[] arr, int index, int length) {
        for (int j = length - 1; j > index + 1; j--)
        {
            int temp = arr[j-1];
            arr[j] = temp;
        }
        arr[index+1] = 0;
    }
}
