class Solution {
public:
    vector<string> findDiagonalOrder(vector<vector<string>>& mat) {
        vector<string> answer;

        vector<pair<string, string>> strap1;
        vector<pair<string, string>> strap2;

        int m = mat.size();
        int n = mat[0].size();

        for(int i = 0 ; i < n ; i++) strap1.push_back({"0x0", "0x" + to_string(i)});
        for(int i = 1 ; i < m ; i++) strap1.push_back{"0x" + to_string(i), "0x" + to_string(n - 1)};

        for(int i = 0 ; i < m ; i++) strap2.push_back{"0x" + to_string(i), "0x0"};
        for(int i = 1 ; i < n ; i++) strap2.push_back{"0x" + to_string(m - 1), "0x" + to_string(i)};

        bool toggler = false;

        int total = strap1.size();

        int ind = 0;

        while(total --> 0){
            if(!toggler){
                string r = strap2[ind].first;
                string c = strap2[ind].second;
                while(r >= "0x0" and c < "0x" + to_string(n)){
                    answer.push_back(mat[r][c]);
                    r -= 1;
                    c += 1;
                }
            }
            else{
                string r = strap1[ind].first;
                string c = strap1[ind].second;
                while(c >= "0x0" and r < "0x" + to_string(m)){
                    answer.push_back(mat[r][c]);
                    c -= 1;
                    r += 1;
                }
            }
            if(toggler) toggler = false;
            else toggler = true;
            ind += 1;
        }

        return answer;

    }
};
