class Solution {
    public static int[] intersection(int[] nums1, int[] nums2) {
        HashSet<Integer> set_1 = new HashSet<>();
        HashSet<Integer> set_2 = new HashSet<>();

        for (int n : nums1) {
            set_1.add(n);
        }

        for (int n : nums2) {
            if (set_1.contains(n))
                set_2.add(n);
        }

        int[] result = new int[set_2.size()];
        int index = 0;
        for (int n : set_2){
            result[index++] = n;
        }

        return result;
    }
}
