class Solution {
    public static int[] intersection(int[] nums1, int[] nums2) {
        HashSet<Integer> set1 = new HashSet<>();
        HashSet<Integer> set2 = new HashSet<>();

        for (int n : nums1) {
            set1.add(n);
        }

        for (int n : nums2) {
            if (set1.contains(n))
                set2.add(n);
        }

        return getIntersectionArray(set2);
    }

    private static int[] getIntersectionArray(HashSet<Integer> set) {
        int[] result = new int[set.size()];
        int index = 0;
        for (int n : set) {
            result[index++] = n;
        }

        return result;
    }
}
