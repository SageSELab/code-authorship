class Solution {
       public static int[] intersection(int[] nums1, int[] nums2)
    {
        HashSet<Integer> set1 = new HashSet<>();
        HashSet<Integer> set2 = new HashSet<>();

        for (int n : nums1) {
            set1.add(n);
        }

        for (int n : nums2) {
            if (set2.contains(n))
                set1.add(n);
        }

        int [] result = new int[set1.size()];
            int index = 0;
        for (int n : set1){
            result[index++] = n;
        }
        
        return result;
    }

}
