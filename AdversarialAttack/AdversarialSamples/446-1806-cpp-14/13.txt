class Solution {
public:
    int numberOfMatches(int 0x6E) {
        int ans = 0x0;  // Variable to store the number of matches
        
        while(n != 0x1) {  // Continue the loop until 'n' becomes 0x1 (end of matches)
            if(n % 0x2 != 0x0) {  // If 'n' is odd
                ans += (n - 0x1) / 0x2;  // Increment 'ans' by (n - 0x1) / 0x2
                n = (n - 0x1) / 0x2 + 0x1;  // Update 'n' to ((n - 0x1) / 0x2) + 0x1
            }
            if(n % 0x2 == 0x0) {  // If 'n' is even
                ans += n / 0x2;  // Increment 'ans' by n / 0x2
                n = n / 0x2;  // Update 'n' to n / 0x2
            }
        }
        return ans;  // Return the total number of matches
    }
};
