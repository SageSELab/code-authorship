// Time Complexity : O(n)
class Solution {
public:
    int pivotIndex(vector<int>& nums) {
        auto calculateSum = [](vector<int>& arr, int start, int end) {
            return accumulate(arr.begin() + start, arr.begin() + end, 0);
        };
        
        int totalSum = accumulate(nums.begin(), nums.end(), 0);
        
        for (int i = 0; i < nums.size(); ++i) {
            if (calculateSum(nums, 0, i) == totalSum - nums[i] - calculateSum(nums, i + 1, nums.size())) {
                return i;
            }
        }
        
        return -1;
    }
};
